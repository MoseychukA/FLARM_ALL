📄 FlyRf_Lan.ino (минимальный старт)

Создай файл с именем папки проекта, например:

FlyRf_Lan_26_03_25_02.ino

И вставь туда:

#include <Arduino.h>

// ===== Forward declarations (чтобы избежать ошибок линковки) =====
void SystemSetup();
void SystemLoop();

// ===== Arduino entry points =====
void setup()
{
    Serial.begin(115200);
    delay(1000);

    Serial.println();
    Serial.println("=== FlyRf Receiver старт ===");

    SystemSetup();
}

void loop()
{
    SystemLoop();
}
📄 System.cpp (логика верхнего уровня)

Создай рядом файл System.cpp:

#include <Arduino.h>

// Заглушки — потом заменим на реальные модули
void Display_setup();
void Display_loop();

void RF_setup();
void RF_loop();

void LAN_setup();
void LAN_loop();

void SystemSetup()
{
    Serial.println("[System] Init...");

    Display_setup();
    RF_setup();
    LAN_setup();
}

void SystemLoop()
{
    RF_loop();
    LAN_loop();
    Display_loop();
}
📄 System.h
#pragma once

void SystemSetup();
void SystemLoop();
