Спасибо за детали по SPI и нужный UDP-порт. Вот как корректно инициализировать
 W5500 на ESP32-S3 для Arduino IDE, использовать hardware SPI, выставить ваши пины и UDP-порт 10110.

Пример LANRF.h
#ifndef LANRF_H
#define LANRF_H

#include <Ethernet.h>
#include <EthernetUdp.h>

// Размер буфера UDP
#define LAN_PACKET_BUFSIZE 256

// Пины SPI под ваш wiring
#define W5500_CS    10
#define W5500_SCLK  12
#define W5500_MISO  13
#define W5500_MOSI  11
#define W5500_RESET 9

extern EthernetUDP Lan_Udp;
extern char LANpacketBuffer[LAN_PACKET_BUFSIZE];

void LAN_setup();
void LAN_loop();
size_t Raw_Receive_UDP_LAN(uint8_t *buf, size_t maxlen);
void Raw_Transmit_UDP_LAN(const uint8_t *buf, size_t len, IPAddress dst_ip, uint16_t dst_port);
void LAN_fini();

#endif // LANRF_H
Пример LANRF.cpp
#include "LANRF.h"
#include <SPI.h>

// MAC-адрес для W5500 укажите уникальный!
byte mac[] = { 0xDE, 0xAD, 0xBE, 0xEF, 0xFE, 0xED };
IPAddress ip(192,168,1,177); // или ваш IP

EthernetUDP Lan_Udp;
char LANpacketBuffer[LAN_PACKET_BUFSIZE];

uint16_t LAN_UDP_PORT = 10110;

void LAN_setup() {
  // SPI begin c вашими пинами
  SPI.begin(W5500_SCLK, W5500_MISO, W5500_MOSI, W5500_CS);

  // Подтягиваем RESET — если нужно сбросить чип
  pinMode(W5500_RESET, OUTPUT);
  digitalWrite(W5500_RESET, LOW);
  delay(10);
  digitalWrite(W5500_RESET, HIGH);
  delay(100);

  Ethernet.init(W5500_CS);  // указать Chip Select ваш пин CS
  Ethernet.begin(mac, ip);
  Lan_Udp.begin(LAN_UDP_PORT);
  Serial.print("Ethernet UDP started at port: ");
  Serial.println(LAN_UDP_PORT);
}

void LAN_loop() {
  int pktSize = Lan_Udp.parsePacket();
  if (pktSize > 0 && pktSize < LAN_PACKET_BUFSIZE) {
    Lan_Udp.read(LANpacketBuffer, pktSize);
    LANpacketBuffer[pktSize] = '\0';
    Serial.print("LAN UDP Received: ");
    Serial.println(LANpacketBuffer); // вывод принятого
  }
}

size_t Raw_Receive_UDP_LAN(uint8_t *buf, size_t maxlen) {
  int pktSize = Lan_Udp.parsePacket();
  if (pktSize > 0 && pktSize < maxlen) {
    return Lan_Udp.read(buf, pktSize);
  }
  return 0;
}

void Raw_Transmit_UDP_LAN(const uint8_t *buf, size_t len, IPAddress dst_ip, uint16_t dst_port) {
  Lan_Udp.beginPacket(dst_ip, dst_port);
  Lan_Udp.write(buf, len);
  Lan_Udp.endPacket();
}

void LAN_fini() {
  Lan_Udp.stop();
  Ethernet.end();
}

  // Пример передачи по LAN UDP:
  const char *msg = "Test UDP LAN";
  IPAddress dst_ip(192,168,1,100); // замените на нужный целевой IP

Важно!
Все W5500 пины обязательно указывайте через SPI.begin(...), иначе библиотека будет использовать дефолтные.
Не забудьте корректно держать оба стека (WiFi и Ethernet) раздельно — объекты UDP разные.
Ethernet.begin может занять время — при первой загрузке подождите и проверьте, что IP получен корректно.

#include "LANRF.h"
#include <SPI.h>

byte mac[] = { 0xDE, 0xAD, 0xBE, 0xEF, 0xFE, 0xED };  // Любой уникальный
EthernetUDP Lan_Udp;
char LANpacketBuffer[LAN_PACKET_BUFSIZE];

void LAN_setup() {
  // Инициируем SPI с вашими пинами для W5500
  SPI.begin(W5500_SCLK, W5500_MISO, W5500_MOSI, W5500_CS);
  pinMode(W5500_RESET, OUTPUT);
  digitalWrite(W5500_RESET, LOW);
  delay(10);
  digitalWrite(W5500_RESET, HIGH);
  delay(100);

  Ethernet.init(W5500_CS);

  if (Ethernet.begin(mac) == 0) {
    Serial.println(F("Failed to configure Ethernet using DHCP"));
    // Можно указать статичный IP вручную через Ethernet.begin(mac, ip);
    while (true) delay(1000); // Остановиться, если критично
  } else {
    Serial.println(F("Ethernet configured via DHCP"));
    Serial.print(F("My IP: "));
    Serial.println(Ethernet.localIP());
  }
  Lan_Udp.begin(LAN_UDP_PORT);
  Serial.print(F("Ethernet UDP started at port: "));
  Serial.println(LAN_UDP_PORT);
}

void LAN_loop() {
  int pktSize = Lan_Udp.parsePacket();
  if (pktSize > 0 && pktSize < LAN_PACKET_BUFSIZE) {
    Lan_Udp.read(LANpacketBuffer, pktSize);
    LANpacketBuffer[pktSize] = '\0';
    Serial.print(F("LAN UDP Received: "));
    Serial.println(LANpacketBuffer);
  }
}

  int pktSize = Lan_Udp.parsePacket();
  if (pktSize > 0 && pktSize < maxlen) {
    return Lan_Udp.read(buf, pktSize);
  }
  return 0;
}
