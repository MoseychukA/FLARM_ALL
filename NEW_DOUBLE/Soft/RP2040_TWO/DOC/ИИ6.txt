Для полной расшифровки пакета, передачи данных в ESP32S3 и интеграции, необходимо выполнить следующие шаги:

1. Расшифровка и обработка пакета rm в decode_adsb_frame
Добавьте вызовы парсинга всех сопровождающих данных:

CPR (глобальная/локальная, even/odd)
Высота (Q=0/1)
TC=19: вертикальная скорость и баро‑гео разница
Виды пакетов (DF17/18/20/21/18/DF18)
Проверку и коррекцию CRC
Собственно — извлечение координат
Примерно:

bool decode_adsb_frame(const RawMessage &rm, DecodedFrame &df, const Config &cfg) {
    // Расшифровка
    memset(&df, 0, sizeof(df));
    df.lat = NAN;
    df.lon = NAN;
    strcpy(df.callsign, "");

    // 1. Проверка CRC и исправление
    CrcStatus cs = check_and_fix_crc(rm);
    if (cs == CRC_FAIL) return false;

    // 2. Расшифровка типа и ICAO
    int df_type = decode_df(rm);
    hex_icao(rm, df.icao);
    // 3. Извлечение типа пакета
    int tc = get_tc(rm);
    const uint8_t *me = &rm.payload[5]; // ME поле
    // 4. Расшифровка данных
    if (df_type == 17 || df_type == 18) {
        // Позиция + высота
        bool qbit = me[1] & 0x10;
        uint16_t ac13 = ((me[0] & 0x0F) << 9) | (me[1] << 1) | ((me[2] >> 7) & 1);
        df.altitude_m_baro = alt_from_ac13(ac13, qbit);

        // CPR поля
        uint32_t latc = ((me[2] & 0x7F) << 10) | (me[3] << 2) | (me[4] >> 6);
        uint32_t lonc = ((me[4] & 0x3F) << 12) | (me[5] << 4) | (me[6] >> 4);
        bool odd = me[6] & 0x04;

        // Хранение пар even/odd для глобальной или локальной распаковки
        CprStoreEntry *ce = cpr_get(df.icao);
        int oi = odd ? 1 : 0;
        ce->latc[oi] = latc;
        ce->lonc[oi] = lonc;
        ce->t_ms[oi] = millis();
        ce->surface = (tc >= 5 && tc <= 8); // surface пакет
        ce->valid[oi] = true;

        bool have_pair = ce->valid[0] && ce->valid[1] &&
                         (abs((int)ce->t_ms[0] - (int)ce->t_ms[1]) <= 10000);
        if (have_pair) {
            float lat, lon;
            if (cpr_decode_global(ce->latc[0], ce->lonc[0], ce->latc[1], ce->lonc[1], ce->surface, lat, lon)) {
                df.lat = lat;
                df.lon = lon;
            }
        } else {
            // Локальный расчет
            float lat, lon;
            if (cpr_decode_local(latc, lonc, ce->surface, cfg.ref_lat, cfg.ref_lon, lat, lon)) {
                df.lat = lat;
                df.lon = lon;
            }
        }

        // Высота (Q=1 — геоид, Q=0 — Gillham AC13)
        // в случае Q=0 уже исправлено в стороннем коде, он возвращает значение
    } else if (df_type == 19) {
        // TC=19 — Векторная скорость + vertical / baro-geo
        int subtype = (me[0] & 0x7);
        if (subtype == 1 || subtype == 2) {
            // Вектор скорости по земле
            int vew = ((me[1] & 0x7F) << 3) | (me[2] >> 5);
            int vns = ((me[2] & 0x1F) << 6) | (me[3] >> 2);
            bool sgn_ew = me[1] & 0x80;
            bool sgn_ns = me[3] & 0x02;
            int ew = vew - 1;
            int ns = vns - 1;
            float vx = (sgn_ew ? -ew : ew);
            float vy = (sgn_ns ? -ns : ns);
            float spd = sqrtf(vx*vx + vy*vy);
            df.speed_kmh = (int)round(spd * 1.852f);
            float trk = atan2f(vx, vy) * 180.0f / PI;
            if (trk < 0) trk += 360;
            df.track_deg = (int)round(trk);

            // Вертикальная скорость и баро-гео разность не реализованы в этом скелете — добавьте при необходимости
        } else if (subtype == 3 || subtype == 4) {
            // Курс и воздушная скорость
            int hdg_enc = ((me[1] & 0x7F) << 3) | (me[2] >> 5);
            bool hdg_valid = me[1] & 0x80;
            float hdg = hdg_valid ? (hdg_enc * 360.0f / 1024) : NAN;
            int as = ((me[2] & 0x1F) << 6) | (me[3] >> 2);
            int airspeed = as - 1;
            df.speed_kmh = (int)round(airspeed * 1.852f);
            df.track_deg = (int)round(hdg);
        }
    }

    // Другие пакеты
    return true;
}

Вот пример полной реализации функции преобразования Raw1090Packet в RawMessage и использование его в основном цикле.

Полный пример — исправленная версия
#include <Arduino.h>
#include <atomic>
#include "adsb_decoder.h" // при необходимости для decode_adsb_frame()

// Объявление внешнего массива (если не сделано)
extern std::atomic<Raw1090Packet*> latest_packets[3];

// Функция преобразования
RawMessage convertPacketToRawMessage(const Raw1090Packet &pkt) {
    RawMessage rm;

    // Размер в битах
    rm.buffer_len_bits = pkt.buffer_len_words32 * 32;

    // В байтах (окурс: используем только buffer в виде массива слов)
    rm.bytes = (rm.buffer_len_bits + 7) / 8;

    // Копируем buffer
    for (int i = 0; i < pkt.buffer_len_words32; i++) {
        rm.buffer[i] = pkt.buffer[i];
    }

    // Остальные поля
    rm.source_in = pkt.source_in;
    rm.sigs_dbm_in = pkt.sigs_dbm_in;
    rm.mlat_48mhz_64bit_counts = pkt.mlat_48mhz_64bit_counts;

    return rm;
}

void process_latest_packet(int sm_idx) {
    auto pkt_ptr = latest_packets[sm_idx].load(std::memory_order_relaxed);
    if (pkt_ptr != nullptr) {
        // Создаем копию
        Raw1090Packet pkt_copy;
        memcpy(&pkt_copy, &(*pkt_ptr), sizeof(Raw1090Packet));

        // Конвертация в RawMessage
        RawMessage rm = convertPacketToRawMessage(pkt_copy);

        // Расшифровка
        DecodedFrame df;
        if (decode_adsb_frame(rm, df, g_cfg)) {
            // Формируем FlightOutput
            FlightOutput fo = { 0 };
            fo.addr = df.icao;
            fo.Squawk = df.squawk;
            memcpy(fo.flight, df.callsign, sizeof(fo.flight));
            fo.altitude = df.altitude_m_geo;
            fo.pressure_altitude = df.altitude_m_baro;
            fo.speed = df.speed_kmh;
            fo.course = df.track_deg;
            fo.vert_rate = df.vert_rate_mpm;
            fo.latitude = df.lat;
            fo.longitude = df.lon;
            fo.seen = df.seen_time_ms;
            fo.timestamp = millis();
            fo.signal_source = 1;
            fo.aircraft_type = df.aircraft_type;

            // Передача ESP
            send_to_esp32(fo);
            // Вывод
            print_by_format(df, fo, (int)rssi_adc_last);
        }
    }
}

void setup() {
    Serial.begin(115200);
    while (!Serial && millis() < 5000);
    Serial.println("Main loop started");
}

void loop() {
    for (int sm_idx=0; sm_idx<3; sm_idx++) {
        process_latest_packet(sm_idx);
    }

void process_latest_packet(int sm_idx) {
    auto pkt_ptr = latest_packets[sm_idx].load(std::memory_order_relaxed);
    if (pkt_ptr != nullptr) {
        Raw1090Packet pkt_copy;
        memcpy(&pkt_copy, &(*pkt_ptr), sizeof(Raw1090Packet));

        // Расшифровка
        DecodedFrame df;
        if (decode_adsb_frame(convertPacketToRawMessage(pkt_copy), df, g_cfg)) {
            // Заполнить и передать расширенную структуру
            FlightOutput fo = { 0 };
            fo.addr = df.icao;
            // Остальные поля заполняются внутри decode_adsb_frame() или здесь
            send_to_esp32(fo);
            print_by_format(df, fo, (int)rssi_adc_last);
        }
    }
