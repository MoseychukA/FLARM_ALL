    uint16_t crc = 0xFFFF;
    for (size_t i = 0; i < len; i++) {
        crc ^= data[i];
        for (int j = 0; j < 8; j++) {
            if (crc & 1)
                crc = (crc >> 1) ^ 0xA001;
            else
                crc >>= 1;
        }
    }
    return crc;

CRC: реализован классический алгоритм CRC-16, используемый в ADS-B.

используя даже и нечетные сообщения для восстановления координат.

#include "decode.h"

uint16_t calculate_crc(const uint8_t *data, size_t len) {
    uint16_t crc = 0xFFFF;
    for (size_t i = 0; i < len; i++) {
        crc ^= data[i];
        for (int j = 0; j < 8; j++) {
            if (crc & 1)
                crc = (crc >> 1) ^ 0xA001;
            else
                crc >>= 1;
        }
    }
    return crc;
}

bool try_fix_one_bit(uint8_t *data, size_t len, uint16_t expected_crc) {
    for (size_t byte_idx = 0; byte_idx < len; byte_idx++) {
        for (int bit = 0; bit < 8; bit++) {
            data[byte_idx] ^= (1 << bit);
            if (calculate_crc(data, len) == expected_crc) {
                return true;
            }
            data[byte_idx] ^= (1 << bit); // восстановление бита
        }
    }
    return false;
}

uint16_t get_expected_crc(const uint8_t *data, size_t len) {
    if (len >= 2) {
        return (data[len - 2] << 8) | data[len - 1];
    }
    return 0xFFFF;
}

bool parse_raw_packet(const Raw1090Packet &raw, ADSBPacket &outPkt) {
    // Пример извлечения ICAO (6 байт начиная с первых двух слов)
    outPkt.icao = 0;
    if (raw.buffer_len_bits >= 48) {
        for (int i = 0; i < 6; i++) {
            uint8_t byte = (raw.buffer[i/4] >> ((i%4)*8)) & 0xFF;
            outPkt.icao = (outPkt.icao << 8) | byte;
        }
    }

    // В реальности — парсинг сообщения ADS-B
    // Тут – пример
    strcpy(outPkt.flight, "AB1234");
    outPkt.squawk = 7500;
    outPkt.altitude = 12000; // м
    outPkt.speed = 600;      // км/ч
    outPkt.track = 90;       // градусы
    outPkt.vert_rate = 0;

    // Время
    outPkt.seen_time = millis();

    // Игнорируем расчет широты и долготы (заглушки)
    outPkt.lat_msg = 0.0;
    outPkt.lon_msg = 0.0;

    return true;
}

",
                  pkt.icao, pkt.flight, pkt.lat_msg, pkt.lon_msg, pkt.altitude, pkt.speed);
}

// В вашем месте получения пакета
void process_received_packet(Raw1090Packet &rawPacket) {

    uint16_t crc_received = get_expected_crc((uint8_t*)rawPacket.buffer, rawPacket.buffer_len_bits / 8);
    uint8_t data_bytes[rawPacket.buffer_len_bits / 8];

            Serial.println("Ошибка исправлена");

            Serial.println("Не удалось исправить ошибку");
            return; // пропускаем пакет
        }

    Raw1090Packet temp_pkt = rawPacket;

    // Можно заменить buffer в temp_pkt на исправленные данные

void process_packet(Raw1090Packet &rawPacket) {
  // Получим CRC из пакета
  uint16_t crc_received = get_expected_crc((uint8_t*)rawPacket.buffer, rawPacket.buffer_len_bits / 8);
  
  // Создаем массив для данных для CRC
  uint8_t data_bytes[rawPacket.buffer_len_bits / 8];
  memcpy(data_bytes, rawPacket.buffer, sizeof(data_bytes));

  // Рассчитаем CRC
  uint16_t crc_calculated = calculate_crc(data_bytes, sizeof(data_bytes));

  // Проверка CRC
  if (crc_calculated != crc_received) {
    Serial.println("Ошибка CRC, попытка исправления...");
    if (try_fix_one_bit(data_bytes, sizeof(data_bytes), crc_received)) {
      Serial.println("Ошибка исправлена успешно");
      // Обновляем rawPacket данными с исправлением
      memcpy(rawPacket.buffer, data_bytes, sizeof(data_bytes));
    } else {
      Serial.println("Не удалось исправить ошибку");
      return; // выходим из функции
    }
  }

  // Создаём временной пакет для парсинга
  Raw1090Packet temp_pkt = rawPacket;

  // Обновляем байты из исправленных данных
  memcpy(temp_pkt.buffer, data_bytes, sizeof(data_bytes));

  // Распарсим в структуру
  ADSBPacket decodedPkt;
  if (parse_raw_packet(temp_pkt, decodedPkt)) {
    // Декодируем координаты
    decode_cpr((uint8_t*)temp_pkt.buffer, decodedPkt.lat_msg, decodedPkt.lon_msg);

    // Выводим информацию
    print_aircraft(decodedPkt);
  } else {
    Serial.println("Ошибка парсинга пакета");
  }
}

MiniADS_B.ino
#include <Arduino.h>
#include "decode.h"

// Имитация получения пакета (заглушка)
bool receive_packet(Raw1090Packet *pkt) {
    // Здесь должна быть ваша логика получения пакета
    // Для примера — создаст фиктивный пакет
    static uint32_t counter = 0;
    if (counter > 10) return false; // имитируем завершение
    counter++;

    // Заполняем тестовыми данными
    for (int i = 0; i < Raw1090Packet::kMaxPacketLenWords32; i++) {
        pkt->buffer[i] = i + counter;
    }
    pkt->buffer_len_bits = 8 * 12; // 12 слов
    return true;
}

void process_packet(Raw1090Packet &rawPacket) {
    uint16_t crc_received = get_expected_crc((uint8_t*)rawPacket.buffer, rawPacket.buffer_len_bits / 8);
  
    uint8_t data_bytes[rawPacket.buffer_len_bits / 8];
    memcpy(data_bytes, rawPacket.buffer, sizeof(data_bytes));

    uint16_t crc_calculated = calculate_crc(data_bytes, sizeof(data_bytes));

    if (crc_calculated != crc_received) {
        Serial.println("CRC mismatch, attempting correction...");
        if (try_fix_one_bit(data_bytes, sizeof(data_bytes), crc_received)) {
            Serial.println("Error corrected");
            memcpy(rawPacket.buffer, data_bytes, sizeof(data_bytes));
        } else {
            Serial.println("Uncorrectable error");
            return;
        }
    }

    // Копируем исправленные данные
    Raw1090Packet temp_pkt = rawPacket;
    memcpy(temp_pkt.buffer, data_bytes, sizeof(data_bytes));

    ADSBPacket decodedPkt;
    if (parse_raw_packet(temp_pkt, decodedPkt)) {
        decode_cpr((uint8_t*)temp_pkt.buffer, decodedPkt.lat_msg, decodedPkt.lon_msg);
        print_aircraft(decodedPkt);
    } else {
        Serial.println("Parsing error");
    }
}

void setup() {
  Serial.begin(115200);
  while (!Serial);
  Serial.println("Starting...");
}

void loop() {
  Raw1090Packet pkt;
  if (receive_packet(&pkt)) {
    process_packet(pkt);
  }
  delay(1000);
}

#include "decode.h"

uint16_t calculate_crc(const uint8_t *data, size_t len) {
    uint16_t crc = 0xFFFF;
    for (size_t i = 0; i < len; i++) {
        crc ^= data[i];
        for (int j = 0; j < 8; j++) {
            if (crc & 1)
                crc = (crc >> 1) ^ 0xA001;
            else
                crc >>= 1;
        }
    }
    return crc;
}

bool try_fix_one_bit(uint8_t *data, size_t len, uint16_t expected_crc) {
    for (size_t byte_idx = 0; byte_idx < len; byte_idx++) {
        for (int bit = 0; bit < 8; bit++) {
            data[byte_idx] ^= (1 << bit);
            if (calculate_crc(data, len) == expected_crc) {
                return true;
            }
            data[byte_idx] ^= (1 << bit);
        }
    }
    return false;
}

uint16_t get_expected_crc(const uint8_t *data, size_t len) {
    if (len >= 2) {
        return (data[len - 2] << 8) | data[len - 1];
    }
    return 0xFFFF;
}

bool parse_raw_packet(const Raw1090Packet &raw, ADSBPacket &outPkt) {
    // Пример: извлечь ICAO из первых 6 байт
    outPkt.icao = 0;
    if (raw.buffer_len_bits >= 48) {
        for (int i = 0; i < 6; i++) {
            uint8_t byte = (raw.buffer[i/4] >> ((i%4)*8)) & 0xFF;
            outPkt.icao = (outPkt.icao << 8) | byte;
        }
    }
    // Заглушки для других полей
    strcpy(outPkt.flight, "AB1234");
    outPkt.squawk = 7500;
    outPkt.altitude = 12000;
    outPkt.speed = 600;
    outPkt.track = 90;
    outPkt.vert_rate = 0;
    outPkt.lat_msg = 55.7558;
    outPkt.lon_msg = 37.6173;
    outPkt.seen_time = millis();
    return true;
}

Вот пример полной реализации функции finish_processing для файла adsbee.cpp. 
Она принимает уже обработанный Raw1090Packet и передает его в общий механизм обмена данными:

#include "shared.h"

void finish_processing(const Raw1090Packet &rx_packet, uint64_t icao_value) {

    parse_raw_packet(rx_packet, shared_packet);

    // Устанавливаем флаг, что есть новый пакет для main.ino
    new_packet_ready = true;
}

#include "shared.h"

void transfer_packet(const RawPacket &rx_packet) {
    size_t offset = 0;

    // Копируем buffer (обычно 56 или 112 байт, можно больше)
    size_t buffer_size = sizeof(rx_packet.buffer);
    if (buffer_size > sizeof(shared_packet_bytes))
        buffer_size = sizeof(shared_packet_bytes);
    memcpy(&shared_packet_bytes[offset], rx_packet.buffer, buffer_size);
    offset += buffer_size;

    // копируем другие поля (можно расширить при необходимости)
    memcpy(&shared_packet_bytes[offset], &rx_packet.buffer_len_bits, sizeof(rx_packet.buffer_len_bits));
    offset += sizeof(rx_packet.buffer_len_bits);

    memcpy(&shared_packet_bytes[offset], &rx_packet.source, sizeof(rx_packet.source));
    offset += sizeof(rx_packet.source);

    memcpy(&shared_packet_bytes[offset], &rx_packet.sigs_dbm, sizeof(rx_packet.sigs_dbm));
    offset += sizeof(rx_packet.sigs_dbm);

    memcpy(&shared_packet_bytes[offset], &rx_packet.sigq_db, sizeof(rx_packet.sigq_db));
    offset += sizeof(rx_packet.sigq_db);

    memcpy(&shared_packet_bytes[offset], &rx_packet.mlat_48mhz_64bit_counts, sizeof(rx_packet.mlat_48mhz_64bit_counts));
    offset += sizeof(rx_packet.mlat_48mhz_64bit_counts);

    // Запоминаем длину байтов
    shared_packet_bytes_len = offset;

    // Устанавливаем флаг
    new_packet_ready = true;
}

Понял вас. В вашем коде есть такой вызов:

Но это неправильно, потому что вы копируете sigq_db в shared_packet_bytes, а затем увеличиваете offset на размер buffer_len_bits, что — несоответствие и ошибка.
