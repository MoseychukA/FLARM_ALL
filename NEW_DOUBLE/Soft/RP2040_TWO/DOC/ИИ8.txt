Вот полный пример, который показывает, как передать содержимое rx_packet_[sm_index], 
добавляемое в очередь raw_1090_packet_queue, в .ino. 
Это реализуется через глобальную переменную, куда вы копируете пакет, 
и свойство флага о наличии нового пакета.

1. Объявление в shared.h
#ifndef SHARED_H
#define SHARED_H

#include "decode.h"

extern volatile bool raw_packet_available; // флаг наличия пакета
extern RawPacket shared_raw_packet;        // глобальный буфер пакета

#endif
2. Определение в shared.cpp
#include "shared.h"

volatile bool raw_packet_available = false;
RawPacket shared_raw_packet;
3. В adsbee.cpp — после raw_1090_packet_queue.Push(rx_packet_[sm_index])
#include "shared.h"

void save_packet_for_ino(const RawPacket& pkt) {
    memcpy(&shared_raw_packet, &pkt, sizeof(RawPacket));
    raw_packet_available = true;
}

// После получения пакета
save_packet_for_ino(rx_packet_[sm_index]);
4. В .ino — для вывода содержимого
#include "shared.h"

void dump_packet(const RawPacket &pkt) {
    Serial.println("Содержимое пакета:");
    uint16_t total_bytes = (pkt.buffer_len_bits + 7) / 8; // число байт
    uint8_t *data = (uint8_t*)pkt.buffer;
    for (uint16_t i = 0; i < total_bytes; i++) {
        if (i > 0) Serial.print(" ");
        if (data[i] < 16) Serial.print("0");
        Serial.print(data[i], HEX);
    }
    Serial.println();
}

void setup() {
    Serial.begin(115200);
    while (!Serial);
    Serial.println("Готово");
}

void loop() {
    if (raw_packet_available) {
        noInterrupts();
        RawPacket pkt = shared_raw_packet; // копируем
        raw_packet_available = false;
        interrupts();

        dump_packet(pkt); // выводим
    }
    delay(100);
}
Итог:
В adsbee.cpp вызывается save_packet_for_ino() после Push().
В .ino — при флаге raw_packet_available отображается содержимое пакета.
Размер выводимых данных — buffer_len_bits / 8.
