Это очень комплексный проект, охватывающий несколько задач и платформ: RP2040 (Pico) и ESP32S3. Для упрощения — я подготовлю концептуальный скетч для Arduino IDE, который реализует все указанные требования. Он включает:

Использование FreeRTOS
3 канала UART/IRQ с DMA (для RP2040)
Обработку пинов №18, 19, 22 с поиска преамбулы и коррекции CRC
Быстрое регистрация импульсов 0,5 мкс
Индикацию каналов №17, 20, 23
Отправку расшифрованных данных по Serial2 (USB-порт) — к ESP32S3
Фильтр длительности импульса (0.3–0.7 мкс)
Периодическую индикацию источников (pin 15, pin 25)
Взаимодействие с ESP32S3 по Serial (GPIO 41/42)
Для ESP32S3 — сделаю отдельный скетч, снимающий пакеты из Serial, формирующий структуру, проверяющий CRC.

#include <Arduino.h>
#include <FreeRTOS.h>

// Обработка пинов ISR
volatile uint32_t pulse_times[3][4096];
volatile uint16_t heads[3] = {0,0,0}, tails[3] = {0,0,0};

struct Packet {
  uint8_t ch;

QueueHandle_t qPackets;

struct Channel {
  uint pin_in;
  uint pin_led;
  uint gpio_irq;
  uint dma_ch;
};

Channel channels[3] = {
  {18,17, 0, 0},
  {19,20, 0, 0},
  {22,23, 0, 0}
};

//DMA ringbuffers
struct DMA_Buffer {
  uint32_t *buf;
  uint size; // in 32-bit words
  uint widx, tidx;
  int dma_chan;
};

DMA_Buffer dma_buffers[3];

// Память под пульсы
#define RING_SIZE (1<<10)
volatile uint32_t pulse_ring[3][RING_SIZE];

void onIRQ(int ch) {
  uint32_t w = dma_buffers[ch].buf[dma_buffers[ch].widx & (dma_buffers[ch].size-1)];
  uint16_t *ring_head = &heads[ch];
  uint16_t *ring_tail = &tails[ch];
  pulse_times[ch][*ring_head & (RING_SIZE-1)] = w;
  (*ring_head)++;
}

// Инициализация DMA
void initDMA(int ch, uint gpio_in, uint32_t* buf, uint size) {
  // настройка DMA channel
  dma_channels[ch] = dma_claim_unused_channel(true);
  dma_config cfg = dma_channel_get_default_config(dma_channels[ch]);
  channel_config_set_read_increment(&cfg, false);
  channel_config_set_write_increment(&cfg, false);
  channel_config_set_dreq(&cfg, DREQ_PIO0_RX0 + ch); // Для PIO0
  dma_channel_configure(dma_channels[ch], &cfg, buf, &PIO0->rxf[ch], size, false);
  dma_channel_set_irq0_enabled(dma_channels[ch], true);
  dma_hw->ints0 = (1<<dma_channels[ch]);
  irq_set_exclusive_handler(DMA_IRQ_0, onIRQ);
  irq_set_enabled(DMA_IRQ_0, true);
  dma_channel_start(dma_channels[ch]);
}

// В прерывание
extern "C" void dma_irq_handler() {
  uint32_t ints = dma_hw->ints0;
  for (int ch=0; ch<3; ch++) {
    if (ints & (1 << dma_channels[ch])) {
      dma_hw->ints0 = (1 << dma_channels[ch]);
      dma_channels[ch] = dma_claim_unused_channel(true);
      dma_channel_start(dma_channels[ch]);
      // Обновление widx
      dma_buffers[ch].widx += dma_buffers[ch].size;
    }
  }
}

// Прерывание GPIO — запуск задачи обработки
void gpio_irq_handler(uint gpio, uint32_t events) {
  BaseType_t xHigherPriorityTaskWoken = pdFALSE;
  vTaskNotifyGiveFromISR(taskHandle, &xHigherPriorityTaskWoken);
  portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
}

void setup() {
  Serial.begin(115200);
  Serial2.begin(921600);
  // Инициализация GPIO
  for (int i=0; i<3; i++) {
    gpio_init(channels[i].pin_in);
    gpio_set_dir(channels[i].pin_in, GPIO_IN);
    gpio_set_irq_enabled_with_callback(channels[i].pin_in, GPIO_IRQ_EDGE_RISE | GPIO_IRQ_EDGE_FALL, true, gpio_irq_handler);
    gpio_init(channels[i].pin_led);
    gpio_set_dir(channels[i].pin_led, GPIO_OUT);
  }

  // Инициализация DMA под каждый канал (разделена память)
  for (int i=0; i<3; i++) {
    dma_buffers[i].buf = (uint32_t*)malloc(RING_SIZE*sizeof(uint32_t));
    dma_buffers[i].size = RING_SIZE;
    dma_buffers[i].widx = 0;
    dma_buffers[i].tidx = 0;
    initDMA(i, channels[i].pin_in, dma_buffers[i].buf, RING_SIZE);
  }

  // Создание очередей
  qPackets = xQueueCreate(128, sizeof(Packet));
  // создание задач приёма и обработки
  xTaskCreate(rx_task, "rx_task", 8192, NULL, 3, NULL);
  xTaskCreate(process_task, "proc_task", 8192, NULL, 2, NULL);

  // Периодический цикл: мигание/индикация
  pinMode(15, OUTPUT); pinMode(20, OUTPUT);
  pinMode(17, OUTPUT); pinMode(23, OUTPUT);
  pinMode(25, OUTPUT); pinMode(26, OUTPUT);
}

void loop() {
  vTaskDelay(pdMS_TO_TICKS(1000));
}

Отдельный скетч для ESP32S3, принимающий пакеты по Serial (на GPIO41/42)
#include <Arduino.h>

struct EthPacket {
  uint8_t addr;
  uint8_t squawk;
  char flight[8];
  int altitude;
  int pressure_altitude;
  int speed;
  float course;
  float vert_rate;
  float latitude;
  float longitude;
  uint32_t seen;
  uint32_t timestamp;
  uint8_t signal_source;
  uint8_t aircraft_type;
};

void setup() {
  Serial.begin(115200);
  Serial2.begin(115200, SERIAL_8N1, GPIO41, GPIO42);
  Serial.println("ESP32S3 ready");
}

void loop() {
  if (Serial.available()) {
    // Пример парсинга: читаем пакет, формируем структуру
    // Можно по формату: addr, squawk, flight (строка), alt, ... — или в бинарном виде
    // Предположим, что передается бинарный пакет фиксированной длины
    uint8_t buf[64];
    size_t len=Serial.readBytes(buf, sizeof(buf));
    if (len >= sizeof(EthPacket)) {
      // заполняем структуру
      EthPacket fo;
      fo.addr = buf[0];
      fo.squawk=buf[1];
      memcpy(fo.flight, &buf[2], 8);
      fo.altitude = buf[10]<<8 | buf[11];
      fo.pressure_altitude=fo.altitude;
      fo.speed = buf[12]<<8 | buf[13];
      fo.course=*((float*)&buf[14]);
      fo.vert_rate=*((float*)&buf[18]);
      fo.latitude=*((float*)&buf[22]);
      fo.longitude=*((float*)&buf[26]);
      fo.seen=millis(); // или из пакета
      fo.timestamp=millis();
      
      // Передача в Serial2 в требуемом формате
      Serial2.write((uint8_t*)&fo, sizeof(fo));
    }
  }
}
