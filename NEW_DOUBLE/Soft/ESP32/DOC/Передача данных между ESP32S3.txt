/* =====================  ESP32-S3 RS485 RECEIVER  ========================
 *  - Arduino IDE
 *  - Serial1: TX=GPIO39, RX=GPIO38
 *  - RS485 DE/RE: GPIO40 (LOW=RX)
 *  - FreeRTOS task pinned to Core 0
 *  - Разбор протокола, CRC32, размещение в Container/ThisAircraft
 *  - Вывод принятых данных в USB Serial
 * ======================================================================= */

#include <Arduino.h>

#define MAX_TRACKING_OBJECTS 8

typedef struct UFO {
  time_t    timestamp;
  uint32_t  addr;
  uint8_t   addr_type;
  float     latitude;
  float     longitude;
  float     old_latitude;
  float     old_longitude;
  float     altitude;
  float     pressure_altitude;
  float     course;
  float     speed;
  uint8_t   aircraft_type;
  char      flight[16];
  int       vert_rate;
  int       Squawk;
  time_t    timemsg;

  float     vs;

  float     geoid_separation;
  uint16_t  hdop;
  int8_t    rssi;

  float     distance;
  float     bearing;

  uint8_t   signal_source;
  time_t    seen;
  uint8_t   hour_msg;
  uint8_t   min_msg;
  uint16_t  delay_time_msg;
  uint8_t   callsign[8];
} ufo_t;

ufo_t Container[MAX_TRACKING_OBJECTS];
ufo_t ThisAircraft;

// Должно совпадать с источником
typedef struct __attribute__((packed)) {
  bool     new_flag_M;
  uint8_t  new_buttton_M;
  bool     setMessageRead_M;
  bool     MessageRead_M;
  bool     SOS_Sprite_on_off_M;
  bool     SOS_View_on_off_M;
  bool     new_SOS_flag_M;
  bool     confirm_message_M;
  char     msg_resp_M[60];
  bool     isValidGNSS_M;
  uint8_t  FLYRF_MODE_TEST_M;
  uint16_t analog_u16;
} tx_flags_t;

// UART/RS485
static const int RS485_DE_PIN = 40;  // LOW = RX
static const int UART_TX_PIN  = 39;
static const int UART_RX_PIN  = 38;
static const uint32_t BAUD    = 921600;

// Протокол кадра
#pragma pack(push,1)
typedef struct {
  uint8_t  magic[2];   // 'U','F'
  uint8_t  version;    // 1
  uint8_t  type;       // 1..N
  uint16_t seq;
  uint16_t payload_len;
  uint16_t reserved;
} frame_hdr_t;
#pragma pack(pop)

enum : uint8_t {
  PKT_UFO_ITEM = 1,   // [uint8_t index][ufo_t]
  PKT_THIS_AC  = 2,   // [ufo_t]
  PKT_FLAGS    = 3    // [tx_flags_t]
};

// CRC32
static uint32_t crc32_acc(const uint8_t* data, size_t len, uint32_t crc = 0xFFFFFFFF) {
  while (len--) {
    crc ^= *data++;
    for (int i=0;i<8;i++) crc = (crc >> 1) ^ (0xEDB88320 & -(crc & 1));
  }
  return ~crc;
}

// Режим RX постоянно (мы только принимаем)
static void rs485_setRX() { digitalWrite(RS485_DE_PIN, LOW); }

// Приёмный буфер (с запасом)
static uint8_t inbuf[4096];
static size_t  inlen = 0;

static void printUFO(const ufo_t& u, const char* prefix) {
  Serial.printf("[%s] addr=%06X lat=%.5f lon=%.5f alt=%.1f spd=%.1f flt=%s vr=%d sq=%d\n",
    prefix, (unsigned)u.addr, u.latitude, u.longitude, u.altitude, u.speed, u.flight, u.vert_rate, u.Squawk);
}

static void handleFlags(const tx_flags_t& f) {
  Serial.printf("[FLAGS] new=%d btn=%u setRead=%d read=%d SOSspr=%d SOSview=%d newSOS=%d confirm=%d GNSS=%d mode=%u analog=%u\n",
    f.new_flag_M, f.new_buttton_M, f.setMessageRead_M, f.MessageRead_M,
    f.SOS_Sprite_on_off_M, f.SOS_View_on_off_M, f.new_SOS_flag_M, f.confirm_message_M,
    f.isValidGNSS_M, f.FLYRF_MODE_TEST_M, f.analog_u16);
  Serial.printf("[FLAGS] msg_resp=\"%s\"\n", f.msg_resp_M);
}

// Поиск и обработка кадров в inbuf
static void processBuffer() {
  size_t pos = 0;
  while (inlen - pos >= sizeof(frame_hdr_t) + 4) { // минимум header + CRC
    frame_hdr_t* hdr = (frame_hdr_t*)(inbuf + pos);
    if (!(hdr->magic[0]=='U' && hdr->magic[1]=='F' && hdr->version==1)) {
      // сдвигаться по одному байту до совпадения magic
      pos++;
      continue;
    }
    size_t need = sizeof(frame_hdr_t) + hdr->payload_len + 4;
    if (inlen - pos < need) break; // ждём остаток

    // Проверка CRC
    uint32_t crc_rx;
    memcpy(&crc_rx, inbuf + pos + sizeof(frame_hdr_t) + hdr->payload_len, sizeof(crc_rx));
    uint32_t crc_calc = crc32_acc(inbuf + pos, sizeof(frame_hdr_t) + hdr->payload_len);
    if (crc_rx != crc_calc) {
      Serial.printf("[RX] CRC ERROR (seq=%u, type=%u) drop\n", hdr->seq, hdr->type);
      pos += 1; // ищем следующий magic
      continue;
    }

    // Разбор полезной нагрузки
    uint8_t* payload = inbuf + pos + sizeof(frame_hdr_t);
    switch (hdr->type) {
      case PKT_UFO_ITEM: {
        if (hdr->payload_len < 1 + sizeof(ufo_t)) { Serial.println("[RX] Bad UFO payload size"); break; }
        uint8_t idx = payload[0];
        if (idx < MAX_TRACKING_OBJECTS) {
          memcpy(&Container[idx], payload+1, sizeof(ufo_t));
          printUFO(Container[idx], "UFO");
        }
        break;
      }
      case PKT_THIS_AC: {
        if (hdr->payload_len != sizeof(ufo_t)) { Serial.println("[RX] Bad ThisAircraft size"); break; }
        memcpy(&ThisAircraft, payload, sizeof(ufo_t));
        printUFO(ThisAircraft, "OWN");
        break;
      }
      case PKT_FLAGS: {
        if (hdr->payload_len != sizeof(tx_flags_t)) { Serial.println("[RX] Bad FLAGS size"); break; }
        tx_flags_t f;
        memcpy(&f, payload, sizeof(f));
        handleFlags(f);
        break;
      }
      default:
        Serial.printf("[RX] Unknown type=%u\n", hdr->type);
        break;
    }

    pos += need; // перейти к следующему кадру
  }

  // Сдвиг непрочитанного остатка в начало
  if (pos > 0) {
    memmove(inbuf, inbuf + pos, inlen - pos);
    inlen -= pos;
  }
}

// Задача приема (ядро 0)
static void rxTask(void* arg) {
  const TickType_t period = pdMS_TO_TICKS(2);
  for (;;) {
    // Подчитать доступные байты
    int avail = Serial1.available();
    if (avail > 0) {
      int toRead = min(avail, (int)(sizeof(inbuf) - inlen));
      int got = Serial1.readBytes(inbuf + inlen, toRead);
      inlen += got;
      if (inlen >= sizeof(frame_hdr_t) + 4) processBuffer();
    }
    vTaskDelay(period);
  }
}

void setup() {
  pinMode(RS485_DE_PIN, OUTPUT);
  rs485_setRX(); // постоянно прием

  Serial.begin(115200);
  Serial.println("\n[RX] RS485 Receiver starting...");

  Serial1.begin(BAUD, SERIAL_8N1, UART_RX_PIN, UART_TX_PIN);

  xTaskCreatePinnedToCore(rxTask, "rxTask", 8192, nullptr, 2, nullptr, 0);
}

void loop() {
  // Ничего — всё в задаче
}

/* =====================  ESP32-S3 RS485 TRANSMITTER  ======================
 *  - Arduino IDE
 *  - Serial2: TX=GPIO18, RX=GPIO17
 *  - RS485 DE/RE: GPIO21 (HIGH=TX, LOW=RX)
 *  - FreeRTOS task pinned to Core 0
 *  - Baud: 921600 (устойчиво и быстро ~92 КБ/с; один пакет << 0.3 c)
 * ======================================================================= */

#include <Arduino.h>

// -------------------- Пользовательские типы и данные ---------------------
#define MAX_TRACKING_OBJECTS 8

typedef struct UFO {
  time_t    timestamp;
  uint32_t  addr;
  uint8_t   addr_type;
  float     latitude;
  float     longitude;
  float     old_latitude;
  float     old_longitude;
  float     altitude;
  float     pressure_altitude;
  float     course;        /* CoG */
  float     speed;         /* ground speed in knots */
  uint8_t   aircraft_type;
  char      flight[16];    // Flight number
  int       vert_rate;     // Vertical rate.
  int       Squawk;        // Squawk
  time_t    timemsg;       // Время передачи сообщения о координатах стороннего самолета

  float     vs; /* feet per minute */

  float     geoid_separation; /* metres */
  uint16_t  hdop; /* cm */
  int8_t    rssi; /* SX1276 only */

  /* 'legacy' specific data */
  float     distance;
  float     bearing;

  uint8_t   signal_source;
  time_t    seen;           // Time at which the last packet was received
  uint8_t   hour_msg;
  uint8_t   min_msg;
  uint16_t  delay_time_msg;
  uint8_t   callsign[8];
} ufo_t;

// Глобальные контейнеры, как в задании
ufo_t Container[MAX_TRACKING_OBJECTS];
ufo_t ThisAircraft;

// Доп. инфо источника (будет уходить в отдельном пакете)
typedef struct __attribute__((packed)) {
  bool     new_flag_M;
  uint8_t  new_buttton_M;
  bool     setMessageRead_M;
  bool     MessageRead_M;
  bool     SOS_Sprite_on_off_M;
  bool     SOS_View_on_off_M;
  bool     new_SOS_flag_M;
  bool     confirm_message_M;
  char     msg_resp_M[60];
  bool     isValidGNSS_M;
  uint8_t  FLYRF_MODE_TEST_M;
  uint16_t analog_u16;  // добавлено по требованию (аналоговый код)
} tx_flags_t;

// -------------------- RS485/UART пины и скорость -------------------------
static const int RS485_DE_PIN = 21;  // управление направлением
static const int UART_TX_PIN  = 18;
static const int UART_RX_PIN  = 17;
static const uint32_t BAUD    = 921600; // быстро, устойчиво для RS485-трансиверов

// -------------------- Протокол кадра (фрейма) ----------------------------
/* Фрейм:
 *  [0..1]  magic = 'U','F'
 *  [2]     version = 1
 *  [3]     type (1=UFO item, 2=ThisAircraft, 3=Flags+Analog)
 *  [4..5]  seq (LE)
 *  [6..7]  payload_len (LE)
 *  [8..9]  reserved=0
 *  [10.. ] payload
 *  [end-4..end-1] CRC32 от header(0..9) + payload
 */

#pragma pack(push,1)
typedef struct {
  uint8_t  magic[2];   // 'U','F'
  uint8_t  version;    // 1
  uint8_t  type;       // 1..N
  uint16_t seq;
  uint16_t payload_len;
  uint16_t reserved;
} frame_hdr_t;
#pragma pack(pop)

enum : uint8_t {
  PKT_UFO_ITEM = 1,       // payload: [uint8_t index][ufo_t]
  PKT_THIS_AC  = 2,       // payload: [ufo_t]
  PKT_FLAGS    = 3        // payload: [tx_flags_t]
};

// -------------------- CRC32 ----------------------------------------------
static uint32_t crc32_acc(const uint8_t* data, size_t len, uint32_t crc = 0xFFFFFFFF) {
  while (len--) {
    crc ^= *data++;
    for (int i=0;i<8;i++) crc = (crc >> 1) ^ (0xEDB88320 & -(crc & 1));
  }
  return ~crc;
}

// -------------------- Управление RS485 -----------------------------------
static void rs485_setTX(bool tx) {
  digitalWrite(RS485_DE_PIN, tx ? HIGH : LOW);
}

// Безопасная отправка буфера как RS485 (полудуплекс)
static void rs485_write(const uint8_t* buf, size_t len) {
  rs485_setTX(true);
  delayMicroseconds(10); // небольшая задержка на включение драйвера
  Serial2.write(buf, len);
  Serial2.flush();       // дождаться опустошения TX FIFO
  delayMicroseconds(10);
  rs485_setTX(false);
}

// -------------------- Буфер для фреймов ----------------------------------
static uint8_t outbuf[2048]; // с большим запасом (каждый пакет << 2KB)

// -------------------- Тестовые данные ------------------------------------
static void fillFakeUFO(ufo_t& u, uint32_t seed, const char* flt = "UF1234") {
  time_t nowt = time(nullptr);
  u.timestamp = nowt;
  u.addr = 0xA00000 | (seed & 0xFFFF);
  u.addr_type = 1;
  u.latitude  = 55.7f + (seed % 100) * 0.001f;
  u.longitude = 37.6f + (seed % 100) * 0.001f;
  u.old_latitude  = u.latitude  - 0.01f;
  u.old_longitude = u.longitude + 0.01f;
  u.altitude = 1200.0f + (seed % 500);
  u.pressure_altitude = u.altitude + 10.0f;
  u.course = fmodf( (float)(seed*11), 360.0f );
  u.speed  = 120.0f + (seed % 50);
  u.aircraft_type = 1;
  snprintf(u.flight, sizeof(u.flight), "%s", flt);
  u.vert_rate = (int)((seed % 20) - 10) * 100;
  u.Squawk = 7000 + (seed % 100);
  u.timemsg = nowt;
  u.vs = (float)u.vert_rate;
  u.geoid_separation = 34.0f;
  u.hdop = 75;
  u.rssi = -50;
  u.distance = 12.3f + (seed % 10);
  u.bearing  = 45.0f + (seed % 90);
  u.signal_source = 2;
  u.seen = nowt;
  u.hour_msg = (uint8_t)((millis()/3600000)%24);
  u.min_msg  = (uint8_t)((millis()/60000)%60);
  u.delay_time_msg = 50;
  for (int i=0;i<8;i++) u.callsign[i] = (uint8_t)('A'+ (seed+i)%26);
}

static void makeTestData() {
  for (int i=0;i<MAX_TRACKING_OBJECTS;i++) {
    char flt[16];
    snprintf(flt, sizeof(flt), "UF%04d", 1000+i);
    fillFakeUFO(Container[i], i+1, flt);
  }
  fillFakeUFO(ThisAircraft, 0xBEEF, "OWNR01");
}

// -------------------- Формирование и отправка фреймов --------------------
static uint16_t g_seq = 1;

template<typename PAYLOAD_FILLER>
static void sendFrame(uint8_t type, uint16_t payload_len, PAYLOAD_FILLER filler) {
  frame_hdr_t hdr;
  hdr.magic[0]='U'; hdr.magic[1]='F';
  hdr.version = 1;
  hdr.type    = type;
  hdr.seq     = g_seq++;
  hdr.payload_len = payload_len;
  hdr.reserved = 0;

  // собрать в outbuf: header + payload + crc32
  memcpy(outbuf, &hdr, sizeof(hdr));
  filler(outbuf + sizeof(hdr)); // заполнить payload

  uint32_t crc = crc32_acc(outbuf, sizeof(hdr) + payload_len);
  memcpy(outbuf + sizeof(hdr) + payload_len, &crc, sizeof(crc));

  size_t total = sizeof(hdr) + payload_len + sizeof(crc);
  rs485_write(outbuf, total);
}

// Отправить один элемент Container[i]
static void sendUFOItem(uint8_t idx) {
  struct __attribute__((packed)) {
    uint8_t index;
    ufo_t   u;
  } payload;
  payload.index = idx;
  payload.u     = Container[idx];

  sendFrame(PKT_UFO_ITEM, sizeof(payload), [&](uint8_t* p){
    memcpy(p, &payload, sizeof(payload));
  });
}

// Отправить ThisAircraft
static void sendThisAircraft() {
  sendFrame(PKT_THIS_AC, sizeof(ThisAircraft), [&](uint8_t* p){
    memcpy(p, &ThisAircraft, sizeof(ThisAircraft));
  });
}

// Отправить флаги + аналог
static void sendFlagsAndAnalog() {
  tx_flags_t f = {};
  // Тестовые значения
  f.new_flag_M = true;
  f.new_buttton_M = 7;
  f.setMessageRead_M = true;
  f.MessageRead_M    = false;
  f.SOS_Sprite_on_off_M = false;
  f.SOS_View_on_off_M   = true;
  f.new_SOS_flag_M      = (millis()/1000)%2;
  f.confirm_message_M   = true;
  snprintf(f.msg_resp_M, sizeof(f.msg_resp_M), "OK %lu", (unsigned long)millis());
  f.isValidGNSS_M = true;
  f.FLYRF_MODE_TEST_M = 3;
  // «Аналоговый» код (имитация): 0..4095
  f.analog_u16 = (uint16_t)(millis() % 4096);

  sendFrame(PKT_FLAGS, sizeof(f), [&](uint8_t* p){
    memcpy(p, &f, sizeof(f));
  });
}

// -------------------- FreeRTOS таск (ядро 0) -----------------------------
static void txTask(void* arg) {
  // Генерация тестовых данных каждые 500 мс + отправка пакетами
  const TickType_t period = pdMS_TO_TICKS(500);

  for (;;) {
    makeTestData();

    // Отправляем Container по элементам (8 небольших пакетов)
    for (uint8_t i=0;i<MAX_TRACKING_OBJECTS;i++) {
      sendUFOItem(i);
      // Межпакетная пауза небольшая (минимальная), чтобы не захлебнуться
      vTaskDelay(pdMS_TO_TICKS(5));
    }

    // Отдельным пакетом — ThisAircraft
    sendThisAircraft();
    vTaskDelay(pdMS_TO_TICKS(5));

    // Отдельным пакетом — флаги + аналог
    sendFlagsAndAnalog();

    vTaskDelay(period);
  }
}

// -------------------- setup/loop -----------------------------------------
void setup() {
  pinMode(RS485_DE_PIN, OUTPUT);
  rs485_setTX(false);

  Serial.begin(115200);
  Serial.println("\n[TX] RS485 Transmitter starting...");

  Serial2.begin(BAUD, SERIAL_8N1, UART_RX_PIN, UART_TX_PIN);

  // Запуск FreeRTOS задачи на Core 0
  xTaskCreatePinnedToCore(txTask, "txTask", 8192, nullptr, 2, nullptr, 0);
}

void loop() {
  // пусто: вся работа в задаче
}
