#include <Arduino.h>
#include <HardwareSerial.h>
#include <freertos/FreeRTOS.h>
#include <freertos/semphr.h>

#define RS485_SERIAL Serial2
#define RS485_TX_PIN 18
#define RS485_RX_PIN 17
#define RS485_DE_PIN 21
#define RS485_BAUD   115200
#define RS485_CONFIG SERIAL_8N1

#define MAX_TRACKING_OBJECTS 8
#define PREAMBLE 0x55AA
#define PROTO_VER 0x01
#define MAX_PAYLOAD 600

// -- Ваши структуры --

typedef struct {
  uint32_t  addr;
  int       squawk;
  uint8_t   callsign[8];
  float     altitude;
  float     pressure_altitude;
  float     course;
  float     speed;
  int       vert_rate;
  float     latitude;
  float     longitude;
  int8_t    rssi;
  uint16_t  last_message_signal_strength_dbm;
  uint16_t  last_message_signal_quality_db;
} ufo_t;

typedef struct {
  bool     new_flag_M;
  uint8_t  new_buttton_M;
  bool     setMessageRead_M;
  bool     MessageRead_M;
  uint8_t  Time_Hour_M;
  uint8_t  Time_Minute_M;
  bool     new_SOS_flag_M;
  char     msg_resp_M[170];
  bool     isValidGNSS_M;
} aux_t;

// --- Глобальное ---
ufo_t Container[MAX_TRACKING_OBJECTS];
aux_t auxData;

SemaphoreHandle_t containerSemaphore;

// --- Вспомогательные функции ---
uint32_t swap32(uint32_t val) {
    return ((val & 0xFF) << 24) | ((val & 0xFF00) << 8) |
           ((val & 0xFF0000) >> 8) | ((val & 0xFF000000) >> 24);
}
float swapFloat(float val) {
    uint32_t temp; memcpy(&temp, &val, 4);
    temp = swap32(temp);
    float res; memcpy(&res, &temp, 4);
    return res;
}
uint16_t toBigEndian16(uint16_t val) {
    return ((val & 0x00FF) << 8) | ((val & 0xFF00) >> 8);
}
uint16_t crc16_ibm(const uint8_t* data, size_t len) {
    uint16_t crc = 0;
    while (len--) {
        crc ^= *data++;
        for (uint8_t i = 0; i < 8; ++i)
            crc = crc & 1 ? (crc >> 1) ^ 0xA001 : (crc >> 1);
    }
    return crc;
}

// --- Сериализация ufo_t ---
void serializeUFO(const ufo_t* src, uint8_t* buf) {
    size_t p = 0;
    uint32_t tmp32;
    int tmpint;
    tmp32 = swap32(src->addr);             memcpy(&buf[p], &tmp32, 4); p+=4;
    tmpint = swap32(src->squawk);          memcpy(&buf[p], &tmpint, 4); p+=4;
    memcpy(&buf[p], src->callsign, 8);     p+=8;
    float f;
    f = swapFloat(src->altitude);          memcpy(&buf[p], &f, 4); p+=4;
    f = swapFloat(src->pressure_altitude); memcpy(&buf[p], &f, 4); p+=4;
    f = swapFloat(src->course);            memcpy(&buf[p], &f, 4); p+=4;
    f = swapFloat(src->speed);             memcpy(&buf[p], &f, 4); p+=4;
    tmpint = swap32(src->vert_rate);       memcpy(&buf[p], &tmpint, 4); p+=4;
    f = swapFloat(src->latitude);          memcpy(&buf[p], &f, 4); p+=4;
    f = swapFloat(src->longitude);         memcpy(&buf[p], &f, 4); p+=4;
    buf[p++] = src->rssi;
    uint16_t u16 = toBigEndian16(src->last_message_signal_strength_dbm);
    memcpy(&buf[p], &u16, 2); p+=2;
    u16 = toBigEndian16(src->last_message_signal_quality_db);
    memcpy(&buf[p], &u16, 2); p+=2;
}

// --- Сериализация aux_t ---
void serializeAux(const aux_t* src, uint8_t* buf) {
    size_t p = 0;
    buf[p++] = src->new_flag_M ? 1 : 0;
    buf[p++] = src->new_buttton_M;
    buf[p++] = src->setMessageRead_M ? 1 : 0;
    buf[p++] = src->MessageRead_M ? 1 : 0;
    buf[p++] = src->Time_Hour_M;
    buf[p++] = src->Time_Minute_M;
    buf[p++] = src->new_SOS_flag_M ? 1 : 0;
    memcpy(&buf[p], src->msg_resp_M, 170); p+=170;
    buf[p++] = src->isValidGNSS_M ? 1 : 0;
}

// --- Передача ---
void sendPacket() {
    if (xSemaphoreTake(containerSemaphore, (TickType_t)10) == pdTRUE) {
        uint8_t payload[MAX_PAYLOAD];
        size_t ptr = 0;
        for(int i=0;i<MAX_TRACKING_OBJECTS;i++) {
            serializeUFO(&Container[i], &payload[ptr]);
            ptr += 43;
        }
        serializeAux(&auxData, &payload[ptr]);
        ptr += 178;

        uint16_t preamble = PREAMBLE;
        uint8_t proto = PROTO_VER;
        uint16_t crc = crc16_ibm(payload, ptr);

        digitalWrite(RS485_DE_PIN, HIGH);
        delayMicroseconds(10);
        RS485_SERIAL.write((uint8_t*)&preamble, 2);
        RS485_SERIAL.write(proto);
        RS485_SERIAL.write(payload, ptr);
        RS485_SERIAL.write((uint8_t*)&crc, 2);
        RS485_SERIAL.flush();
        digitalWrite(RS485_DE_PIN, LOW);

        xSemaphoreGive(containerSemaphore);
    }
}

// --- Приём статуса кнопок ---
void receiveButtons() {
    while (RS485_SERIAL.available() >= 2) {
        uint8_t rxBuf[2];
        RS485_SERIAL.readBytes(rxBuf, 2);
        if (rxBuf[0] == 0xAB) {
            uint8_t btn1 = rxBuf[1]&1, btn2 = (rxBuf[1]>>1)&1;
            Serial.printf("Buttons: %d %d
", btn1, btn2);
        } else {
            // Ошибка в протоколе, очиститься до первой 0xAB
            while (RS485_SERIAL.available()>0 && RS485_SERIAL.peek()!=0xAB) RS485_SERIAL.read();
        }
    }
}

void setup() {
    Serial.begin(115200);
    RS485_SERIAL.begin(RS485_BAUD, RS485_CONFIG, RS485_RX_PIN, RS485_TX_PIN);
    pinMode(RS485_DE_PIN, OUTPUT); digitalWrite(RS485_DE_PIN, LOW);
    containerSemaphore = xSemaphoreCreateBinary(); xSemaphoreGive(containerSemaphore);
}

void loop() {
    sendPacket();
    receiveButtons();
    delay(500);
}

#include <Arduino.h>

// -- Swap, CRC
uint32_t swap32(uint32_t val) {
    return ((val & 0xFF) << 24) | ((val & 0xFF00) << 8) |
           ((val & 0xFF0000) >> 8) | ((val & 0xFF000000) >> 24);
}
float swapFloat(float val) {
    uint32_t temp; memcpy(&temp, &val, 4);
    temp = swap32(temp);
    float res; memcpy(&res, &temp, 4);
    return res;
}
uint16_t fromBigEndian16(uint16_t val){ return ((val&0xFF)<<8)|((val>>8)&0xFF);}
uint16_t crc16_ibm(const uint8_t* data, size_t len) {
    uint16_t crc = 0;
    while (len--) {
        crc ^= *data++;
        for (uint8_t i = 0; i < 8; ++i)
            crc = crc & 1 ? (crc >> 1) ^ 0xA001 : (crc >> 1);

void deserializeUFO(const uint8_t* buf, ufo_t* dst) {
    size_t p = 0;
    uint32_t tmp32; int tmpint;
    memcpy(&tmp32, &buf[p], 4);           dst->addr = swap32(tmp32); p+=4;
    memcpy(&tmpint, &buf[p], 4);          dst->squawk = swap32(tmpint); p+=4;
    memcpy(dst->callsign,  &buf[p], 8);   p+=8;
    float f;
    memcpy(&f, &buf[p], 4); dst->altitude = swapFloat(f); p+=4;
    memcpy(&f, &buf[p], 4); dst->pressure_altitude = swapFloat(f); p+=4;
    memcpy(&f, &buf[p], 4); dst->course = swapFloat(f); p+=4;
    memcpy(&f, &buf[p], 4); dst->speed = swapFloat(f); p+=4;
    memcpy(&tmpint, &buf[p], 4); dst->vert_rate = swap32(tmpint); p+=4;
    memcpy(&f, &buf[p], 4); dst->latitude = swapFloat(f); p+=4;
    memcpy(&f, &buf[p], 4); dst->longitude = swapFloat(f); p+=4;
    dst->rssi = buf[p++];
    uint16_t u16;
    memcpy(&u16, &buf[p], 2); dst->last_message_signal_strength_dbm = fromBigEndian16(u16); p+=2;
    memcpy(&u16, &buf[p], 2); dst->last_message_signal_quality_db = fromBigEndian16(u16); p+=2;
}

// --- Приём пакета: синхронизация по преамбуле, защита от ошибок
// --- если не хватает байт — выход; если CRC не совпал — очистка до следующей преамбулы
void receivePacket() {
    static uint8_t rxBuf[MAX_PAYLOAD+8];
    static size_t rxCount = 0;
    while (RS485_SERIAL.available() > 0) {
        rxBuf[rxCount++] = RS485_SERIAL.read();
        if (rxCount >= 3) {
            uint16_t preamble = rxBuf[0] | (rxBuf[1]<<8);
            if (preamble != PREAMBLE) {
                // Сдвигаем до вероятной преамбулы
                for (size_t i = 1; i < rxCount; ++i)
                    if ((rxBuf[i] | (rxBuf[i+1]<<8)) == PREAMBLE) {
                        memmove(rxBuf, &rxBuf[i], rxCount-i);
                        rxCount -= i;
                        break;
                    }
                if (preamble != PREAMBLE) { rxCount = 0; continue; }
            }
            // После удачной синхронизации пытаемся считать пакет целиком
            if (rxCount >=  (3 + MAX_TRACKING_OBJECTS*43 + 178 + 2)) {
                size_t payloadLen = MAX_TRACKING_OBJECTS*43 + 178;
                if (rxBuf[2] != PROTO_VER) { rxCount = 0; continue;}
                uint16_t pk_crc = rxBuf[3+payloadLen] | (rxBuf[4+payloadLen]<<8);
                uint16_t cl_crc = crc16_ibm(&rxBuf[3], payloadLen);
                if (pk_crc != cl_crc) { Serial.println("CRC error"); rxCount = 0; continue;}
                if (xSemaphoreTake(containerSemaphore, (TickType_t)10) == pdTRUE) {
                    for (int i=0; i<MAX_TRACKING_OBJECTS;i++)
                        deserializeUFO(&rxBuf[3]+i*43, &Container[i]);
                    deserializeAux(&rxBuf[3]+MAX_TRACKING_OBJECTS*43, &auxData);
                    dataReceived = true;
                    xSemaphoreGive(containerSemaphore);
                    Serial.println("Packet OK");
                }
                rxCount = 0;
            }
        }
        // Переполнение (опечатка в размере?) - сброситься
        if (rxCount >= sizeof(rxBuf)) rxCount = 0;
    }
}

void sendButtonsState(bool button1, bool button2) {
    uint8_t outbuf[2];
    outbuf[0] = 0xAB; // marker
    outbuf[1] = (button1 ? 1 : 0) | ((button2 ? 1 : 0)<<1);
    digitalWrite(RS485_DE_PIN, HIGH);
    delayMicroseconds(10);
    RS485_SERIAL.write(outbuf, 2);
    RS485_SERIAL.flush();
    digitalWrite(RS485_DE_PIN, LOW);
}

void loop() {
    receivePacket();
    bool b1 = (digitalRead(BUTTON1_PIN)==LOW);
    bool b2 = (digitalRead(BUTTON2_PIN)==LOW);
    sendButtonsState(b1, b2);
    delay(500);
}
Краткие рекомендации:
При ошибке преамбулы и CRC — автоматический сброс приёма к новому пакету.
Все операции с Container/aux через семафор, чтобы не получить race condition.
Продуманы все конвертации (swap/float/endian).
Сериализация и разборка структур полностью ручные!
Режим работы Button1/2 — INPUT_PULLUP: кнопка замыкает на GND.
Размер буфера достаточно на максимальный пакет. Проверки переполнения есть.
Поясню работу протокола или помогу свести к библиотеке — если потребуется!
