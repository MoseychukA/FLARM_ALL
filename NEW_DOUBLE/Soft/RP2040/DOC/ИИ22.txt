#include "dma_ring.h"
#include "hardware/regs/dreq.h"

static void __isr __time_critical_func(dma_irq0_handler)();
static DmaRing *g_rings[12] = {0};

bool dma_ring_init(DmaRing &r, PIO pio, uint sm, size_t words) {
if ((words & (words - 1)) != 0) return false;
r.buf = (uint32_t*)malloc(words * sizeof(uint32_t));
if (!r.buf) return false;
r.words = words; r.widx = 0; r.ridx = 0; r.pio = pio; r.sm = sm;
r.dma_ch = dma_claim_unused_channel(true);

dma_channel_config c = dma_channel_get_default_config(r.dma_ch);
channel_config_set_read_increment(&c, false);
channel_config_set_write_increment(&c, true);
channel_config_set_dreq(&c, pio == pio0 ? (DREQ_PIO0_RX0 + sm) : (DREQ_PIO1_RX0 + sm));
channel_config_set_transfer_data_size(&c, DMA_SIZE_32);
dma_channel_configure(r.dma_ch, &c, r.buf, &r.pio->rxf[sm], r.words, false);

dma_channel_set_irq0_enabled(r.dma_ch, true);

static bool irq_installed = false;
if (!irq_installed) {
irq_set_exclusive_handler(DMA_IRQ_0, dma_irq0_handler);
irq_set_enabled(DMA_IRQ_0, true);
irq_installed = true;
}
g_rings[r.dma_ch] = &r;
dma_channel_start(r.dma_ch);
return true;
}

void dma_ring_deinit(DmaRing &r) {
if (r.dma_ch >= 0) {
dma_channel_abort(r.dma_ch);
dma_channel_set_irq0_enabled(r.dma_ch, false);
g_rings[r.dma_ch] = nullptr;
dma_channel_unclaim(r.dma_ch);
}
if (r.buf) { free(r.buf); r.buf = nullptr; }
}

size_t dma_ring_available(const DmaRing &r) {
return (r.widx - r.ridx) & (r.words - 1);
}

bool dma_ring_get_word(DmaRing &r, uint32_t &w) {
if (!dma_ring_available(r)) return false;
w = r.buf[r.ridx];
r.ridx = (r.ridx + 1) & (r.words - 1);
return true;
}

static void __isr __time_critical_func(dma_irq0_handler)() {
uint32_t pending = dma_hw->ints0;
for (int ch=0; ch<12; ch++) if (pending & (1u<<ch)) {
auto *r = g_rings[ch];
if (r) {
r->widx = (r->widx + r->words) & (r->words - 1);
dma_hw->ints0 = (1u<<ch);
dma_channel_set_trans_count(ch, r->words, false);
dma_channel_start(ch);
} else {
dma_hw->ints0 = (1u<<ch);
}
}
}

#include <stdint.h>
#include <stddef.h>

// CRC-24 для Mode S
typedef struct {
uint32_t tbl[256];
uint8_t inited;
} crc24_ctx_t;

static crc24_ctx_t g_crc24_ctx;

static inline void crc24_init_table() {
if (g_crc24_ctx.inited) return;
const uint32_t POLY = 0x864CFB;
for (uint32_t i=0; i<256; i++) {
uint32_t crc = i << 16;
for (int j=0;j<8;j++){ crc <<= 1; if (crc & 0x1000000) crc ^= POLY; }
g_crc24_ctx.tbl[i] = crc & 0xFFFFFF;
}
g_crc24_ctx.inited = 1;
}

static inline uint32_t crc24_update(uint32_t crc, uint8_t data) {
uint8_t idx = (uint8_t)((crc >> 16) ^ data);
crc = ((crc << 8) & 0xFFFFFF) ^ g_crc24_ctx.tbl[idx];
return crc & 0xFFFFFF;
}

static inline uint32_t modes_crc24(const uint8_t* data, size_t nbytes) {
crc24_init_table();
uint32_t crc = 0;
for (size_t i=0;i<nbytes;i++) crc = crc24_update(crc, data[i]);
return crc & 0xFFFFFF;
}

static inline uint32_t modes_parity_syndrome(const uint8_t* full, size_t nbytes) {
uint32_t parity = (full[nbytes-3]<<16)|(full[nbytes-2]<<8)|full[nbytes-1];
uint32_t crc = modes_crc24(full, nbytes-3);
return (crc ^ parity) & 0xFFFFFF;
}

// Исправление одиночной ошибки бита
static inline int modes_try_single_bit_fix(uint8_t* msg, size_t nbits) {
crc24_init_table();
const size_t nbytes = (nbits + 7) / 8;
for (size_t b=0; b<nbits; b++) {
size_t i = b >> 3;
uint8_t m = 1u << (7 - (b & 7));
msg[i] ^= m;
uint32_t synd = modes_parity_syndrome(msg, nbytes);
if (synd == 0) return (int)b;
msg[i] ^= m;
}
return -1;
}

int decode_altitude_gillham(uint16_t ac) {
int D1=(ac>>0)&1, D2=(ac>>1)&1, D4=(ac>>2)&1;
int A1=(ac>>3)&1, A2=(ac>>4)&1, A4=(ac>>5)&1;
int B1=(ac>>6)&1, B2=(ac>>7)&1, B4=(ac>>8)&1;
int C1=(ac>>9)&1, C2=(ac>>10)&1, C4=(ac>>11)&1;
int f100=(D1|D2|D4);
auto gray2bin=[](int g){ int b=0; for(; g; g>>=1) b^=g; return b; };
int n500 = gray2bin((A4<<2)|(A2<<1)|A1);
int n5k = gray2bin((B4<<2)|(B2<<1)|B1);
int n20k = gray2bin((C4<<2)|(C2<<1)|C1);
static const int tbl100[8]={0,100,200,300,400,500,600,700};
int idx100=gray2bin((D4<<2)|(D2<<1)|D1);
int alt=(n20k16000)+(n5k2000)+(n500*500);
if (f100) alt += tbl100[idx100];
alt -= 1300;
if (alt<-1200 || alt>126000) return -1;
return alt;
}

bool decode_velocity_tc19(const uint8_t* me, double& gs, double& trk, int& vs, int& baro_geo) {

int subtype=(acc>>(56-8-37))&0x7; gs=NAN; trk=NAN; vs=INT32_MIN; baro_geo=INT32_MIN;
if (subtype==1 || subtype==2) {
int ew_dir=(acc>>(56-8-38))&1; int ew_spd=(acc>>(56-8-46))&0x1FF;
int ns_dir=(acc>>(56-8-47))&1; int ns_spd=(acc>>(56-8-55))&0x1FF;
double vE=(ew_spd==0)?0:(ew_spd-1); vE=(ew_dir?-vE:vE);
double vN=(ns_spd==0)?0:(ns_spd-1); vN=(ns_dir?-vN:vN);
gs=sqrt(vEvE+vNvN);
trk=fmod(atan2(vE, vN)180.0/M_PI + 360.0, 360.0);
} else if (subtype==3) {
int hdg=((acc>>(56-8-46))&0x1FF);
int spd=((acc>>(56-8-55))&0x1FF);

} else if (subtype==4) {

int tas=((acc>>(56-8-55))&0x1FF);
if (tas) { trk=hdg*360.0/512.0; gs=tas-1; }
}
return (!isnan(gs));
}

if (xQueueReceive(q_packets, &pkt, pdMS_TO_TICKS(10)) == pdPASS) decode_and_output(pkt);
while (xQueueReceive(q_fixed_packets, &pkt, 0) == pdPASS) {
  Serial.print("FIXED_RAW="); for (uint8_t i=0;i<(pkt.nbits+7)/8;i++){ if(pkt.data[i]<16) Serial.print('0'); Serial.print(pkt.data[i],HEX);} Serial.print("\r
");
}

if (xTaskGetTickCount() - lastBlink >= pdMS_TO_TICKS(1000)){
  fast_toggle(PIN_BLINK_CORE0);
  lastBlink = xTaskGetTickCount();
}
}
}

static uint16_t crc16_ccitt(const uint8_t* data, size_t len){
uint16_t crc=0xFFFF;
for(size_t i=0;i<len;i++){
crc ^= (uint16_t)data[i] << 8;
for(int j=0;j<8;j++){
if(crc & 0x8000) crc=(crc<<1) ^ 0x1021; else crc<<=1;
}
}
return crc;
}
