Ниже — рабочий шаблон обмена по RS‑485 между двумя ESP32‑S3 (источник и приёмник) в стиле Arduino/Apduino IDE с использованием FreeRTOS (все задачи закреплены за ядром 0). Передаются только изменившиеся объекты, фреймы с CRC‑16, подтверждение ACK/NACK, сегментация “по объектам” (каждый объект — отдельный пакет). В источнике — генерация тестовых данных; в приёмнике — вывод в Serial. Управление направлением RS‑485 через DE/RE.

Ключевые параметры:

Интерфейсы и пины:
Источник: Serial2 TX=GPIO18, RX=GPIO17, DE/RE=GPIO21
Приёмник: Serial1 TX=GPIO39, RX=GPIO38, DE/RE=GPIO40
Скорость: 921600 бод (можно менять через define)
Ограничение по времени на пакет: при такой скорости и размере пакета <256 байт время передачи единичного фрейма < 3 мс
Протокол:
Преамбула: 0xAA 0x55
ver (1), type (1), index (1), seq (1), length (2), payload (length), CRC16-IBM (2) — CRC считается по полям [ver..payload]
Типы: 0x01 — Container[i], 0x02 — ThisAircraft, 0x03 — Analog, 0x04 — AuxFlags, 0x05 — Heartbeat, 0xF0 — ACK, 0xF1 — NACK
ACK payload: orig_type (1), orig_index (1), orig_seq (1), status (1) — 0x06 ACK, 0x15 NACK
Передаём “только изменённые данные”: отправляем целиком объект/структуру, но лишь при обнаружении изменений относительно последней отправленной копии
Контроль приёма: проверка CRC, ответ ACK/NACK, повтор до 3 попыток
Допполя источника: передаются отдельным фреймом AuxFlags + код аналогового сигнала uint16_t
Схема обмена (мермаид):

Код 1. Источник (ESP32S3, Serial2: TX18, RX17, DE/RE21)

enum : uint8_t {
  PKT_CONTAINER = 0x01,
  PKT_THISAC    = 0x02,
  PKT_ANALOG    = 0x03,
  PKT_AUX       = 0x04,
  PKT_HEARTBEAT = 0x05,
  PKT_ACK       = 0xF0,
  PKT_NACK      = 0xF1
};

#pragma pack(push,1)
struct FrameHeader {
  uint16_t preamble; // 0xAA55
  uint8_t  ver;      // 0x01
  uint8_t  type;
  uint8_t  index;    // 0..7 для контейнера, 0xFF для ThisAircraft, 0 для прочих
  uint8_t  seq;      // счётчик кадров
  uint16_t length;   // длина payload
};
#pragma pack(pop)

#define PREAMBLE 0x55AAu  // будем писать в LE как 0xAA,0x55
#define PROTO_VER 0x01
#define MAX_PAYLOAD 256

// CRC16-IBM (Modbus)
static uint16_t crc16_ibm(const uint8_t* data, size_t len) {
  uint16_t crc = 0xFFFF;
  for (size_t i = 0; i < len; ++i) {
    crc ^= data[i];
    for (int j = 0; j < 8; ++j) {
      if (crc & 1) crc = (crc >> 1) ^ 0xA001;
      else crc >>= 1;
    }
  }
  return crc;
}

static void rs485SetTX(bool enable) {
  digitalWrite(RS485_DE_PIN, enable ? HIGH : LOW);
  // Небольшие задержки для стабилизации драйвера
  if (enable) delayMicroseconds(50);
}

  uint8_t  seq;
  uint16_t length;

// ================== Вспомогательные ===================
static bool memdiff(const void* a, const void* b, size_t n) {
  return memcmp(a, b, n) != 0;
}

static uint8_t nextSeq() {
  return ++seqCounter; // переполнение допустимо
}

static void enqueueFrame(uint8_t type, uint8_t index, const void* data, uint16_t len) {
  if (len > MAX_PAYLOAD) return;
  TxItem item{};
  item.type = type;
  item.index = index;
  item.seq = nextSeq();
  item.length = len;
  memcpy(item.payload, data, len);
  xQueueSend(txQueue, &item, 0);
}

static void sendAckLike(uint8_t type, uint8_t index, uint8_t seq, bool positive) {
  uint8_t payload[4];
  payload[0] = type;
  payload[1] = index;
  payload[2] = seq;
  payload[3] = positive ? 0x06 : 0x15;

  FrameHeader hdr;
  hdr.preamble = PREAMBLE;
  hdr.ver = PROTO_VER;
  hdr.type = positive ? PKT_ACK : PKT_NACK;
  hdr.index = 0;
  hdr.seq = 0;
  hdr.length = sizeof(payload);

  uint8_t header_and_payload[sizeof(FrameHeader)-2 + sizeof(payload)]; // CRC считаем с ver
  memcpy(&header_and_payload[0], &hdr.ver, sizeof(FrameHeader)-2);     // исключаем preamble
  memcpy(&header_and_payload[sizeof(FrameHeader)-2], payload, sizeof(payload));
  uint16_t crc = crc16_ibm(header_and_payload, sizeof(header_and_payload));

  // Отправка
  if (xSemaphoreTake(serialMutex, pdMS_TO_TICKS(50)) == pdTRUE) {
    rs485SetTX(true);
    RS485_SERIAL.write((uint8_t*)&hdr, sizeof(hdr));
    RS485_SERIAL.write(payload, sizeof(payload));
    RS485_SERIAL.write((uint8_t*)&crc, sizeof(crc));
    RS485_SERIAL.flush();
    delayMicroseconds(200);
    rs485SetTX(false);
    xSemaphoreGive(serialMutex);
  }
}

// ============== Парсер входящего потока ================
typedef struct {
  uint8_t type;
  uint8_t index;
  uint8_t seq;
  bool    isAck;
  bool    crcOk;
} AckInfo;

static bool parseOneFrame(Stream& s, FrameHeader& hdr, uint8_t* payloadBuf, uint16_t& crc) {
  // Поиск преамбулы
  static uint8_t state = 0;
  uint8_t b;
  while (s.available()) {
    b = s.read();
    if (state == 0 && b == 0xAA) { state = 1; continue; }
    if (state == 1 && b == 0x55) { // нашли 0xAA 0x55
      // читаем остальной заголовок
      size_t need = sizeof(FrameHeader) - 2;
      uint8_t* p = ((uint8_t*)&hdr) + 2;
      size_t got = 0;
      while (got < need) {
        int v = s.read();
        if (v < 0) { state = 2; goto WAIT_CONT; }
        p[got++] = (uint8_t)v;
      }
      state = 3;
WAIT_CONT:
      if (state != 3) return false;
      // Проверка версий и длины
      if (hdr.ver != PROTO_VER || hdr.length > MAX_PAYLOAD) {
        state = 0;
        continue;
      }
      // читаем payload
      for (uint16_t i = 0; i < hdr.length; ++i) {
        int v = -1;
        while ((v = s.read()) < 0) { delayMicroseconds(50); }
        payloadBuf[i] = (uint8_t)v;
      }
      // читаем CRC
      uint8_t* c = (uint8_t*)&crc;
      for (int i = 0; i < 2; ++i) {
        int v = -1;
        while ((v = s.read()) < 0) { delayMicroseconds(50); }
        c[i] = (uint8_t)v;
      }
      state = 0;
      return true;
    } else {
      state = 0;
    }
  }
  return false;
}

void RxTask(void* arg) {
  FrameHeader hdr;
  uint8_t payload[MAX_PAYLOAD];
  uint16_t rxCrc;

  for (;;) {
    // Ждём данные
    if (!RS485_SERIAL.available()) {
      vTaskDelay(pdMS_TO_TICKS(1));
      continue;
    }

    if (!parseOneFrame(RS485_SERIAL, hdr, payload, rxCrc)) {
      vTaskDelay(pdMS_TO_TICKS(1));
      continue;
    }

    // Проверка CRC
    uint8_t buf[sizeof(FrameHeader)-2 + MAX_PAYLOAD];
    memcpy(buf, &hdr.ver, sizeof(FrameHeader)-2);
    memcpy(buf + sizeof(FrameHeader)-2, payload, hdr.length);
    uint16_t calc = crc16_ibm(buf, (sizeof(FrameHeader)-2) + hdr.length);

    bool ok = (calc == rxCrc);

    if (hdr.type == PKT_ACK || hdr.type == PKT_NACK) {
      // Сообщим TxTask
      if (ok && hdr.length == 4) {
        uint8_t orig_type = payload[0];
        uint8_t orig_index = payload[1];
        uint8_t orig_seq = payload[2];
        uint8_t status = payload[3];
        AckInfo a{orig_type, orig_index, orig_seq, true, status == 0x06};
        xQueueSend(ackQueue, &a, 0);
      }
      continue;
    }

    // Для источника данные от приёмника не предполагаются => игнор
    // Но в общем случае можно ответить ACK/NACK на любые принятые кадры.
    // Здесь вернём ACK/NACK, чтобы TxTask на приёмнике (если будет) мог работать.
    sendAckLike(hdr.type, hdr.index, hdr.seq, ok);
  }
}

void TxTask(void* arg) {
  const TickType_t ACK_TIMEOUT = pdMS_TO_TICKS(50);
  TxItem item;
  for (;;) {
    if (xQueueReceive(txQueue, &item, portMAX_DELAY) == pdTRUE) {
      // Сформировать фрейм
      FrameHeader hdr;
      hdr.preamble = PREAMBLE;
      hdr.ver = PROTO_VER;
      hdr.type = item.type;
      hdr.index = item.index;
      hdr.seq = item.seq;
      hdr.length = item.length;

      uint8_t header_and_payload[sizeof(FrameHeader)-2 + MAX_PAYLOAD];
      memcpy(header_and_payload, &hdr.ver, sizeof(FrameHeader)-2);
      memcpy(header_and_payload + sizeof(FrameHeader)-2, item.payload, item.length);
      uint16_t crc = crc16_ibm(header_and_payload, (sizeof(FrameHeader)-2) + item.length);

      // Отправка + ожидание ACK с повторами
      const int MAX_RETRY = 3;
      bool sentOk = false;
      for (int attempt = 0; attempt < MAX_RETRY && !sentOk; ++attempt) {
        if (xSemaphoreTake(serialMutex, pdMS_TO_TICKS(50)) == pdTRUE) {
          rs485SetTX(true);
          RS485_SERIAL.write((uint8_t*)&hdr, sizeof(hdr));
          RS485_SERIAL.write(item.payload, item.length);
          RS485_SERIAL.write((uint8_t*)&crc, sizeof(crc));
          RS485_SERIAL.flush();
          delayMicroseconds(200);
          rs485SetTX(false);
          xSemaphoreGive(serialMutex);
        }

        // Ждём ACK
        AckInfo a{};
        TickType_t start = xTaskGetTickCount();
        for (;;) {
          if (xQueueReceive(ackQueue, &a, pdMS_TO_TICKS(5)) == pdTRUE) {
            if (a.crcOk && a.type == item.type && a.index == item.index && a.seq == item.seq) {
              if (a.crcOk) sentOk = true;
              break;
            }
          }
          if (xTaskGetTickCount() - start > ACK_TIMEOUT) break;
        }
      }

      // Можно логировать результат
      if (!sentOk) {
        Serial.printf("[TX] FAIL type=%02X idx=%u seq=%u
", item.type, item.index, item.seq);
      }
    }
  }
}

void DataGenTask(void* arg) {
  // Имитация тестовых данных
  for (int i = 0; i < MAX_TRACKING_OBJECTS; ++i) {
    memset(&Container[i], 0, sizeof(ufo_t));
    Container[i].addr = 0xABC000 + i;
    snprintf(Container[i].flight, sizeof(Container[i].flight), "FL%03d", i);
  }
  memset(&ThisAircraft, 0, sizeof(ThisAircraft));
  ThisAircraft.addr = 0x777777;

  uint32_t t0 = millis();

  for (;;) {
    uint32_t ms = millis() - t0;
    float t = ms / 1000.0f;

    // Обновляем контейнеры
    for (int i = 0; i < MAX_TRACKING_OBJECTS; ++i) {
      ufo_t &u = Container[i];
      u.timestamp = time(nullptr);
      u.latitude  = 55.0f + 0.01f * sinf(t + i);
      u.longitude = 37.0f + 0.01f * cosf(t + i);
      u.old_latitude = u.latitude - 0.0005f;
      u.old_longitude = u.longitude - 0.0005f;
      u.altitude = 1000.0f + 10.0f * i + 5.0f * sinf(t * 0.5f);
      u.course   = fmodf(180.0f + 10.0f * sinf(t + i), 360.0f);
      u.speed    = 120.0f + 2.0f * cosf(t + i);
      u.vert_rate = int(100.0f * sinf(t + i));
      u.Squawk    = 7000 + i;
      u.timemsg   = time(nullptr);
      u.vs        = u.vert_rate;
      u.geoid_separation = 30.0f;
      u.hdop = 85;
      u.rssi = -60;
      u.distance = 5.0f + i;
      u.bearing = 90.0f + i * 10.0f;
      u.signal_source = 1;
      u.seen = time(nullptr);
      u.hour_msg = (uint8_t)((ms/3600000)%24);
      u.min_msg  = (uint8_t)((ms/60000)%60);
      u.delay_time_msg = (uint16_t)(ms % 1000);
      memset(u.callsign, 0, sizeof(u.callsign));
      u.callsign[0] = 'U'; u.callsign[1] = 'F'; u.callsign[2] = 'O'; u.callsign[3] = '0' + i;
    }

    // Обновляем собственный борт
    ThisAircraft.timestamp = time(nullptr);
    ThisAircraft.latitude  = 55.75f + 0.015f * sinf(t);
    ThisAircraft.longitude = 37.61f + 0.015f * cosf(t);
    ThisAircraft.altitude  = 1200.0f + 20.0f * sinf(t * 0.3f);
    snprintf(ThisAircraft.flight, sizeof(ThisAircraft.flight), "OWN%03u", (unsigned)((ms/1000)%999));

    // Аналоговый код (симуляция)
    analog_code_M = (uint16_t)((sinf(t) * 0.5f + 0.5f) * 65535);

    // Доп.флаги
    AuxFlags.new_flag_M = (ms/1000)%2;
    AuxFlags.new_buttton_M = (uint8_t)((ms/1000)%4);
    AuxFlags.setMessageRead_M = ((ms/3000)%2);
    AuxFlags.MessageRead_M = ((ms/5000)%2);
    AuxFlags.SOS_Sprite_on_off_M = ((ms/7000)%2);
    AuxFlags.SOS_View_on_off_M = ((ms/11000)%2);
    AuxFlags.new_SOS_flag_M = ((ms/13000)%2);
    AuxFlags.confirm_message_M = ((ms/17000)%2);
    snprintf(AuxFlags.msg_resp_M, sizeof(AuxFlags.msg_resp_M), "Test msg t=%lu", (unsigned long)(ms/1000));
    AuxFlags.isValidGNSS_M = true;
    AuxFlags.FLYRF_MODE_TEST_M = (uint8_t)((ms/2000)%255);

    vTaskDelay(pdMS_TO_TICKS(100)); // обновление 10 Гц
  }
}

void DeltaCheckTask(void* arg) {
  for (;;) {
    // Проверка контейнеров
    for (int i = 0; i < MAX_TRACKING_OBJECTS; ++i) {
      if (!hasLastContainer[i] || memdiff(&Container[i], &lastSentContainer[i], sizeof(ufo_t))) {
        enqueueFrame(PKT_CONTAINER, (uint8_t)i, &Container[i], sizeof(ufo_t));
        lastSentContainer[i] = Container[i];
        hasLastContainer[i] = true;
      }
    }

    // Проверка собственного борта
    if (!hasLastThis || memdiff(&ThisAircraft, &lastSentThisAircraft, sizeof(ufo_t))) {
      enqueueFrame(PKT_THISAC, 0xFF, &ThisAircraft, sizeof(ufo_t));
      lastSentThisAircraft = ThisAircraft;
      hasLastThis = true;
    }

    // Аналог
    if (!hasLastAnalog || analog_code_M != lastSentAnalog) {
      enqueueFrame(PKT_ANALOG, 0x00, &analog_code_M, sizeof(analog_code_M));
      lastSentAnalog = analog_code_M;
      hasLastAnalog = true;
    }

    // Доп.флаги
    if (!hasLastAux || memdiff(&AuxFlags, &lastSentAux, sizeof(aux_t))) {
      enqueueFrame(PKT_AUX, 0x00, &AuxFlags, sizeof(aux_t));
      lastSentAux = AuxFlags;
      hasLastAux = true;
    }

    // Пульс
    uint8_t hb = 0xA5;
    enqueueFrame(PKT_HEARTBEAT, 0x00, &hb, 1);

    vTaskDelay(pdMS_TO_TICKS(100)); // 10 Гц проверка
  }
}

  Serial.begin(115200);
  delay(500);
  Serial.println("RS485 Source start");

  pinMode(RS485_DE_PIN, OUTPUT);
  rs485SetTX(false);

  RS485_SERIAL.begin(RS485_BAUD, RS485_CONFIG, RS485_RX_PIN, RS485_TX_PIN);

  txQueue = xQueueCreate(32, sizeof(TxItem));
  ackQueue = xQueueCreate(32, sizeof(AckInfo));
  serialMutex = xSemaphoreCreateMutex();

  // Все задачи на ядре 0
  xTaskCreatePinnedToCore(RxTask, "RxTask", 8192, nullptr, 3, nullptr, 0);
  xTaskCreatePinnedToCore(TxTask, "TxTask", 8192, nullptr, 3, nullptr, 0);
  xTaskCreatePinnedToCore(DataGenTask, "DataGenTask", 8192, nullptr, 2, nullptr, 0);
  xTaskCreatePinnedToCore(DeltaCheckTask, "DeltaCheckTask", 8192, nullptr, 2, nullptr, 0);
}

static uint16_t crc16_ibm(const uint8_t* data, size_t len) {
  uint16_t crc = 0xFFFF;
  for (size_t i = 0; i < len; ++i) {
    crc ^= data[i];
    for (int j = 0; j < 8; ++j) {
      if (crc & 1) crc = (crc >> 1) ^ 0xA001;
      else crc >>= 1;
    }
  }
  return crc;
}

static void rs485SetTX(bool enable) {
  digitalWrite(RS485_DE_PIN, enable ? HIGH : LOW);
  if (enable) delayMicroseconds(50);
}

// ============== FreeRTOS объекты (ядро 0) ==============
SemaphoreHandle_t serialMutex;

// ============== Парсер и ACK ===========================
static bool parseOneFrame(Stream& s, FrameHeader& hdr, uint8_t* payloadBuf, uint16_t& crc) {
  static uint8_t state = 0;
  uint8_t b;
  while (s.available()) {
    b = s.read();
    if (state == 0 && b == 0xAA) { state = 1; continue; }
    if (state == 1 && b == 0x55) {
      size_t need = sizeof(FrameHeader) - 2;
      uint8_t* p = ((uint8_t*)&hdr) + 2;
      size_t got = 0;
      while (got < need) {
        int v = s.read();
        if (v < 0) { state = 2; goto WAIT_CONT; }
        p[got++] = (uint8_t)v;
      }
      state = 3;
WAIT_CONT:
      if (state != 3) return false;
      if (hdr.ver != PROTO_VER || hdr.length > MAX_PAYLOAD) {
        state = 0; continue;
      }
      for (uint16_t i = 0; i < hdr.length; ++i) {
        int v = -1; while ((v = s.read()) < 0) { delayMicroseconds(50); }
        payloadBuf[i] = (uint8_t)v;
      }
      uint8_t* c = (uint8_t*)&crc;
      for (int i = 0; i < 2; ++i) {
        int v = -1; while ((v = s.read()) < 0) { delayMicroseconds(50); }
        c[i] = (uint8_t)v;
      }
      state = 0;
      return true;
    } else {
      state = 0;
    }
  }
  return false;
}

static void sendAckLike(uint8_t type, uint8_t index, uint8_t seq, bool positive) {
  uint8_t payload[4];
  payload[0] = type;
  payload[1] = index;
  payload[2] = seq;
  payload[3] = positive ? 0x06 : 0x15;

  FrameHeader hdr;
  hdr.preamble = PREAMBLE;
  hdr.ver = PROTO_VER;
  hdr.type = positive ? PKT_ACK : PKT_NACK;
  hdr.index = 0;
  hdr.seq = 0;
  hdr.length = sizeof(payload);

  uint8_t header_and_payload[sizeof(FrameHeader)-2 + sizeof(payload)];
  memcpy(&header_and_payload[0], &hdr.ver, sizeof(FrameHeader)-2);
  memcpy(&header_and_payload[sizeof(FrameHeader)-2], payload, sizeof(payload));
  uint16_t crc = crc16_ibm(header_and_payload, sizeof(header_and_payload));

  if (xSemaphoreTake(serialMutex, pdMS_TO_TICKS(50)) == pdTRUE) {
    rs485SetTX(true);
    RS485_SERIAL.write((uint8_t*)&hdr, sizeof(hdr));
    RS485_SERIAL.write(payload, sizeof(payload));
    RS485_SERIAL.write((uint8_t*)&crc, sizeof(crc));
    RS485_SERIAL.flush();
    delayMicroseconds(200);
    rs485SetTX(false);
    xSemaphoreGive(serialMutex);
  }
}

// ============== Приёмная задача ========================
void RxTask(void* arg) {
  FrameHeader hdr;
  uint8_t payload[MAX_PAYLOAD];
  uint16_t rxCrc;

  for (;;) {
    if (!RS485_SERIAL.available()) {
      vTaskDelay(pdMS_TO_TICKS(1));
      continue;
    }

    if (!parseOneFrame(RS485_SERIAL, hdr, payload, rxCrc)) {
      vTaskDelay(pdMS_TO_TICKS(1));
      continue;
    }

    uint8_t buf[sizeof(FrameHeader)-2 + MAX_PAYLOAD];
    memcpy(buf, &hdr.ver, sizeof(FrameHeader)-2);
    memcpy(buf + sizeof(FrameHeader)-2, payload, hdr.length);
    uint16_t calc = crc16_ibm(buf, (sizeof(FrameHeader)-2) + hdr.length);

    bool ok = (calc == rxCrc);

    // ACK/NACK для служебных
    if (hdr.type == PKT_ACK || hdr.type == PKT_NACK) {
      // Приёмник служебные подтверждения от источника не использует
      continue;
    }

    if (!ok) {
      sendAckLike(hdr.type, hdr.index, hdr.seq, false);
      Serial.printf("[RX] CRC FAIL type=%02X idx=%u seq=%u
", hdr.type, hdr.index, hdr.seq);
      continue;
    }

    // Обработка полезных данных
    switch (hdr.type) {
      case PKT_CONTAINER: {
        if (hdr.index < MAX_TRACKING_OBJECTS && hdr.length == sizeof(ufo_t)) {
          memcpy(&Container[hdr.index], payload, sizeof(ufo_t));
          Serial.printf("[RX] Container[%u]: lat=%.6f lon=%.6f alt=%.1f flight=%s
",
            hdr.index,
            Container[hdr.index].latitude,
            Container[hdr.index].longitude,
            Container[hdr.index].altitude,
            Container[hdr.index].flight);
          sendAckLike(hdr.type, hdr.index, hdr.seq, true);
        } else {
          sendAckLike(hdr.type, hdr.index, hdr.seq, false);
        }
      } break;

      case PKT_THISAC: {
        if (hdr.length == sizeof(ufo_t)) {
          memcpy(&ThisAircraft, payload, sizeof(ufo_t));
          Serial.printf("[RX] ThisAircraft: lat=%.6f lon=%.6f alt=%.1f flight=%s
",
            ThisAircraft.latitude,
            ThisAircraft.longitude,
            ThisAircraft.altitude,
            ThisAircraft.flight);
          sendAckLike(hdr.type, hdr.index, hdr.seq, true);
        } else {
          sendAckLike(hdr.type, hdr.index, hdr.seq, false);
        }
      } break;

      case PKT_ANALOG: {
        if (hdr.length == sizeof(uint16_t)) {
          memcpy(&analog_code_M, payload, sizeof(uint16_t));
          Serial.printf("[RX] Analog code: %u
", (unsigned)analog_code_M);
          sendAckLike(hdr.type, hdr.index, hdr.seq, true);
        } else {
          sendAckLike(hdr.type, hdr.index, hdr.seq, false);
        }
      } break;

      case PKT_AUX: {
        if (hdr.length == sizeof(aux_t)) {
          memcpy(&AuxFlags, payload, sizeof(aux_t));
          Serial.printf("[RX] AUX: new_flag=%d btn=%u MSG='%s' GNSS=%d MODE=%u
",
            (int)AuxFlags.new_flag_M,
            AuxFlags.new_buttton_M,
            AuxFlags.msg_resp_M,
            (int)AuxFlags.isValidGNSS_M,
            AuxFlags.FLYRF_MODE_TEST_M);
          sendAckLike(hdr.type, hdr.index, hdr.seq, true);
        } else {
          sendAckLike(hdr.type, hdr.index, hdr.seq, false);
        }
      } break;

      case PKT_HEARTBEAT: {
        // Можно ничего не делать, только ACK
        sendAckLike(hdr.type, hdr.index, hdr.seq, true);
      } break;

      default: {
        sendAckLike(hdr.type, hdr.index, hdr.seq, false);
      } break;
    }
  }
}

// ================== Arduino ============================
void setup() {
  Serial.begin(115200);
  delay(500);
  Serial.println("RS485 Receiver start");

  pinMode(RS485_DE_PIN, OUTPUT);
  rs485SetTX(false);

  RS485_SERIAL.begin(RS485_BAUD, RS485_CONFIG, RS485_RX_PIN, RS485_TX_PIN);

  serialMutex = xSemaphoreCreateMutex();

  // Все задачи на ядре 0
  xTaskCreatePinnedToCore(RxTask, "RxTask", 8192, nullptr, 3, nullptr, 0);
}

  // Вся логика в задачах на core 0
}
Примечания по эксплуатации:

На обеих платах убедитесь, что DE и /RE RS‑485 трансивера объединены и заведены на указанный пин (GPIO21 на источнике, GPIO40 на приёмнике).
Общая “земля” между платами обязательна.
Скорость 921600 может потребовать качественных трансиверов/проводки. При проблемах снизьте до 460800 или 230400, изменив RS485_BAUD.
“Только изменённые данные” реализовано по принципу “отправка объекта целиком при изменении”. При желании можно оптимизировать до диффов по полям и/или порогов для float.
Ограничение 0.3 секунды на пакет выдерживается с большим запасом: при 921600 бод и размере до ~200 байт время передачи единичного кадра ~2–3 мс. Даже с повторами и ожиданием ACK в коде установлены таймауты в десятки миллисекунд.
Если нужно, могу:

Добавить журналирование статистики (потери, повторы, задержки).
Ввести пороги изменений для float полей.
Реализовать сборку нескольких объектов в один кадр с фрагментацией.

Принято. Теперь передаём все данные всегда, без проверки изменений. Меняем только источник. Приёмник можно не трогать.

Что именно правим в источнике:

Убираем логику сравнения со “снапшотами” (memdiff, lastSent* и hasLast* можно удалить или оставить неиспользуемыми).
В задаче DeltaCheckTask всегда ставим в очередь все 8 объектов Container[i], затем ThisAircraft, analog_code_M и AuxFlags, плюс Heartbeat — каждый цикл.
Ниже — замена для функции DeltaCheckTask и (по желанию) добавление периода отправки через define.

Вставьте/замените в источнике:

// Частота передачи всех данных
#define TX_PERIOD_MS 100  // 10 Гц; при необходимости увеличить/уменьшить

void DeltaCheckTask(void* arg) {
  const TickType_t period = pdMS_TO_TICKS(TX_PERIOD_MS);
  TickType_t last = xTaskGetTickCount();

  for (;;) {
    // 1) Всегда отправляем все объекты контейнера
    for (int i = 0; i < MAX_TRACKING_OBJECTS; ++i) {
      enqueueFrame(PKT_CONTAINER, (uint8_t)i, &Container[i], sizeof(ufo_t));
    }

    // 2) Всегда отправляем ThisAircraft
    enqueueFrame(PKT_THISAC, 0xFF, &ThisAircraft, sizeof(ufo_t));

    // 3) Всегда отправляем analog_code_M
    enqueueFrame(PKT_ANALOG, 0x00, &analog_code_M, sizeof(analog_code_M));

    // 4) Всегда отправляем AuxFlags
    enqueueFrame(PKT_AUX, 0x00, &AuxFlags, sizeof(aux_t));

    // 5) Heartbeat
    uint8_t hb = 0xA5;
    enqueueFrame(PKT_HEARTBEAT, 0x00, &hb, 1);

    // Периодическая отправка
    vTaskDelayUntil(&last, period);
  }
}
Можно удалить следующие элементы, если больше не нужны:

Функцию memdiff
Массивы/флаги lastSentContainer, hasLastContainer, lastSentThisAircraft/hasLastThis, lastSentAux/hasLastAux, lastSentAnalog/hasLastAnalog
Присваивания lastSent* внутри прежней DeltaCheckTask
Остальное (TxTask, RxTask, протокол, ACK/NACK, контроль CRC, имитация данных, вывод на приёмнике в Serial) — без изменений.

Комментарий по нагрузке:

При TX_PERIOD_MS = 100 мс вы отправляете 8 Container + 1 ThisAircraft + Analog + Aux + Heartbeat = 12 кадров за 100 мс, 
то есть ~120 кадров/сек. На 921600 бод и типичных размерах кадров это далеко в пределах лимитов и 
укладывается в требование “передача пакета не дольше 0,3 с” с большим запасом.
 При необходимости увеличьте период до 50 мс (20 Гц) или уменьшите — обе стороны справятся.
