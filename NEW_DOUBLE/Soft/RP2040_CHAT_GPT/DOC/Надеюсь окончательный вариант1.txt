src/ufo.h
#pragma once
#include <stdint.h>
#include <time.h>

struct DecodedADSB {
  time_t    timestamp;
  uint32_t  addr;
  uint8_t   addr_type;
  float     latitude;
  float     longitude;
  float     altitude;
  float     pressure_altitude;
  float     course;
  float     speed;
  uint8_t   aircraft_type;
  char      flight[16];
  int       vert_rate;
  int       Squawk;
  time_t    seen;
  uint8_t   signal_source;
};
//========================================

src/main.cpp
#include <Arduino.h>
#include "ufo.h"

#define RX_PIN 42
#define TX_PIN 41

void setup() {
  Serial.begin(115200);
  Serial1.begin(115200, SERIAL_8N1, RX_PIN, TX_PIN);
  Serial.println("ESP32-S3 Receiver Ready");
}

void loop() {
  static DecodedADSB fo;
  if (Serial1.available() >= (int)sizeof(fo)) {
    size_t r = Serial1.readBytes((char*)&fo, sizeof(fo));
    if (r == sizeof(fo)) {
      Serial.printf("RX ICAO=%06X FLT=%s LAT=%.6f LON=%.6f ALT(ft)=%.1f SPD(kt)=%.1f TRK=%.1f VR=%d SQ=%d\n",
        fo.addr, fo.flight, fo.latitude, fo.longitude,
        fo.altitude/0.3048f, fo.speed, fo.course, fo.vert_rate, fo.Squawk);
    }
  }
}

Примечания по сборке/запуску

Если используете Arduino IDE:
— создайте папки/файлы как выше в одном скетче (можно как вкладки).
— Файл .pio Arduino распознаёт и сам вызовет pioasm. Заглушечный .pio.h у нас пустой — допустимо.

Скорость семплирования/временные окна:
— В проекте выставлено 2 МГц (шаг 0.5 мкс), этого достаточно под условие задачи.
— При сильных помехах повысьте BASE_THR, K_THR, включите профиль PROFILE_HIGH_EMI (можно добавить обработку профилей в setup() и через Serial командой).

CPR: реализованы global/локальная реконструкция. Пара even/odd кэшируется 10 с.
Укажите ваши REF_LAT/REF_LON в config.h.

Gillham: реализованы Q=1 (25 ft) и Q=0 (AC13 Gray) алгоритмически. Для предельной точности можно добавить таблицы исключений («дыры»).

CRC и одно-битная коррекция: выполняется в сборщике; если удачно — пакет сохраняется.

RAW/JSON: до CRC мы уже не печатаем «сырые» биты (т.к. CRC проверен в сборке), но включён OUTPUT_RAW_BEFORE_CRC для отладки — формат MSB-first.

RSSI: analogRead(RSSI_PIN) читается на каждый пакет и кладётся в AdsbPacket.rssi (можете нормировать/усреднять).

Dead time, маски и EMA: настроены константами, меняйте в config.h.

Три канала параллельно: три SM PIO + три DMA кольца; простая обработка без IRQ DMA. Для боевой версии добавьте IRQ и ведите точный указатель wIdx из DMA callback.

Выбор профилей/команд через USB: можно расширить, добавив парсер команд в loop().
