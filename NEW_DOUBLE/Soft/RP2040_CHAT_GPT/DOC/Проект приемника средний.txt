#include "adsb_crc.h"

// Mode-S CRC-24 generator poly
static const uint32_t GEN = 0xFFF409u;

uint32_t adsb_crc24(const uint8_t *msg, size_t nbits) {
  // bits without the last 24 parity bits
  size_t paybits = nbits - 24;
  uint32_t crc = 0;
  for (size_t i=0;i<paybits;i++) {
    size_t byte = i >> 3;
    int bit = 7 - (i & 7);
    uint8_t b = (msg[byte] >> bit) & 1;
    if (((crc >> 23) & 1) ^ b) crc = ((crc << 1) ^ GEN) & 0xFFFFFFu;
    else crc = (crc << 1) & 0xFFFFFFu;
  }
  return crc;
}

bool adsb_parity_ok(const uint8_t *msg, size_t nbits) {
  size_t nb = nbits/8;
  uint32_t p = (msg[nb-3]<<16) | (msg[nb-2]<<8) | msg[nb-1];
  return adsb_crc24(msg, nbits) == p;
}

bool adsb_fix_single_bit(uint8_t *msg, size_t nbits) {
  if (adsb_parity_ok(msg, nbits)) return true;
  // try single-bit flip across payload bits only
  size_t paybits = nbits - 24;
  for (size_t i=0;i<paybits;i++) {
    size_t byte = i >> 3;
    int bit = 7 - (i & 7);
    msg[byte] ^= (1u << bit);
    if (adsb_parity_ok(msg, nbits)) return true;
    msg[byte] ^= (1u << bit); // revert
  }
  return false;
}

float cpr_dlon(float lat, bool odd) {
  int nl = NL(lat);
  if (nl > 0) return 360.0f/(nl - (odd?1:0));
  return 360.0f;
}

bool cpr_global_decode(const cpr_pair_t* pair, float* lat, float* lon) {
  if (!(pair->have_even && pair->have_odd)) return false;
  const float Dlat0 = 360.0f/60.0f;
  const float Dlat1 = 360.0f/59.0f;
  float lat0 = Dlat0 * (pair->lat_even / 131072.0f);
  float lat1 = Dlat1 * (pair->lat_odd  / 131072.0f);
  int j = floorf(59*lat0 - 60*lat1 + 0.5f);
  float rlat0 = Dlat0 * (cpr_mod(j,60) + pair->lat_even/131072.0f);
  float rlat1 = Dlat1 * (cpr_mod(j,59) + pair->lat_odd /131072.0f);
  if (rlat0 >= 270) rlat0 -= 360;
  if (rlat1 >= 270) rlat1 -= 360;
  float rlat = (pair->t_odd_ms > pair->t_even_ms) ? rlat1 : rlat0;

  int nl = NL(rlat);
  if (nl == 0) return false;
  float Dlon = 360.0f/(nl - ((pair->t_odd_ms>pair->t_even_ms)?1:0));
  float lon0 = pair->lon_even/131072.0f;
  float lon1 = pair->lon_odd /131072.0f;
  int m = floorf(lon0*(nl-1) - lon1*nl + 0.5f);
  float rlon = Dlon * (cpr_mod(m, nl) + ((pair->t_odd_ms>pair->t_even_ms)?lon1:lon0));
  if (rlon > 180) rlon -= 360;
  *lat = rlat; *lon = rlon;
  return true;
}

bool cpr_local_decode(float refLat, float refLon, uint32_t cprLat, uint32_t cprLon, bool odd, float* outLat, float* outLon) {
  float Dlat = odd ? (360.0f/59.0f) : (360.0f/60.0f);
  float lat = Dlat * ( (int)floorf(refLat/Dlat) + cprLat/131072.0f );
  if (lat >= 270) lat -= 360;
  int nl = NL(lat) - (odd?1:0);
  float Dlon = (nl>0) ? 360.0f/nl : 360.0f;
  float lon = Dlon * ( (int)floorf(refLon/Dlon) + cprLon/131072.0f );
  if (lon > 180) lon -= 360;
  *outLat = lat; *outLon = lon;
  return true;
}

static inline uint32_t be24(const uint8_t* p){ return (p[0]<<16)|(p[1]<<8)|p[2]; }
static inline uint32_t be32(const uint8_t* p){ return ((uint32_t)p[0]<<24)|((uint32_t)p[1]<<16)|((uint32_t)p[2]<<8)|p[3]; }

static uint32_t get_bits(const uint8_t* b, int start, int len) {
  // start: bit index from MSB=0 across entire frame
  uint32_t v=0;
  for (int i=0;i<len;i++) {
    int bit = start+i;
    int byte = bit>>3;
    int off = 7-(bit&7);
    v = (v<<1) | ((b[byte]>>off)&1);
  }
  return v;
}

static float kts_to_kmh(float k){ return k*1.852f; }

// altitude decode (DF17, TC=9/10/11 field)
static int32_t decode_altitude_m(uint32_t altfield) {
  // 13 bits, bit Q at position 4 (from MSB): b(8..0) but in standard: use bitmaps.
  // Quick path: if Q=1, N= (AC13 >> 1) * 25 - 1000 ft
  bool q = (altfield & 0x10) != 0;
  if (q) {
    int N = (( (altfield & 0x0F) | ((altfield & 0x1FE0)>>1) ) ); // collapse bit hole
    int ft = N*25 - 1000;
    return (int32_t)lrintf(ft * 0.3048f);
  } else {
    int ft = gillham_ac13_to_alt_ft((uint16_t)altfield, false);
    if (ft == INT32_MIN) return INT32_MIN;
    return (int32_t)lrintf(ft * 0.3048f);
  }
}

// callsign (TC 1-4)
static void decode_callsign(const uint8_t* b, char out[16]) {
  static const char tbl[] = "#ABCDEFGHIJKLMNOPQRSTUVWXYZ#####_###############0123456789######";
  // ME[1..6] bytes; 8 6-bit chars
  uint64_t v = ((uint64_t)b[4]<<40)|((uint64_t)b[5]<<32)|((uint64_t)b[6]<<24)|((uint64_t)b[7]<<16)|((uint64_t)b[8]<<8)|b[9];
  for (int i=0;i<8;i++){
    int c = (v >> (42 - i*6)) & 0x3F;
    out[i] = (c< (int)sizeof(tbl)-1) ? tbl[c] : ' ';
  }
  out[8]=0;
  // copy into 16-char buffer padded with spaces
  memset(out+8,' ',7); out[15]=0;
}

// TC=19 velocity
static void decode_tc19(const uint8_t* b, uint16_t* speed_kmh, uint16_t* track_deg, int16_t* vr_mpm) {
  uint8_t subtype = (get_bits(b, 37, 3)); // ME[1] bits 5..7
  *vr_mpm = 0;
  if (subtype==1 || subtype==2){
    // Ground speed (vector), east/west, north/south
    int ew_dir = get_bits(b, 46,1);
    int ew_spd = get_bits(b, 47,10);
    int ns_dir = get_bits(b, 57,1);
    int ns_spd = get_bits(b, 58,10);
    if (ew_spd==0 || ns_spd==0) { *speed_kmh=0; *track_deg=0; return; }
    float ve = (ew_spd-1) * (ew_dir? -1.0f : 1.0f);
    float vn = (ns_spd-1) * (ns_dir? -1.0f : 1.0f);
    float v = sqrtf(ve*ve + vn*vn);
    float trk = atan2f(ve, vn) * 180.0f/M_PI; if (trk<0) trk+=360.0f;
    *speed_kmh = (uint16_t)lrintf(kts_to_kmh(v));
    *track_deg = (uint16_t)lrintf(trk);
    int vr_sgn = get_bits(b, 68,1) ? -1 : 1;
    int vr = get_bits(b, 69,9) - 1;
    *vr_mpm = (int16_t)(vr_sgn * vr * 64 * 0.3048f * 3.28084f); // (ft/min)≈64fpm steps => placeholder
  } else if (subtype==3 || subtype==4){
    // Heading + ground/true speed
    int hdg_stat = get_bits(b, 46,1);
    int hdg = get_bits(b, 47,10);
    int spd_typ = get_bits(b, 57,1); // 0=GS,1=TAS
    int spd = get_bits(b, 58,10);
    float track = (hdg/1024.0f)*360.0f;
    *track_deg = (uint16_t)lrintf(track);
    *speed_kmh = (uint16_t)lrintf(kts_to_kmh(spd-1));
    int vr_sgn = get_bits(b, 68,1) ? -1 : 1;
    int vr = get_bits(b, 69,9) - 1;
    *vr_mpm = (int16_t)(vr_sgn * vr); // leave in raw steps, adjust upstream if needed
  } else {
    *speed_kmh=0; *track_deg=0; *vr_mpm=0;
  }
}

static void format_squawk(uint16_t code, char out[8]) {
  // 13-bit squawk Gillham-like -> 4 octal
  out[0] = ((code>>9)&7)+'0';
  out[1] = ((code>>6)&7)+'0';
  out[2] = ((code>>3)&7)+'0';
  out[3] = ((code>>0)&7)+'0';
  out[4]=0;
}

// High-level decoder (DF17/DF18 and partial DF20/21 BDS)
bool adsb_decode(const adsb_packet_t* pkt, adsb_decoded_t* out) {
  if (!pkt || !out) return false;
  // Optionally dump raw before CRC
  dump_raw(pkt);

  // CRC and single-bit correction
  uint8_t data[MAX_BYTES]; memcpy(data, pkt->data, sizeof(data));
  bool ok = adsb_parity_ok(data, pkt->nbits);
  bool corrected = false;
  if (!ok) {
    // Try single-bit correction
    corrected = adsb_fix_single_bit(data, pkt->nbits);
    ok = corrected;
  }
  if (!ok) return false;

  // Basic header
  uint8_t df = (data[0] >> 3) & 0x1F;
  uint8_t ca = data[0] & 0x07;
  (void)ca;

  uint32_t icao = be24(&data[1]); // for DF17/18 after CRC removal this may differ; proceed as common SDR method

  memset(out, 0, sizeof(*out));
  out->addr = icao;
  out->seen_time = millis();
  out->type = 0;

  if (df==17 || df==18) {
    uint8_t tc = (data[4] >> 3) & 0x1F;

    if (tc>=1 && tc<=4) { // callsign
      decode_callsign(data, out->flight);
    }
    else if (tc==19) { // velocity
      uint16_t spd, hdg; int16_t vr;
      decode_tc19(data, &spd, &hdg, &vr);
      out->speed = spd;
      out->track = hdg;
      out->vert_rate = vr;
    }
    else if (tc==9 || tc==10 || tc==11) { // airborne position
      bool odd = (data[6] & 0x04) != 0; // time flag
      uint32_t altfield = ((data[5]&0x07)<<8) | data[6]; // rough, will re-extract precise bits
      // Proper altitude field (13 bits in ME[3..5]): bits 19..31 overall ME
      uint32_t re_alt = ((get_bits(data, 40, 7) << 6) | get_bits(data, 48,6)); // AC13 mixed; Q @ bit 51
      int32_t alt_m = decode_altitude_m(re_alt);
      if (alt_m != INT32_MIN) out->altitude = alt_m;

      uint32_t cprLat = (get_bits(data, 54, 17));
      uint32_t cprLon = (get_bits(data, 71, 17));

      // pair store
      ICAOPair &pair = g_pairs[out->addr & 0xFF];
      if (odd) { pair.pair.have_odd=true;  pair.pair.t_odd_ms=millis(); pair.pair.lat_odd=cprLat; pair.pair.lon_odd=cprLon; }
      else     { pair.pair.have_even=true; pair.pair.t_even_ms=millis(); pair.pair.lat_even=cprLat; pair.pair.lon_even=cprLon; }
      pair.pair.icao = out->addr;

      // Try global (if both within 10s)
      if (pair.pair.have_even && pair.pair.have_odd &&
          (abs((int)pair.pair.t_even_ms - (int)pair.pair.t_odd_ms) <= 10000)) {
        float lat, lon;
        if (cpr_global_decode(&pair.pair, &lat, &lon)) { out->lat=lat; out->lon=lon; }
      } else {
        // Fallback local decode
        cpr_local_decode(g_ref_lat, g_ref_lon, cprLat, cprLon, odd, &out->lat, &out->lon);
      }
    }
    else if (tc==5 || tc==6 || tc==7 || tc==8) { // surface pos: can decode similarly if needed
      // Not fully implemented here
    }

    // Squawk (DF17 extended?) Typically DF5/DF21 hold identity; DF17 has ID only in BDS 1,0 not included here
    // We'll leave squawk empty unless DF 5/21 appear (handled below).
  }
  else if (df==5 || df==21) {
    // Identity reply: squawk code in 13 bits
    uint16_t id = (uint16_t)get_bits(data, 19, 13);
    char sq[8]={0}; format_squawk(id, sq);
    strncpy(out->flight, sq, sizeof(out->flight));
  }
  else if (df==20 || df==21) {
    // BDS 6,0/6,1 skeleton (baro-geo diff, VS) – requires IC bit fields per transponder implementation
    // TODO: parse specific fields when ME field indicates 6,0/6,1
  }

  // default types
  if (out->type==0) out->type = 9;

  // Coherence: when only velocity seen, we still output partial
  return true;
}

.program adsb_sampler
; Sample a single GPIO into RX FIFO at fixed rate.
; Each PUSH transfers 32 samples (bits) to FIFO (packed).
; SM clock is configured externally to SAMPLE_HZ * 32 (since we collect 1 bit/clk and then push every 32)
; Pin is configured externally via sideset/pindir; here we read IN pins.

.pull block         ; init Y = 31 for loop count
mov y, ~null

.wrap_target
  set x, 31        ; 32-bit pack
pack_loop:
  in pins, 1       ; shift 1 sample into ISR (LSB-first accumulation in ISR)
  jmp x-- pack_loop
  push block       ; push 32 bits
  jmp y-- wrap_target
.wrap

// ---------- DMA ring buffers per channel ----------
static uint32_t __attribute__((aligned(4))) ring1[RING_BYTES/4];
static uint32_t __attribute__((aligned(4))) ring2[RING_BYTES/4];
static uint32_t __attribute__((aligned(4))) ring3[RING_BYTES/4];

static volatile size_t r1_w=0, r2_w=0, r3_w=0;

static PIO pio_inst = pio0;
static int sm1, sm2, sm3;
static int dma1=-1, dma2=-1, dma3=-1;

static void dma_setup_channel(int &dma, int sm, uint32_t* ring, volatile size_t* wridx){
  dma = dma_claim_unused_channel(true);
  dma_channel_config c = dma_channel_get_default_config(dma);
  channel_config_set_transfer_data_size(&c, DMA_SIZE_32);
  channel_config_set_read_increment(&c, false);
  channel_config_set_write_increment(&c, true);
  channel_config_set_dreq(&c, pio_get_dreq(pio_inst, sm, false));
  dma_channel_configure(dma, &c, ring, &pio_inst->rxf[sm], RING_BYTES/4, false);
  *wridx = 0;
  dma_hw->ints0 = 1u<<dma; // clear
  dma_channel_set_irq0_enabled(dma, true);
}

static void irq_dma_handler() {
  uint32_t st = dma_hw->ints0;
  if (st & (1u<<dma1)) {
    dma_hw->ints0 = (1u<<dma1);
    r1_w = (r1_w + RING_BYTES/4) % (RING_BYTES/4);
    dma_channel_set_write_addr(dma1, &ring1[0], true);
  }
  if (st & (1u<<dma2)) {
    dma_hw->ints0 = (1u<<dma2);
    r2_w = (r2_w + RING_BYTES/4) % (RING_BYTES/4);
    dma_channel_set_write_addr(dma2, &ring2[0], true);
  }
  if (st & (1u<<dma3)) {
    dma_hw->ints0 = (1u<<dma3);
    r3_w = (r3_w + RING_BYTES/4) % (RING_BYTES/4);
    dma_channel_set_write_addr(dma3, &ring3[0], true);
  }
}

static void pio_start_sm(PIO pio, int sm, uint pin){

  pio_sm_set_clkdiv(pio, sm, div);
  pio_sm_set_enabled(pio, sm, true);
}

// ---------- Preamble correlator (80-sample weighted) ----------
static bool corr_preamble_and_get_nbits(const uint32_t* buf, size_t bitpos, uint8_t* nbits) {
  // Very compact placeholder: try to detect both (DF=short/long) windows
  // Real implementation should compute score over 80-sample mask with penalties.
  // Here we accept preamble and choose length by following energy window check.
  *nbits = DF_LONG_BITS; // try long by default
  return true;
}

// ---------- Bitstream to bytes ----------
static void extract_bits_to_bytes(const uint32_t* buf, size_t startBit, uint8_t nbits, uint8_t* out) {
  // buf stores 32 samples (bits) per uint32_t, MSB-first from IN pins
  // We consider each bit==1 sample as high; NRZ sampling (already filtered by front-end).
  memset(out,0,(nbits+7)/8);
  for (size_t i=0;i<nbits;i++){
    size_t pos = startBit + i;
    size_t w = pos >> 5;
    int bit = 31 - (pos & 31);
    int v = (buf[w] >> bit) & 1;
    if (v) out[i>>3] |= (1u << (7-(i&7)));
  }
}

// ---------- Channel sampler/decoder (runs on core1) ----------
static void channel_worker(uint8_t chan, volatile size_t* wridx, uint32_t* ring, uint gpio_led) {
  size_t rd = 0;
  uint32_t last_t0_us = 0;
  for(;;){
    // scan ring buffer for preamble signature (placeholder fast path)
    if (rd != *wridx) {
      // quick mock: treat every 400 bits as a candidate to keep demo going
      size_t startBit = rd*32;
      uint8_t nbits=0;
      if (corr_preamble_and_get_nbits(ring, startBit+PREAMBLE_SAMPLES, &nbits)) {
        uint32_t nowus = micros();
        if (nowus - last_t0_us > DEAD_TIME_US) {
          GPIO_SET(gpio_led);
          adsb_packet_t p={0};
          p.chan = chan;
          p.t0_us = nowus - PREAMBLE_SAMPLES/4; // rough
          p.nbits = nbits;
          extract_bits_to_bytes(ring, startBit+PREAMBLE_SAMPLES, nbits, p.data);
          p.rssi = analogRead(PIN_RSSI);
          p.crc_ok = adsb_parity_ok(p.data, p.nbits);
          if (!p.crc_ok) p.corrected = adsb_fix_single_bit(p.data, p.nbits);
          enqueue_packet_from_core1(p);
          GPIO_CLR(gpio_led);
          last_t0_us = nowus;
        }
      }
      rd = (rd + 1) % (RING_BYTES/4);
    } else {
      tight_loop_contents();
    }
  }
}

void setup1(){

  pinMode(LED_PRE1, OUTPUT);
  pinMode(LED_PRE2, OUTPUT);
  pinMode(LED_PRE3, OUTPUT);

  // Init PIO state machines

  pio_start_sm(pio_inst, sm2, PIN_CH2);
  pio_start_sm(pio_inst, sm3, PIN_CH3);

  // DMA
  irq_set_exclusive_handler(DMA_IRQ_0, irq_dma_handler);
  irq_set_enabled(DMA_IRQ_0, true);
  dma_setup_channel(dma1, sm1, ring1, &r1_w);
  dma_setup_channel(dma2, sm2, ring2, &r2_w);
  dma_setup_channel(dma3, sm3, ring3, &r3_w);

  dma_channel_start(dma1);
  dma_channel_start(dma2);
  dma_channel_start(dma3);

  // Heartbeat on core1
  absolute_time_t t = get_absolute_time();
  for(;;){
    // background processing: drive per-channel workers a bit each cycle
    static uint32_t slice=0;
    slice++;
    channel_worker(1, &r1_w, ring1, LED_PRE1); // note: simplistic; in production spawn 3 workers
    channel_worker(2, &r2_w, ring2, LED_PRE2);
    channel_worker(3, &r3_w, ring3, LED_PRE3);

    // toggle LED every 500 ms
    if (to_ms_since_boot(t) % 500 == 0) GPIO_TGL(LED_CORE1);
  }
}

#include "adsb_crc.h"

// Таблица для CRC-24 (ADS-B)
static const uint32_t crcTable[256] = {
  // TODO: заполнить таблицу
};

static uint32_t adsb_crc24(const uint8_t *data, size_t len) {
  uint32_t crc = 0;
  for (size_t i = 0; i < len; i++) {
    uint8_t idx = ((crc >> 16) ^ data[i]) & 0xFF;
    crc = (crc << 8) ^ crcTable[idx];
    crc &= 0xFFFFFF;
  }
  return crc;
}

bool adsb_check_crc(const AdsbPacket &pkt) {
  size_t nbytes = pkt.bitlen / 8;
  if (nbytes < 3) return false;
  uint32_t crc = adsb_crc24(pkt.raw, nbytes - 3);
  uint32_t received = (pkt.raw[nbytes - 3] << 16) |
                      (pkt.raw[nbytes - 2] << 8) |
                      (pkt.raw[nbytes - 1]);
  return crc == received;
}

bool adsb_try_correct(AdsbPacket &pkt) {
  size_t nbytes = pkt.bitlen / 8;
  for (size_t byte = 0; byte < nbytes; byte++) {
    for (uint8_t bit = 0; bit < 8; bit++) {
      pkt.raw[byte] ^= (1 << bit);
      if (adsb_check_crc(pkt)) return true;
      pkt.raw[byte] ^= (1 << bit); // откат
    }
  }
  return false;
}

.program adsb_sampler
; Простая программа захвата уровня с пина каждые 0.5 мкс
; Частота системного такта 125 МГц → 1 такт = 8 нс
; Для дискретизации 0.5 мкс = 62 такта
; Мы будем использовать set pindirs + in pins

; захватывает один бит с input pin → в FIFO
; ожидание ~62 тактов (0.496 мкс)

loop:
    in pins, 1       ; захватить 1 бит в ISR
    push block       ; отправить в FIFO
    jmp loop


⚠️ В реальном проекте понадобится развести три программы (по одной на канал, т.к. у RP2040 один state machine = один pin или pin group).
 Здесь одна и та же программа будет собрана трижды для пинов 18, 19 и 22.

// Вспомогательные макросы для доступа к битам
static inline uint32_t get_bits(const uint8_t *data, int start, int len) {
    uint32_t val = 0;
    for (int i = 0; i < len; i++) {
        int bitpos = start + i;
        int byte = bitpos / 8;
        int bit = 7 - (bitpos % 8);
        if (data[byte] & (1 << bit)) {
            val |= (1 << (len - 1 - i));
        }
    }
    return val;
}

// ------------------------------------------------------------

uint32_t adsb_get_icao(const AdsbPacket &pkt) {
    // Биты 9–32 в DF17 — ICAO-адрес
    return get_bits(pkt.raw, 8, 24);
}

bool adsb_get_flight(const AdsbPacket &pkt, char *dst, size_t dstlen) {
    // Тип сообщений DF17: ME поле = 56 бит
    // Flight ID: тип кодирования TC=1–4
    uint8_t tc = get_bits(pkt.raw, 32, 5);
    if (tc < 1 || tc > 4) return false;

    memset(dst, 0, dstlen);
    for (int i = 0; i < 8 && i < (int)dstlen - 1; i++) {
        uint8_t c = get_bits(pkt.raw, 40 + i * 6, 6);
        dst[i] = (c >= 1 && c <= 26) ? ('A' + c - 1) :
                 (c >= 48 && c <= 57) ? ('0' + c - 48) : ' ';
    }
    return true;
}

float adsb_get_altitude(const AdsbPacket &pkt) {
    uint8_t tc = get_bits(pkt.raw, 32, 5);
    if (tc == 9 || tc == 11) {
        // Altitude encoded in 13 bits
        uint16_t altbits = get_bits(pkt.raw, 40, 13);
        // Проверяем Q-бит
        if (altbits & (1 << 4)) {
            int n = ((altbits & 0x1F0) >> 1) | (altbits & 0xF);
            return n * 25 * 0.3048; // в метры
        } else {
            return gillham2alt(altbits) * 0.3048;
        }
    }
    return NAN;
}

float adsb_get_speed(const AdsbPacket &pkt) {
    uint8_t tc = get_bits(pkt.raw, 32, 5);
    if (tc == 19) {
        uint8_t subtype = get_bits(pkt.raw, 37, 3);
        if (subtype == 1 || subtype == 2) {
            // Векторная скорость
            int ew = get_bits(pkt.raw, 45, 10);
            int ns = get_bits(pkt.raw, 55, 10);
            float vx = (ew & 0x200) ? -(ew & 0x1FF) : (ew & 0x1FF);
            float vy = (ns & 0x200) ? -(ns & 0x1FF) : (ns & 0x1FF);
            float v = sqrt(vx * vx + vy * vy);
            return v * 1.852; // knots → km/h
        }
    }
    return NAN;
}

float adsb_get_course(const AdsbPacket &pkt) {
    uint8_t tc = get_bits(pkt.raw, 32, 5);
    if (tc == 19) {
        uint8_t subtype = get_bits(pkt.raw, 37, 3);
        if (subtype == 1 || subtype == 2) {
            int ew = get_bits(pkt.raw, 45, 10);
            int ns = get_bits(pkt.raw, 55, 10);
            float vx = (ew & 0x200) ? -(ew & 0x1FF) : (ew & 0x1FF);
            float vy = (ns & 0x200) ? -(ns & 0x1FF) : (ns & 0x1FF);
            if (vx == 0 && vy == 0) return NAN;
            return atan2(vx, vy) * 180.0 / PI;
        }
    }
    return NAN;
}

int adsb_get_vert_rate(const AdsbPacket &pkt) {
    uint8_t tc = get_bits(pkt.raw, 32, 5);
    if (tc == 19) {
        int vr = get_bits(pkt.raw, 77, 9);
        int sign = get_bits(pkt.raw, 76, 1) ? -1 : 1;
        return sign * vr * 64; // ft/min
    }
    return 0;
}

bool adsb_decode(const AdsbPacket &pkt, AdsbFrame &frame) {
    memset(&frame, 0, sizeof(frame));

    frame.addr = adsb_get_icao(pkt);
    adsb_get_flight(pkt, frame.flight, sizeof(frame.flight));
    frame.squawk = adsb_get_squawk(pkt);

    frame.speed = adsb_get_speed(pkt);

    frame.lon = lon;

    frame.aircraft_type = 9;
    frame.signal_source = 1;
    frame.seen_time = millis();
    frame.timestamp = millis();

    return true;

Блок 12. main.cpp (для ESP32S3)

#include <Arduino.h>
#include "ufo.h"

// Конфиг UART
#define UART_RX 42
#define UART_TX 41
#define UART_BAUD 115200

HardwareSerial SerialADS(2);

void setup() {
    Serial.begin(115200);
    SerialADS.begin(UART_BAUD, SERIAL_8N1, UART_RX, UART_TX);

    Serial.println("ESP32S3 ADS-B Receiver started...");
}

void loop() {
    static UfoFrame frame;

    // Ждём полный пакет
    if (SerialADS.available() >= sizeof(UfoFrame)) {
        size_t n = SerialADS.readBytes((uint8_t*)&frame, sizeof(UfoFrame));
        if (n == sizeof(UfoFrame)) {
            // Выводим в читаемом виде
            Serial.print("ICAO: ");
            Serial.print(frame.addr, HEX);
            Serial.print("  Flight: ");
            Serial.print(frame.flight);
            Serial.print("  Alt: ");
            Serial.print(frame.altitude);
            Serial.print(" m  Speed: ");
            Serial.print(frame.speed);
            Serial.print(" km/h  Lat: ");
            Serial.print(frame.latitude, 6);
            Serial.print("  Lon: ");
            Serial.print(frame.longitude, 6);
            Serial.print("  Course: ");
            Serial.print(frame.course);
            Serial.print("  VR: ");
            Serial.println(frame.vert_rate);
        }
    }
}

Как это работает

RP2040

Принимает сигналы от SDR/приёмника на PIO DMA.

Выделяет импульсы → собирает пакет ADS-B → проверяет CRC.

Декодирует в AdsbFrame.

Отправляет структуру в ESP32S3 по Serial2.

ESP32S3

Принимает UfoFrame.

Отображает его в Serial USB.

Можно расширить: отправлять JSON, NMEA, CSV и т.д.
