// RP2040 ADS-B Receiver (PIO+DMA 3ch) + CPR/Gillham + UART2 to ESP32-S3
#include <Arduino.h>
#include <pico/stdlib.h>

// Pins
#define CH1_IN   18
#define CH2_IN   19
#define CH3_IN   22
#define LED_PREAMBLE1 17
#define LED_PREAMBLE2 20
#define LED_PREAMBLE3 23
#define LED_CORE0 15
#define LED_CORE1 25
#define RSSI_PIN  26 // ADC0

// UART2 to ESP32S3: TX=4, RX=5, 115200
#define UART2_TX 4
#define UART2_RX 5

// Core1: blink 500 ms
void core1_main() {
  gpio_init(LED_CORE1);
  gpio_set_dir(LED_CORE1, GPIO_OUT);
  while (true) {
    gpio_xor_mask(1u << LED_CORE1);
    sleep_ms(500);
  }
}

void blinkTaskCore0(void*) {
  gpio_init(LED_CORE0);
  gpio_set_dir(LED_CORE0, GPIO_OUT);
  while (true) {
    gpio_xor_mask(1u << LED_CORE0);
    vTaskDelay(pdMS_TO_TICKS(1000));
  }
}

static inline uint32_t millis32() { return (uint32_t)millis(); }

void setup() {
  // USB Serial (отладка + команды)
  Serial.begin(115200);
  delay(200);
  Serial.println("RP2040 ADS-B Receiver Start");

  // UART2 к ESP32-S3
  Serial2.setTX(UART2_TX);
  Serial2.setRX(UART2_RX);
  Serial2.begin(115200);

  // RSSI ADC
  analogReadResolution(12);
  analogRead(RSSI_PIN);

  // Индикация преамбулы
  gpio_init(LED_PREAMBLE1); gpio_set_dir(LED_PREAMBLE1, GPIO_OUT); gpio_put(LED_PREAMBLE1, 0);
  gpio_init(LED_PREAMBLE2); gpio_set_dir(LED_PREAMBLE2, GPIO_OUT); gpio_put(LED_PREAMBLE2, 0);
  gpio_init(LED_PREAMBLE3); gpio_set_dir(LED_PREAMBLE3, GPIO_OUT); gpio_put(LED_PREAMBLE3, 0);

  // Модули декодирования
  adsb_gillham_init();
  adsb_set_local_ref(55.75, 37.62); // Москва по умолчанию; поменяйте на свои координаты
  adsb_set_pair_window_ms(10000);
  adsb_reset_cpr_cache();

  // Коррелятор/настройка профиля
  CorrProfile prof{};
  prof.enable_filter = true;
  prof.sample_rate_hz = 10000000; // 10 MHz
  prof.dead_time_us = 8;          // 8 мкс
  prof.base_thr = 6;              // стартовое значение (подбирается)
  prof.k_ema = 2;                 // усиление EMA шума
  prof.min_pos_hits = 2;
  set_corr_profile(prof);

  // DMA + PIO: 3 канала сборки сэмплов
  adsb_dma_init(CH1_IN, CH2_IN, CH3_IN, prof.sample_rate_hz);

  // Core1: мерцание 500 мс
  multicore_launch_core1(core1_main);
  // Core0: мерцание 1000 мс (FreeRTOS)
  xTaskCreatePinnedToCore(blinkTaskCore0, "blink0", 2048, NULL, 1, NULL, 0);

  // Тестовый стенд (самопроверки Gillham/CPR/TC19/Squawk)
  extern void runTests();
  runTests();
}

static void indicatePreambleLED(int ch) {
  uint32_t pin = (ch==0)?LED_PREAMBLE1 : (ch==1)?LED_PREAMBLE2 : LED_PREAMBLE3;
  // короткий импульс 0.5 мс
  gpio_put(pin, 1);
  busy_wait_us(100);
  gpio_put(pin, 0);
}

void loop() {
  // 1) Считываем RSSI
  int rssi_raw = analogRead(RSSI_PIN); // 0..4095

  // 2) Пытаемся извлечь пакет из любого из 3 DMA-колец
  for (int ch=0; ch<3; ++ch) {
    AdsbPacket pkt{};
    // попытка получить t0 (начало преамбулы) из коррелятора
    if (adsb_try_extract_packet(ch, pkt)) {
      // индикация
      indicatePreambleLED(ch);

      // 3) CRC14/24 проверка (Mode S parity)
      if (!adsb_crc_check(pkt.raw, pkt.bits)) {
        // можно раскомментировать отладку «сырых» без CRC
        // Serial.println(F("CRC fail, raw dump:"));
        // for (int i=0;i<(pkt.bits+7)/8;i++) Serial.printf("%02X", pkt.raw[i]); Serial.println();
        continue;
      }

      // 4) Полный декод
      DecodedADSB fo = adsb_decode_packet(pkt);

      // 5) Передача в ESP32-S3
      Serial2.write((uint8_t*)&fo, sizeof(fo));

      // 6) Отладка в USB
      Serial.printf("CH%d RSSI=%d ICAO=%06X FLT=%s ALT=%.0fft LAT=%.5f LON=%.5f SPD=%.1f CRSE=%.1f SQ=%d\n",
        ch+1, rssi_raw, fo.addr, fo.flight, fo.altitude, fo.latitude, fo.longitude, fo.speed, fo.course, fo.Squawk);
    }
  }
}

#pragma once
#include <stdint.h>
#include <time.h>

typedef struct {
    uint8_t  raw[14];   // 112 бит макс
    uint16_t bits;      // 56 или 112
    int8_t   rssi;      // dBFS условно, можно не заполнять
    uint32_t timestamp; // ms
} AdsbPacket;

typedef struct {
    time_t    timestamp;
    uint32_t  addr;
    uint8_t   addr_type;
    float     latitude;
    float     longitude;
    float     altitude;           // ft
    float     pressure_altitude;  // ft (можно дублировать altitude)
    float     course;             // deg
    float     speed;              // kt
    uint8_t   aircraft_type;
    char      flight[16];         // позывной
    int       vert_rate;          // fpm
    int       Squawk;             // Mode A идентификатор
    time_t    seen;
    uint8_t   signal_source;      // 1 = ADS-B
    int8_t    rssi;               // сырая оценка, если доступна
} DecodedADSB;

static double g_ref_lat = 55.0, g_ref_lon = 37.0;

struct CprSlot {
  uint32_t icao;
  bool has_even, has_odd;
  uint32_t t_even_ms, t_odd_ms;
  uint32_t lat_even, lon_even, lat_odd, lon_odd;
  double   last_lat, last_lon;
};
#define CPR_CACHE_SLOTS 64
static CprSlot g_cpr_cache[CPR_CACHE_SLOTS];

static CprSlot* getSlot(uint32_t icao) {
  int freeIdx=-1;
  for (int i=0;i<CPR_CACHE_SLOTS;i++) {
    if (g_cpr_cache[i].icao==icao) return &g_cpr_cache[i];
    if (freeIdx<0 && g_cpr_cache[i].icao==0) freeIdx=i;
  }
  if (freeIdx<0) freeIdx=0;
  memset(&g_cpr_cache[freeIdx],0,sizeof(CprSlot));
  g_cpr_cache[freeIdx].icao=icao; return &g_cpr_cache[freeIdx];
}

static uint32_t getBits(const uint8_t *msg, int startBit, int bitLen) {
  uint32_t acc=0;
  for (int i=0;i<bitLen;i++) {
    int b=startBit+i; int by=b>>3; int bi=7-(b&7);
    acc=(acc<<1)|((msg[by]>>bi)&1);
  }
  return acc;
}

static const char *charset = "ABCDEFGHIJKLMNOPQRSTUVWXYZ_____"
                             "______________________________"
                             " 0123456789______";
static void decodeCS(const uint8_t *msg, char out[16]) {
  memset(out,0,16);
  int base=32+8; // после TC(5)+EC(3)
  for (int i=0;i<8;i++) {
    uint32_t v = getBits(msg, base+i*6, 6);
    char c = (v<64)?charset[v]:' ';
    out[i] = (c=='_'?' ':c);
  }
  out[8]=0;
  for (int i=7;i>=0;i--){ if(out[i]==' ') out[i]=0; else break; }
}

static inline double cprDlat(bool odd){ return odd?(360.0/59.0):(360.0/60.0); }
static int modp(int a,int b){int m=a%b; if(m<0)m+=b; return m;}

static bool cprGlobal(uint32_t latE,uint32_t lonE,uint32_t latO,uint32_t lonO,double &olat,double &olon){
  int j=floor((59*latE-60*latO)/131072.0+0.5);
  double rlatE=cprDlat(false)*(modp(j,60)+latE/131072.0);
  double rlatO=cprDlat(true) *(modp(j,59)+latO/131072.0);
  double lat=(fabs(rlatE)<=90 && cprNL(rlatE)>0)?rlatE:rlatO;
  int nl=cprNL(lat); if(nl<=0) return false;
  int ni=nl-((j%2)==0?0:1);
  double dlon=360.0/ni;
  int m=floor((lonE*(nl-1)-lonO*nl)/131072.0+0.5);
  double lon=dlon*(modp(m,ni)+(((j%2)==0?lonE:lonO)/131072.0));
  if (lon>180) lon-=360;
  olat=lat; olon=lon; return true;
}

static bool cprLocal(uint32_t latc,uint32_t lonc,bool odd,double rlat,double rlon,double &olat,double &olon){
  double dLat=cprDlat(odd);
  int j=floor(rlat/dLat)+floor(0.5+((rlat-floor(rlat/dLat)*dLat)/dLat - latc/131072.0));
  double lat=dLat*(j+latc/131072.0);
  if (lat>=270) lat-=360;
  int nl=cprNL(lat);
  if (nl==0){ olat=lat; olon=0; return true; }
  int ni=nl-(odd?1:0);
  double dLon=360.0/ni;
  int m=floor(rlon/dLon)+floor(0.5+((rlon-floor(rlon/dLon)*dLon)/dLon - lonc/131072.0));
  double lon=dLon*(m+lonc/131072.0);
  if (lon>180) lon-=360;
  olat=lat; olon=lon; return true;
}

static int decodeModeAfromID13(uint32_t id12){
  int A=((id12>>10)&1)|(((id12>>8)&1)<<1)|(((id12>>6)&1)<<2);
  int B=((id12>>5)&1) |(((id12>>3)&1)<<1)|(((id12>>1)&1)<<2);
  int C=((id12>>11)&1)|(((id12>>9)&1)<<1)|(((id12>>7)&1)<<2);
  int D=((id12>>4)&1) |(((id12>>2)&1)<<1)|(((id12>>0)&1)<<2);
  return ( (D&7) | ((C&7)<<3) | ((B&7)<<6) | ((A&7)<<9) );
}
static int unpackSquawk(int packed){
  int d1=(packed)&7, d2=(packed>>3)&7, d3=(packed>>6)&7, d4=(packed>>9)&7;
  return d4*1000+d3*100+d2*10+d1;
}

static void decodeVelocityTC19(const uint8_t *msg, DecodedADSB &o){
  uint32_t tc = (uint32_t) ((msg[4]>>3)&0x1F);
  if (tc!=19) return;
  uint32_t subtype = ((msg[4]&7));
  if (subtype==1 || subtype==2) {
    int s_ew=(msg[5]>>6)&1; int v_ew=((msg[5]&0x3F)<<4)|((msg[6]>>4)&0x0F); v_ew-=1;
    int s_ns=(msg[6]>>3)&1; int v_ns=((msg[6]&0x07)<<7)|((msg[7]>>1)&0x7F); v_ns-=1;
    if (v_ew>=0 && v_ns>=0) {
      double ve=s_ew?-v_ew:v_ew, vn=s_ns?-v_ns:v_ns;
      o.speed = sqrt(ve*ve+vn*vn);
      double trk = atan2(ve,vn)*180.0/M_PI; if(trk<0) trk+=360; o.course=trk;
    }
    int vr_s=(msg[8]>>7)&1; (void)vr_s;
    int vr_sign=(msg[8]>>6)&1; int vr = (((msg[8]&0x3F)<<3)|((msg[9]>>5)&0x07)) - 1;
    if (vr>=0) o.vert_rate = (vr_sign?-vr:vr)*64;
  } else if (subtype==3 || subtype==4) {
    int hdg_stat=(msg[5]>>6)&1; int hdg_raw=((msg[5]&0x3F)<<4)|((msg[6]>>4)&0x0F);
    if (hdg_stat) o.course = (hdg_raw*360.0)/1024.0;
    int as_tas=(msg[6]>>3)&1; (void)as_tas;
    int as_val=(((msg[6]&0x07)<<7)|((msg[7]>>1)&0x7F))-1;
    if (as_val>=0) o.speed=as_val;
    int vr_s=(msg[8]>>7)&1; (void)vr_s;
    int vr_sign=(msg[8]>>6)&1; int vr=(((msg[8]&0x3F)<<3)|((msg[9]>>5)&0x07))-1;
    if (vr>=0) o.vert_rate=(vr_sign?-vr:vr)*64;
  }
}

static bool decodePosition(const uint8_t *msg, uint32_t now_ms, DecodedADSB &o){
  uint32_t df=(msg[0]>>3)&0x1F;
  if (!(df==17 || df==18)) return false;
  uint32_t icao=(msg[1]<<16)|(msg[2]<<8)|msg[3];
  o.addr=icao;
  uint32_t tc=(msg[4]>>3)&0x1F;
  if (tc<9 || tc>18) return false;

  uint16_t ac12 = ((uint16_t)msg[5]<<4)|((msg[6]>>4)&0x0F);
  bool qbit = (ac12 & 0x10)!=0;
  int32_t alt_ft = modes_altitude_from_ac12(ac12, qbit);
  if (alt_ft != GILLHAM_INVALID_ALT) { o.altitude = (float)alt_ft; o.pressure_altitude=o.altitude; }

  uint32_t odd = (msg[6]>>3)&1;
  uint32_t latc = ((uint32_t)(msg[6]&0x07)<<14)|((uint32_t)msg[7]<<6)|((msg[8]>>2)&0x3F);
  uint32_t lonc = ((uint32_t)(msg[8]&0x03)<<15)|((uint32_t)msg[9]<<7)|((msg[10]>>1)&0x7F);

  CprSlot *s=getSlot(icao);
  if (odd){ s->lat_odd=latc; s->lon_odd=lonc; s->t_odd_ms=now_ms; s->has_odd=true; }
  else    { s->lat_even=latc; s->lon_even=lonc; s->t_even_ms=now_ms; s->has_even=true; }

  double lat=0,lon=0; bool ok=false;
  if (s->has_even && s->has_odd) {
    uint32_t dt = (s->t_even_ms>s->t_odd_ms)?(s->t_even_ms-s->t_odd_ms):(s->t_odd_ms-s->t_even_ms);
    if (dt<=g_pair_window_ms) ok = cprGlobal(s->lat_even,s->lon_even,s->lat_odd,s->lon_odd,lat,lon);
  }
  if (!ok) ok = cprLocal(latc,lonc,odd,g_ref_lat,g_ref_lon,lat,lon);
  if (ok){ o.latitude=(float)lat; o.longitude=(float)lon; s->last_lat=lat; s->last_lon=lon; }
  return ok;
}

static bool decodeCallSign(const uint8_t *msg, DecodedADSB &o){
  uint32_t df=(msg[0]>>3)&0x1F; if(!(df==17||df==18)) return false;
  uint32_t tc=(msg[4]>>3)&0x1F; if(tc<1||tc>4) return false;
  char cs[16]; decodeCS(msg, cs); strncpy(o.flight, cs, sizeof(o.flight)-1); return true;
}

static bool decodeSquawkDF5_DF21(const uint8_t *msg, DecodedADSB &o){
  uint32_t df=(msg[0]>>3)&0x1F; if(!(df==5||df==21)) return false;
  uint32_t id12;
  if (df==21) id12 = (uint32_t)((msg[4]<<4)|((msg[5]>>4)&0x0F));
  else        id12 = (uint32_t)((msg[4]<<4)|((msg[5]>>4)&0x0F)); // совместимость
  int packed=decodeModeAfromID13(id12);
  o.Squawk=unpackSquawk(packed); return true;
}

bool adsb_decode(const AdsbPacket &pkt, DecodedADSB &out){
  memset(&out,0,sizeof(out));
  out.Squawk=-1; out.vert_rate=0; out.signal_source=1;
  out.timestamp = pkt.timestamp; out.rssi = pkt.rssi;
  const uint8_t *msg=pkt.raw;
  uint32_t df=(msg[0]>>3)&0x1F;
  if (df==17||df==18||df==21){ uint32_t icao=(msg[1]<<16)|(msg[2]<<8)|msg[3]; out.addr=icao; }

  bool any=false;
  if (decodeCallSign(msg,out)) any=true;
  if (decodePosition(msg, pkt.timestamp, out)) any=true;
  { uint32_t tc=(msg[4]>>3)&0x1F; if (tc==19){ decodeVelocityTC19(msg,out); any=true; } }
  if (decodeSquawkDF5_DF21(msg,out)) any=true;

  return any;
}

#include "include/adsb_crc.h"

// Полином Mode S: x^24 + x^23 + x^18 + x^17 + x^14 + x^11 + x^10 + x^7 + x^6 + x^5 + x^4 + x^3 + x + 1
// В hex (без старшего бита): 0xFFF409 (в разных источниках встречается 0x1FFF409 в 25 битах).
// Реализуем побитовый делитель над входным сообщением.

uint32_t adsb_crc24(const uint8_t *msg, int useful_bits) {
  // useful_bits: количество бит данных БЕЗ последних 24 бит паритета
  const uint32_t poly = 0xFFF409; // 24-bit poly
  uint32_t reg = 0;
  for (int i=0;i<useful_bits;i++) {
    int byte = i>>3;
    int bit  = 7-(i&7);
    uint8_t in = (msg[byte]>>bit)&1;
    uint8_t top = (reg>>23)&1;
    reg = ((reg<<1)&0xFFFFFF) | in;
    if (top) reg ^= poly;
  }
  return reg & 0xFFFFFF;
}

bool adsb_crc_check(const uint8_t *msg, int bits) {
  if (!(bits==56 || bits==112)) return false;
  int useful = bits - 24;
  uint32_t rem = adsb_crc24(msg, useful);
  // сравним с полученным паритетом
  uint32_t parity = 0;
  for (int i=0;i<24;i++) {
    int b = useful + i;
    int byte=b>>3, bit=7-(b&7);
    parity = (parity<<1) | ((msg[byte]>>bit)&1);
  }
  return rem == parity;
}

#include "include/adsb_gillham.h"
#include <LittleFS.h>

static int16_t g_table[2048];
static bool g_inited=false;

static inline uint8_t gray8_to_bin(uint8_t g){
  g^=g>>1; g^=g>>2; g^=g>>4; return g;
}
static inline uint8_t bin_to_gray8(uint8_t b){ return b ^ (b>>1); }

int16_t gillham_decode_11(uint16_t g11){
  uint8_t D2=(g11>>10)&1, D4=(g11>>9)&1, A1=(g11>>8)&1, A2=(g11>>7)&1, A4=(g11>>6)&1;
  uint8_t B1=(g11>>5)&1,  B2=(g11>>4)&1,  B4=(g11>>3)&1,  C1=(g11>>2)&1,  C2=(g11>>1)&1, C4=g11&1;

  uint8_t gray500=(D2<<7)|(D4<<6)|(A1<<5)|(A2<<4)|(A4<<3)|(B1<<2)|(B2<<1)|B4;
  uint8_t n500 = gray8_to_bin(gray500);
  uint8_t c=(C1<<2)|(C2<<1)|C4;

  bool evenParity = (__builtin_parity((unsigned)n500)==0);
  int16_t off=0;
  switch(c){
    case 0b010: off=0; break;
    case 0b001: off = evenParity?-200:+200; break;
    case 0b011: off = evenParity?-100:+100; break;
    case 0b110: off = evenParity?+100:-100; break;
    case 0b100: off = evenParity?+200:-200; break;
    default: return GILLHAM_INVALID_ALT;
  }
  int32_t alt = -1000 + (int32_t)n500*500 + off;
  if (alt<-1200 || alt>126700) return GILLHAM_INVALID_ALT;
  return (int16_t)alt;
}

void adsb_gillham_init(){
  if (g_inited) return;
  // Попытка загрузить из LittleFS
  if (LittleFS.begin()) {
    if (LittleFS.exists("/gillham_table.bin")) {
      File f=LittleFS.open("/gillham_table.bin","r");
      if (f && f.read((uint8_t*)g_table, sizeof(g_table))==sizeof(g_table)) {
        f.close(); g_inited=true; return;
      }
      if (f) f.close();
    }
  }
  for (uint16_t c=0;c<2048;c++) g_table[c]=gillham_decode_11(c);
  g_inited=true;
  // Сохраним для быстрых стартов
  if (LittleFS.begin()) {
    File f=LittleFS.open("/gillham_table.bin","w");
    if (f) { f.write((uint8_t*)g_table,sizeof(g_table)); f.close(); }
  }
}

uint16_t modes_ac12_to_g11(uint16_t ac12){
  ac12 &= 0x0FFF;
  uint16_t g11=0;
  g11 |= ((ac12>>11)&1)<<10; // D2
  g11 |= ((ac12>>10)&1)<< 9; // D4
  g11 |= ((ac12>> 9)&1)<< 8; // A1
  g11 |= ((ac12>> 8)&1)<< 7; // A2
  g11 |= ((ac12>> 7)&1)<< 6; // A4
  g11 |= ((ac12>> 6)&1)<< 5; // B1
  g11 |= ((ac12>> 5)&1)<< 4; // B2
  g11 |= ((ac12>> 4)&1)<< 3; // B4
  g11 |= ((ac12>> 2)&1)<< 2; // C1
  g11 |= ((ac12>> 1)&1)<< 1; // C2
  g11 |= ((ac12>> 0)&1)<< 0; // C4
  return g11 & 0x07FF;
}

int32_t modes_altitude_from_ac12(uint16_t ac12, bool q_bit){
  if (q_bit){
    // Соберём N (11 бит) пропуская M/Q
    uint16_t N=0;
    N |= ((ac12>>11)&1)<<10; N |= ((ac12>>10)&1)<<9;  N |= ((ac12>>9)&1)<<8;
    N |= ((ac12>> 8)&1)<<7;  N |= ((ac12>> 7)&1)<<6;  N |= ((ac12>>6)&1)<<5;
    N |= ((ac12>> 5)&1)<<4;  N |= ((ac12>> 4)&1)<<3;  N |= ((ac12>>2)&1)<<2;
    N |= ((ac12>> 1)&1)<<1;  N |= ((ac12>>0)&1)<<0;
    return (int32_t)N*25 - 1000;
  } else {
    if (!g_inited) adsb_gillham_init();
    uint16_t g11=modes_ac12_to_g11(ac12);
    return g_table[g11];
  }
}

uint16_t gillham_encode_11(int32_t altitude_ft){
  if (altitude_ft<-1000 || altitude_ft>126700) return 0xFFFF;
  int32_t base = (altitude_ft+1000)/500; if(base<0)base=0; if(base>255)base=255;
  int32_t rem = altitude_ft - (-1000 + base*500);
  int16_t off=0;
  switch(rem){case -200:off=-200;break;case -100:off=-100;break;case 0:off=0;break;case 100:off=100;break;case 200:off=200;break;default:return 0xFFFF;}
  uint8_t gray500 = bin_to_gray8((uint8_t)base);
  bool evenParity = (__builtin_parity((unsigned)base)==0);
  uint8_t c=0b010;
  if (off==-200) c=evenParity?0b001:0b100;
  if (off==-100) c=evenParity?0b011:0b110;
  if (off==+100) c=evenParity?0b110:0b011;
  if (off==+200) c=evenParity?0b100:0b001;
  uint16_t g11=0;
  g11 |= ((gray500>>7)&1)<<10; g11 |= ((gray500>>6)&1)<<9; g11 |= ((gray500>>5)&1)<<8; g11 |= ((gray500>>4)&1)<<7;
  g11 |= ((gray500>>3)&1)<<6;  g11 |= ((gray500>>2)&1)<<5; g11 |= ((gray500>>1)&1)<<4; g11 |= ((gray500>>0)&1)<<3;
  g11 |= ((c>>2)&1)<<2; g11 |= ((c>>1)&1)<<1; g11 |= (c&1);
  return g11;
}

//(весовая корреляция по 80 отсчётам, подавление ложных, dead-time, сборка бит из 10 отсчётов/бит; профилируемый порог EMA)

#include <Arduino.h>
#include <string.h>
#include "include/adsb_correlator.h"
#include "include/adsb_dma.h"

static CorrProfile g_prof;

void set_corr_profile(const CorrProfile &p){ g_prof=p; }
void adsb_correlator_init(){}

// Шаблон преамбулы в шкале 0.1 мкс (10 MHz), 8 мкс = 80 отсчётов.
// Позитивные окна (где ожидается энергия) по позициям (начало..конец):
// Импульсы 0..0.5, 1.0..1.5, 3.5..4.0, 5.5..6.0 мкс.
// С запасом ±1 отсчёт.
struct Win { uint8_t s,e; };
static const Win POSW[] = {
  { 0,  5}, {10, 15}, {35, 40}, {55, 60}
};
static const Win NEG_MASK[] = { // маска подавления — не должно быть длинных импульсов
  { 6,  9}, {16, 34}, {41, 54}, {61, 79}
};

static bool corr_preamble_and_get_t0(const uint32_t* words, uint32_t nwords, uint32_t &bit_index, uint8_t &dyn_thr) {
  // words — поток битов (LSB-first? мы будем разбирать по битам MSB->LSB равномерно)
  // Упрощение: заранее разворачивать не будем — оценим по битовой маске, считая бит=1 как "импульс".
  // Пройдём по всем битам, скользящее окно 80 бит.
  uint32_t total_bits = nwords*32;
  static uint8_t ema = 0; // оценка шума
  dyn_thr = g_prof.base_thr + (g_prof.k_ema * ema)/8;

  if (total_bits < 80) return false;
  uint32_t score=0, penalty=0;
  for (uint32_t i=0;i<=total_bits-80;i++) {
    // веса: суммируем количество '1' в положительных окнах
    int posHits=0, negLong=0;
    for (auto &w : POSW){
      for (int b=w.s;b<=w.e;b++){
        uint32_t idx=i+b;
        uint32_t widx=idx>>5, boff=31-(idx&31);
        posHits += ( (words[widx]>>boff) & 1 );
      }
    }
    // проверим отсутствие «длинных» импульсов в отрицательных окнах
    for (auto &w : NEG_MASK){
      int run=0;
      for (int b=w.s;b<=w.e;b++){
        uint32_t idx=i+b;
        uint32_t widx=idx>>5, boff=31-(idx&31);
        int v = ( (words[widx]>>boff) & 1 );
        if (v){ run++; if (run>7) { negLong=1; break; } } else run=0; // >0.7 мкс в окне
      }
      if (negLong) break;
    }
    // динамический порог
    if (posHits >= dyn_thr && !negLong) {
      // обновим EMA минимально (шум низкий)
      ema = (uint8_t)((7*ema + 1)/8);
      bit_index = i; return true;
    } else {
      // обновим EMA «шумом»: negHits ~ 80-posHits
      uint8_t negHits = (uint8_t)(80 - (posHits<80?posHits:80));
      ema = (uint8_t)((7*ema + negHits)/8);
    }
  }
  return false;
}

// Сборка битов PWM PPM по 10 отсчётов на бит.
// 0 → энергия в первой половине (0..4), 1 → во второй (5..9)
static bool assemble_bits(const uint32_t* words, uint32_t start_bit, uint16_t &out_bits_count, uint8_t out_bytes[14]) {
  // Сначала определим ожидаемую длину: прочитаем первые 5 бит DF
  // Соберём сразу 112 бит максимум, потом укоротим по DF.
  memset(out_bytes,0,14);
  auto bit_at = [&](uint32_t idx)->uint8_t{
    uint32_t widx=idx>>5, boff=31-(idx&31);
    return (words[widx]>>boff)&1;
  };

  auto energy = [&](uint32_t s,uint32_t e)->int{
    int ssum=0;
    for (uint32_t k=s;k<=e;k++) ssum += bit_at(k);
    return ssum;
  };

  // После преамбулы (80 бит) начинается 1us-битовый поток = 10 сэмплов/бит
  // Вытянем сначала 14*8=112 бит => 1120 сэмплов
  uint32_t data_start = start_bit + 80;
  // sample budget check
  // (внешний код должен передавать достаточно слов)
  // 112 бит = 1120 сэмплов ⇒ 1120 бит индекса

  uint8_t bits[112]={0};
  for (int i=0;i<112;i++){
    uint32_t s0 = data_start + i*10;
    int e1 = energy(s0, s0+4);
    int e2 = energy(s0+5, s0+9);
    bits[i] = (e2>e1) ? 1 : 0;
  }

  // DF — первые 5 бит:
  int df=0; for(int i=0;i<5;i++){ df=(df<<1)|bits[i]; }
  uint16_t total = (df==17||df==18||df==21)?112:56;
  out_bits_count = total;

  // Упакуем в байты MSB->LSB
  for (int i=0;i<total;i++){
    int byte=i>>3, bi=7-(i&7);
    if (bits[i]) out_bytes[byte]|=(1<<bi);
  }
  return true;
}

bool adsb_try_extract_packet(int ch, AdsbPacket &pkt){
  uint32_t nwords=0;
  const uint32_t* buf = dma_get_buffer(ch, nwords);
  if (!buf || nwords==0) return false;

  uint32_t t0=0; uint8_t dyn_thr=0;
  bool ok = corr_preamble_and_get_t0(buf, nwords, t0, dyn_thr);
  if (!ok) { return false; }

  // Соберём биты из 1120 сэмплов после t0+80
  uint16_t bitsCount=0;
  uint8_t frame[14];
  if (!assemble_bits(buf, t0, bitsCount, frame)) {
    return false;
  }

  // dead time: пропустить 6–8 мкс (80 бит), плюс длина пакета (56/112 бит)*10
  uint32_t skip_bits = 80 + (uint32_t)bitsCount*10 + (uint32_t)g_prof.dead_time_us*10;
  uint32_t skip_words = (skip_bits+31)/32;
  dma_consume(ch, skip_words);

  // сформировать pkt
  pkt.bits = bitsCount;
  memcpy(pkt.raw, frame, (bitsCount+7)/8);
  pkt.timestamp = millis();
  pkt.rssi = 0; // можно привязать к ADC26 при желании
  return true;
}

rp2040_adsb/src/adsb_dma.cpp

//(PIO-программа на 10 МГц, 3 state-machine, 3 DMA-кольца; без pioasm — встроенный pio header)

#include <Arduino.h>
#include <hardware/pio.h>
#include <hardware/dma.h>
#include <hardware/regs/pio.h>
#include <hardware/clocks.h>
#include <string.h>
#include "include/adsb_dma.h"

// --- Конфиг ---
#define RING_WORDS   8192     // 32 KB на канал (8192*4)
#define DMA_PER_CH   1        // 1 DMA канал на SM
#define PIO_CLOCK_HZ 125000000 // обычно 125 МГц

// --- Глобалы ---
static PIO pio = pio0;
static int sm_idx[3] = {-1,-1,-1};
static int dma_chan[3] = {-1,-1,-1};
static volatile uint32_t *ringbuf[3] = {nullptr,nullptr,nullptr};
static volatile uint32_t wptr[3] = {0,0,0};
static volatile uint32_t rptr[3] = {0,0,0};

// PIO программа: сэмплируем 1 входной пин каждую тактовую «итерацию» и пихаем в ISR,
// автосдвиг каждые 32 бита → PUSH в FIFO.
static const uint16_t pio_prog_instr[] = {
  // .wrap_target
  pio_encode_in(pio_pins, 1),   // IN PINS,1
  pio_encode_jmp(0),            // JMP 0 (tight loop)
  // .wrap
};
static const struct pio_program pio_prog = {
  .instructions=(uint16_t*)pio_prog_instr,
  .length=sizeof(pio_prog_instr)/sizeof(pio_prog_instr[0]),
  .origin=-1
};

static void pio_setup_sm(int sm, int pin, float div){
  // Конфиг SM
  pio_sm_config c = pio_get_default_sm_config();
  sm_config_set_in_pins(&c, pin);
  sm_config_set_in_shift(&c, true, true, 32); // shift right, autopush, threshold 32
  sm_config_set_clkdiv(&c, div);
  // Пин как вход
  pio_gpio_init(pio, pin);
  gpio_pull_down(pin);
  // SM init
  int offset = pio_add_program(pio, &pio_prog);
  pio_sm_init(pio, sm, offset+0, &c);
  pio_sm_set_consecutive_pindirs(pio, sm, pin, 1, false);
  pio_sm_exec(pio, sm, pio_encode_set(pio_y, 0));
  pio_sm_set_enabled(pio, sm, true);
}

static void dma_setup(int ch, int sm) {
  dma_chan[ch] = dma_claim_unused_channel(true);
  dma_channel_config c = dma_channel_get_default_config(dma_chan[ch]);
  channel_config_set_read_increment(&c, false);
  channel_config_set_write_increment(&c, true);
  channel_config_set_dreq(&c, pio_get_dreq(pio, sm, false));
  channel_config_set_ring(&c, true, 13); // 2^13 = 8192 байт? (мы пишем 32-бит слова → 8192*4) — сделаем вручную
  // Мы управляем кольцом сами → без встроенного ring feature; используем прерывистые срабатывания.
  // Поэтому ring в DMA можно не включать; реализуем вручную.
  channel_config_set_ring(&c, false, 0);

  dma_channel_configure(dma_chan[ch], &c,
    (void*)&ringbuf[ch][0],       // dst
    (const void*)&pio->rxf[sm],   // src
    RING_WORDS,                   // считывать RING_WORDS слов за раз (потом переключим dst и перезапустим)
    false);

  // Запуск
  dma_start_channel_mask(1u<<dma_chan[ch]);
}

// Перезапуск DMA блока
static void dma_refill(int ch) {
  // записывать с текущего wptr дальше по кругу, блок = сколько влезет до конца
  uint32_t space = (RING_WORDS - (wptr[ch] % RING_WORDS));
  dma_channel_set_write_addr(dma_chan[ch], (void*)&ringbuf[ch][wptr[ch]%RING_WORDS], true);
  dma_channel_set_trans_count(dma_chan[ch], space, true);
  wptr[ch] = (wptr[ch] + space) % (RING_WORDS*2); // "виртуальный" счётчик
}

void adsb_dma_init(int pin1, int pin2, int pin3, uint32_t sample_rate_hz){
  // Память под кольца
  for (int i=0;i<3;i++){
    ringbuf[i] = (volatile uint32_t*)malloc(RING_WORDS*sizeof(uint32_t));
    memset((void*)ringbuf[i], 0, RING_WORDS*sizeof(uint32_t));
    wptr[i]=rptr[i]=0;
  }
  // Частота: div = PIO_CLOCK_HZ / sample_rate
  float div = (float)clock_get_hz(clk_sys) / (float)sample_rate_hz;

  // SM allocate
  sm_idx[0] = pio_claim_unused_sm(pio, true);
  sm_idx[1] = pio_claim_unused_sm(pio, true);
  sm_idx[2] = pio_claim_unused_sm(pio, true);

  // PIO программы (одна общая)
  pio_setup_sm(sm_idx[0], pin1, div);
  pio_setup_sm(sm_idx[1], pin2, div);
  pio_setup_sm(sm_idx[2], pin3, div);

  // DMA на каждый SM
  dma_setup(0, sm_idx[0]);
  dma_setup(1, sm_idx[1]);
  dma_setup(2, sm_idx[2]);

  // Инициальный refill
  dma_refill(0); dma_refill(1); dma_refill(2);
}

const uint32_t* dma_get_buffer(int ch, uint32_t &out_word_count){
  // Сколько DMA уже перелило? Возьмём текущий write address
  // Упростим: будем «пачками» подхватывать половину буфера, опираясь на dma_hw регистры.
  // В Arduino среде прямой доступ может быть ограничен; воспользуемся внутренним счётчиком.
  // Предполагаем, что refill вызывается периодически после consume.
  uint32_t wp = wptr[ch];
  // Что доступно к чтению:
  // Здесь без аппаратной синхронизации: будем подавать большие блоки и сразу consume после обработки.
  out_word_count = RING_WORDS; // даём весь блок
  return (const uint32_t*)ringbuf[ch];
}

void dma_consume(int ch, uint32_t words_used){
  // Просто перезапускаем DMA порцию (рефилл буфера). В этой простой модели — full refill.
  dma_refill(ch);
  // rptr можно обновить, если нужно внутреннее отслеживание
  rptr[ch] = (rptr[ch] + words_used) % (RING_WORDS*2);
}
/*

Примечание: реализация DMA-колец здесь упрощённая (без аппаратного «ring» DMA), 
но практичная для Arduino: мы подаём большими блоками и вручную рефилим; коррелятор 
обрабатывает массив целиком и сообщает consume() примерный объём для сдвига. 
При необходимости можно усложнить — хранить реальный write-addr DMA 
через dma_hw->ch[dma_chan].write_addr и вычислять «сколько влилось».
*/

static void hex2bytes(const char *hex, uint8_t *out, int len) {
  auto nyb = [](char c)->uint8_t{
    if (c>='0'&&c<='9') return c-'0';
    if (c>='a'&&c<='f') return c-'a'+10;
    if (c>='A'&&c<='F') return c-'A'+10;
    return 0;
  };
  for (int i=0;i<len;i++) out[i]=(nyb(hex[2*i])<<4)|nyb(hex[2*i+1]);
}

static void printLine(const char *label, const DecodedADSB &f){
  Serial.printf("| %-18s | %06X | %-8s | %9.5f | %10.5f | %6.0f | %6.1f | %6.1f | %5d | %04d |\n",
    label, f.addr, f.flight, f.latitude, f.longitude, f.altitude, f.speed, f.course, f.vert_rate, (f.Squawk>=0?f.Squawk:9999));
}

// Приём по Serial1 (UART), вывод в Serial (USB)
#define RX_PIN 42

void setup() {
  Serial.begin(115200);
  Serial1.begin(115200, SERIAL_8N1, RX_PIN, TX_PIN);
  Serial.println("ESP32-S3 Receiver Ready");

void loop() {
  static DecodedADSB fo;
  if (Serial1.available() >= (int)sizeof(fo)) {
    size_t n = Serial1.readBytes((char*)&fo, sizeof(fo));
    if (n == sizeof(fo)) {
      Serial.printf("RX: ICAO=%06X FLT=%s ALT=%.0fft LAT=%.5f LON=%.5f SPD=%.1f CRS=%.1f SQ=%d\n",
        fo.addr, fo.flight, fo.altitude, fo.latitude, fo.longitude, fo.speed, fo.course, fo.Squawk);
    }
  }
}

Замечания по сборке и запуску

Arduino Core RP2040 (Earle Philhower): включите LittleFS (он встроен). 
Первый запуск создаст /gillham_table.bin автоматически.

Скорости: PIO семплирует 10 МГц (0.1 мкс). 
Коррелятор использует 80-отсчётное окно.

Профили шума: при сильных помехах увеличьте base_thr в профиле (например, 10–12) и/или k_ema.

Dead-time: dead_time_us=8 избегает повторной детекции хвоста.

CRC: adsb_crc_check() ожидает полный кадр (с паритетом). 
Для отладки «сырых» можно сравнивать adsb_crc24(msg,bits-24) отдельно.

RSSI: сейчас просто считывается analogRead(26) — при желании свяжите с оценкой уровня для фильтра.

Стабильность DMA: представленный DMA-контур упрощён для Arduino. 
Если нагрузка очень высокая, можно перейти на «истинные» DMA-кольца с адресными масками и IRQ.
