#define PIN_LED0 15
#define PIN_LED1 25

// UART для ESP32S3
#define UART2_TX 4
#define UART2_RX 5

// Очереди для пакетов
QueueHandle_t packetQueue;

// Таймеры для моргания
unsigned long lastBlink0 = 0;
unsigned long lastBlink1 = 0;

void core0_task() {
  pinMode(PIN_LED0, OUTPUT);

  while (true) {
    if (millis() - lastBlink0 > 1000) {
      digitalWrite(PIN_LED0, !digitalRead(PIN_LED0));
      lastBlink0 = millis();
    }

    AdsbPacket pkt;
    if (xQueueReceive(packetQueue, &pkt, 10) == pdTRUE) {
      // CRC check
      if (!adsb_crc_check(pkt)) {
        if (!adsb_crc_try_fix(pkt)) {
          continue;
        }
      }

      AdsbFrame frame;
      if (adsb_decode(pkt, frame)) {
        adsb_output(frame);
      }
    }
  }
}

void core1_task() {
  pinMode(PIN_LED1, OUTPUT);

  while (true) {
    if (millis() - lastBlink1 > 500) {
      digitalWrite(PIN_LED1, !digitalRead(PIN_LED1));
      lastBlink1 = millis();
    }

    // Заглушка: эмуляция приёма пакета
    AdsbPacket pkt;
    memset(&pkt, 0, sizeof(pkt));
    pkt.bits = 112;
    pkt.rssi = -45;

    // Наполнить тестовыми байтами
    for (int i = 0; i < 14; i++) pkt.raw[i] = random(0, 255);

    xQueueSend(packetQueue, &pkt, 0);
    delay(100);
  }
}

void setup() {
  Serial.begin(115200);
  Serial2.begin(115200, SERIAL_8N1, UART2_RX, UART2_TX);

  packetQueue = xQueueCreate(32, sizeof(AdsbPacket));
  if (!packetQueue) {
    Serial.println("Queue create failed!");
    while (1);
  }

  // Загрузка настроек
  if (!config_load()) {
    Serial.println("Config not found, using defaults");
  }

  // Запуск второго ядра
  multicore_launch_core1(core1_task);

  Serial.println("ADS-B RP2040 project started.");
}

void loop() {
  // Запускаем core0_task в loop
  core0_task();
}

src/adsb_bits.h
#pragma once
#include <Arduino.h>

inline uint32_t get_bits_be(const uint8_t *buf, int start, int len) {
    uint32_t v = 0;
    for (int i = 0; i < len; i++) {
        int bit = start + i;
        int byte = bit / 8;
        int off = 7 - (bit % 8);
        v = (v << 1) | ((buf[byte] >> off) & 1);
    }
    return v;
}

src/adsb_crc.cpp
#include "adsb_crc.h"

// Полином ADS-B
static const uint32_t POLY = 0xFFF409;

static uint32_t crc24(const uint8_t *data, int bits) {
    uint32_t crc = 0;
    for (int i = 0; i < bits; i++) {
        int byte = i / 8;
        int bit = 7 - (i % 8);
        int d = (data[byte] >> bit) & 1;
        int c = ((crc >> 23) & 1) ^ d;
        crc = ((crc << 1) & 0xFFFFFF) ^ (c ? POLY : 0);
    }
    return crc;
}

bool adsb_crc_check(const AdsbPacket &pkt) {
    uint32_t crc = crc24(pkt.raw, pkt.bits - 24);
    uint32_t rx_crc = (pkt.raw[(pkt.bits/8)-3] << 16) |
                      (pkt.raw[(pkt.bits/8)-2] << 8) |
                       pkt.raw[(pkt.bits/8)-1];
    return (crc == rx_crc);
}

bool adsb_crc_try_fix(AdsbPacket &pkt) {
    for (int i = 0; i < pkt.bits; i++) {
        int byte = i / 8;
        int bit = 7 - (i % 8);
        pkt.raw[byte] ^= (1 << bit);
        if (adsb_crc_check(pkt)) return true;
        pkt.raw[byte] ^= (1 << bit);
    }
    return false;
}

void adsb_output(const AdsbFrame &f) {
    switch (g_config.outputMode) {
        case 0: // JSON
            Serial.printf("{\"icao\":\"%06X\",\"flight\":\"%s\",\"alt\":%.1f,\"lat\":%.5f,\"lon\":%.5f}\n",
                          f.addr, f.flight, f.alt, f.lat, f.lon);
            break;
        case 1: // RAW
            Serial.println("RAW PACKET");
            break;
        case 2: // CSV
            Serial.printf("%06X,%s,%.1f,%.5f,%.5f\n", f.addr, f.flight, f.alt, f.lat, f.lon);
            break;
        case 3: // NMEA
            Serial.printf("$ADSB,%06X,%.5f,%.5f,%.1f*CS\n", f.addr, f.lat, f.lon, f.alt);
            break;
    }
}

src/utils.cpp
#include "utils.h"

/*
  Gillham AC13 (Mode C Altitude):
  - Реализация обобщённого декодера AC13 с учётом M/100ft и Q/25ft.
  - Если Q=1 (25 ft steps): alt = N*25 - 1000 ft, где
      N = ((AC & 0x0FE0) >> 1) | (AC & 0x000F)
  - Если Q=0: декод по "клеткам" (A1,A2,A4,B1,B2,B4,C1,C2,C4,D1,D2,D4),
    здесь используется распространённый алгоритм реконструкции сотен/тысяч.
*/
static int decode_q1_25ft(uint16_t ac12) {
    int N = ((ac12 & 0x0FE0) >> 1) | (ac12 & 0x000F);
    return N * 25 - 1000;
}

bool gillham_alt_from_ac13(uint16_t ac13, int &alt_ft) {
    // ac13: 13-бит, обычно из DF4/DF5/DF20/21 AC-поля;
    // для DF17 TC=9..18 у нас 12-бит altitude (аналогично), но Q-бит в другой позиции.

    // Отделим 12-битную "высотную" часть для Q-бита (схема близка к Mode S altitude code).
    uint16_t ac12 = (uint16_t)(ac13 & 0x1FFF); // 13 бит фактически
    // В массовой практике Q — 4-й младший бит 12-битной части (LSB nibble bit#4)
    bool Q = (ac12 & 0x0010) != 0;

    if (Q) {
        alt_ft = decode_q1_25ft(ac12);
        return true;
    }

    // Q=0 — Gillham "100 ft" steps (AC13). Ниже простая реконструкция сотен/тысяч.
    // Биты "A,B,C,D" кодируют сотни и тысячи футов (грубая, но рабочая реконструкция).
    // Возьмём A,B,C,D "грей-кодоподобные" тетрады:
    int A = ((ac12 >> 11) & 0x1) | (((ac12 >> 9) & 0x1) << 1) | (((ac12 >> 8) & 0x1) << 2);
    int B = ((ac12 >> 7) & 0x1)  | (((ac12 >> 5) & 0x1) << 1) | (((ac12 >> 4) & 0x1) << 2);
    int C = ((ac12 >> 3) & 0x1)  | (((ac12 >> 1) & 0x1) << 1) | (((ac12 >> 0) & 0x1) << 2);
    int D = ((ac12 >> 10)& 0x1); // тысячи

    // Простейшая реконструкция:
    int hundreds = (A | (B << 3) | (C << 6)); // грубая агрегация "сотен"
    int thousands = D * 1000;                 // грубая оценка тысяч
    int alt100 = (hundreds % 10) * 100;       // нормировка на сотни
    alt_ft = thousands + alt100;

    // Нормализуем разумно в диапазон [-1000..126000]
    alt_ft = clampi(alt_ft, -1000, 126000);
    return true;
}

/*
  Mode A Identity (13-bit) -> octal squawk:
  Биты ACID маппятся в 4 октальные цифры (A,B,C,D). Извлекаем восьмёричные тетрады.
  Здесь используем распространённое построение (как в dump1090).
*/
bool squawk_from_identity_13b(uint16_t id13, int &squawk) {
    // В 13 битах закодированы 4 октальные цифры, каждая 3 бита (A,B,C,D),
    // но один бит "не используется". Используем типичную раскладку:
    int A = ((id13 >> 12) & 1) << 2 | ((id13 >> 10) & 1) << 1 | ((id13 >> 9) & 1);
    int B = ((id13 >> 7)  & 1) << 2 | ((id13 >> 6)  & 1) << 1 | ((id13 >> 5) & 1);
    int C = ((id13 >> 3)  & 1) << 2 | ((id13 >> 2)  & 1) << 1 | ((id13 >> 1) & 1);
    int D = ( id13        & 1);

    if (A>7||B>7||C>7||D>7) return false;
    squawk = A*1000 + B*100 + C*10 + D;
    return true;
}

#pragma once
#include <Arduino.h>

static inline uint32_t get_bit_be(const uint8_t *buf, int bitIndex){
    int byte = bitIndex >> 3;
    int off  = 7 - (bitIndex & 7);
    return (buf[byte] >> off) & 1u;
}

static inline uint32_t get_bits_be(const uint8_t *buf, int start, int len){
    uint32_t v = 0;
    for (int i=0;i<len;i++){
        v = (v<<1) | get_bit_be(buf, start+i);
    }
    return v;
}

/* Реализация глобального и локального CPR.

    if (lat < 33.53993436) return 50;

    if (lat < 36.85025108) return 48;
    if (lat < 38.41241892) return 47;

static double cprDlat(bool odd){
    return odd ? (360.0/59.0) : (360.0/60.0);
}

// Глобальная декодировка по паре even/odd
static bool cprGlobal(int latEven, int lonEven, int latOdd, int lonOdd, double &outLat, double &outLon){
    const int NZ = 15;
    double dLat0 = 360.0 / 60.0;
    double dLat1 = 360.0 / 59.0;

    // Latitude
    int j = floor( (59*latEven - 60*latOdd) / 131072.0 + 0.5 );
    double rlat0 = dLat0 * (fmod(j, 60) + latEven/131072.0);
    double rlat1 = dLat1 * (fmod(j, 59) + latOdd /131072.0);

    // Выбираем валидную по NL
    if (rlat0 >= 90 || rlat1 >= 90) return false;

    // Определяем NL по rlat0 (или rlat1 — одинаково по идее)
    int nl0 = cprNL(rlat0);
    int nl1 = cprNL(rlat1);
    if (nl0 != nl1) return false;
    int nl = nl0;
    if (nl < 1) nl = 1;

    // Longitude
    double dLon0 = 360.0 / nl;
    double dLon1 = 360.0 / (nl-1 > 0 ? nl-1 : 1);

    int m = floor( (lonEven*(nl-1) - lonOdd*nl) / 131072.0 + 0.5 );
    double rlon0 = dLon0 * (fmod(m, nl) + lonEven/131072.0);
    double rlon1 = dLon1 * (fmod(m, nl-1>0?nl-1:1) + lonOdd /131072.0);

    // Возьмём вариант even (или odd) — обе должны быть консистентны
    outLat = rlat0;
    outLon = rlon0;
    if (outLon > 180) outLon -= 360.0;
    return true;
}

// Локальная декодировка (по реф. точке приёмника)
static double g_refLat = 0.0, g_refLon = 0.0;
void adsb_set_local_ref(float refLatDeg, float refLonDeg){
    g_refLat = refLatDeg; g_refLon = refLonDeg;
}
static bool cprLocal(int latCpr, int lonCpr, bool odd, double &outLat, double &outLon){
    double dlat = cprDlat(odd);
    int j = floor(g_refLat / dlat) + floor(0.5 + fmod(g_refLat, dlat)/dlat) - floor(latCpr/131072.0);
    double rlat = dlat * (j + latCpr/131072.0);
    if (rlat >= 270) rlat -= 360;
    int nl = cprNL(rlat);
    if (nl < 1) nl = 1;
    double dlon = 360.0 / (odd ? nl-1 : nl);
    int m = floor(g_refLon / dlon) + floor(0.5 + fmod(g_refLon, dlon)/dlon) - floor(lonCpr/131072.0);
    double rlon = dlon * (m + lonCpr/131072.0);
    if (rlon > 180) rlon -= 360.0;

    outLat = rlat; outLon = rlon;
    return true;
}

// Кэш последнего even/odd на ICAO (для глобальной CPR)
struct CprCache {
    uint32_t icao;
    uint32_t tEvenMs;
    uint32_t tOddMs;
    int latEven, lonEven;
    int latOdd,  lonOdd;
    bool used;
};
static CprCache g_cpr[32]; // простой LRU по индексу

static CprCache* cprGet(uint32_t icao){
    uint8_t idx = icao ^ (icao>>7) ^ (icao>>14);
    idx &= (uint8_t)(32-1);
    if (!g_cpr[idx].used || g_cpr[idx].icao != icao){
        g_cpr[idx] = {icao,0,0,0,0,0,0,true};
    }
    return &g_cpr[idx];
}

// =============== Декод высоты (TC 9..18) ================
static bool decode_adsb_altitude(uint16_t alt12, int &alt_ft){
    // Q-бит = bit4 в младшей пятёрке (LSB nibble+1), как в Mode S altitude
    bool Q = (alt12 & 0x0010) != 0;
    if (Q) {
        // Q=1: простая формула 25 ft
        int N = ((alt12 & 0x0FE0) >> 1) | (alt12 & 0x000F);
        alt_ft = N * 25 - 1000;
        return true;
    } else {
        // Q=0: Gillham AC13 реконструкция; соберём псевдо-AC13 (добавим D-бит как 0)
        uint16_t ac13 = (alt12 & 0x0FFF); // 12 бит
        int alt;
        if (gillham_alt_from_ac13(ac13, alt)) { alt_ft = alt; return true; }
        return false;
    }
}

// =============== Squawk из DF5 или BDS 1,0 (DF20/21) ================
static bool try_decode_squawk_DF5(const AdsbPacket &pkt, int &squawk){
    // DF = first 5 bits (bits 0..4 of message)
    uint8_t DF = (uint8_t)get_bits_be(pkt.raw, 0, 5);
    if (DF != 5) return false; // Surveillance, identity
    // AC field (identity) bits 19..31 (13 bits) у DF5
    uint16_t id13 = (uint16_t)get_bits_be(pkt.raw, 19, 13);
    return squawk_from_identity_13b(id13, squawk);
}

static bool try_decode_squawk_BDS10(const AdsbPacket &pkt, int &squawk){
    // DF20/21, BDS 1,0 Identity в ME (7 байт). Тут предполагаем, что сообщение действительно BDS1,0.
    uint8_t df = (uint8_t)get_bits_be(pkt.raw,0,5);
    if (df != 20 && df != 21) return false;
    // identity 13 bits часто в старших битах ME; используем распространённый мэппинг:
    // возьмём 13 бит начиная с ME bit 13 (эвристика; при несовпадении — не декодируется)
    uint16_t id13 = (uint16_t)get_bits_be(pkt.raw, 33+13, 13);
    return squawk_from_identity_13b(id13, squawk);
}

// =================== Основной декодер ===================
bool adsb_decode(const AdsbPacket &pkt, AdsbFrame &f) {
    memset(&f, 0, sizeof(f));
    f.squawk = -1;
    f.aircraft_type = 9;
    f.signal_source = 1;
    f.rssi = (int8_t)pkt.rssi;
    f.seen_time = millis();
    f.timestamp = millis();

    uint8_t DF = (uint8_t)get_bits_be(pkt.raw, 0, 5);
    f.addr = (uint32_t)get_bits_be(pkt.raw, 8, 24);

    // Попытка достать Squawk из DF5/DF20/21 (если попадётся)
    int sq;
    if (try_decode_squawk_DF5(pkt, sq) || try_decode_squawk_BDS10(pkt, sq)) {
        f.squawk = sq;
    }

    // DF17/18: ADS-B extended squitter
    if (DF == 17 || DF == 18) {
        uint8_t TC = (uint8_t)get_bits_be(pkt.raw, 32+0, 5); // ME bits 0..4
        // Airborne position TC=9..18; Surface position TC=5..8
        if ((TC >= 9 && TC <= 18) || (TC >= 5 && TC <= 8)) {
            uint16_t alt12 = (uint16_t)get_bits_be(pkt.raw, 32+8, 12);
            bool timeFlag = get_bit_be(pkt.raw, 32+20);
            bool odd = get_bit_be(pkt.raw, 32+21);
            int latCpr = (int)get_bits_be(pkt.raw, 32+22, 17);
            int lonCpr = (int)get_bits_be(pkt.raw, 32+39, 17);

            int alt_ft;
            if (decode_adsb_altitude(alt12, alt_ft)) {
                f.alt = f.pressure_alt = (float)alt_ft;
            }

            // CPR decode
            CprCache *c = cprGet(f.addr);
            uint32_t nowms = millis();
            if (!odd) { c->latEven = latCpr; c->lonEven = lonCpr; c->tEvenMs = nowms; }
            else      { c->latOdd  = latCpr; c->lonOdd  = lonCpr; c->tOddMs  = nowms; }

            bool haveGlobal = false;
            if (c->tEvenMs && c->tOddMs && (abs((int)(c->tEvenMs - c->tOddMs)) <= 10000)) {
                double lat, lon;
                if (cprGlobal(c->latEven, c->lonEven, c->latOdd, c->lonOdd, lat, lon)) {
                    f.lat = lat; f.lon = lon; haveGlobal = true;
                }
            }
            if (!haveGlobal) {
                // Локальный декодинг по реф. точке
                double lat, lon;
                if (cprLocal(odd ? latCpr : latCpr, odd ? lonCpr : lonCpr, odd, lat, lon)) {
                    f.lat = lat; f.lon = lon;
                }
            }
        }
        // TC=19 скорости/курс/VS — предполагается у вас уже реализовано отдельно; можно добавить позже.
    }

    // Если координаты не были декодированы (0,0) — не считаем это ошибкой
    // Полётный номер (flight) и прочие поля можно заполнить из TC=1..4, если они встречались ранее.

    // По умолчанию что-то осмысленное:
    if (f.flight[0] == 0) snprintf(f.flight, sizeof(f.flight), "N/A");

    return true;
}
//========================================================
Как это работает

CPR: кешируем последнюю пару even/odd для каждого ICAO и, если разрыв ≤10 с, выполняем global CPR; иначе — local CPR по опорной точке (задаётся adsb_set_local_ref()).

Высота:

при Q=1 — формула 25 ft шаг;

при Q=0 — задействован Gillham AC13 декодер (обобщённый каркас).

Squawk:

из DF5 (Surveillance, Identity),

либо эвристически из DF20/DF21 (BDS 1,0) в ME (если такое сообщение попало).

Если понадобится, можем точно подогнать битовые смещения для BDS 1,0 под ваш источник (разные транспондеры встречаются).

void hex2bytes(const char *hex, uint8_t *out, int len) {
  for (int i=0; i<len; i++) {
    char c1 = hex[i*2], c2 = hex[i*2+1];

#include "adsb_gillham.h"

// Gillham gray-code to altitude (ft)
// Covers -1000 to 126000 ft. Returns -99999 if invalid.
int gillham2alt(uint16_t ac13) {
    // Extract bits (D1..A1, M, Q, etc.)
    int g[13];
    for (int i = 0; i < 13; i++) {
        g[i] = (ac13 >> i) & 1;
    }

    int qbit = g[4];  // Q-bit = bit 4
    if (qbit == 1) {
        // Q=1 → 25ft resolution (Mode S normal)
        int n = ((ac13 >> 5) & 0xFF) << 4 | ((ac13 & 0xF0) >> 1) | (ac13 & 0x1);
        return n * 25 - 1000;
    }

    // Q=0 → classic Gillham Gray-code (AC13)
    // Decode as per ICAO Annex 10, Table 3.7.2.6.3
    // This is simplified: mapping 4096 codes -> altitude in 100ft steps
    // Here we approximate: return steps of 100ft
    // TODO: complete with full Gray-to-binary mapping table for strict decoding

    // For now: dummy linear approximation
    int alt100 = (ac13 & 0xFFF);  // just treat as binary
    return alt100 * 100 - 1000;
}

src/adsb_gillham.cpp

#include "adsb_gillham.h"

// Таблица на 2048 кодов (11 бит D2..C4). Заполняется в adsb_gillham_init()
static int16_t g_table[2048];
static bool g_inited = false;

// Gray→Binary (8-битный, достаточно для 500-ft компоненты)
static inline uint8_t gray8_to_bin(uint8_t g) {
    g ^= g >> 1;
    g ^= g >> 2;
    g ^= g >> 4;
    return g;
}

// Декод 11-битного Gillham (D2..C4)
int16_t gillham_decode_11(uint16_t g11) {
    // Биты в порядке: D2 D4 A1 A2 A4 B1 B2 B4 C1 C2 C4  (MSB→LSB)
    // Индексация в нашем g11: bit10..bit0
    uint8_t D2 = (g11 >> 10) & 1;
    uint8_t D4 = (g11 >>  9) & 1;
    uint8_t A1 = (g11 >>  8) & 1;
    uint8_t A2 = (g11 >>  7) & 1;
    uint8_t A4 = (g11 >>  6) & 1;
    uint8_t B1 = (g11 >>  5) & 1;
    uint8_t B2 = (g11 >>  4) & 1;
    uint8_t B4 = (g11 >>  3) & 1;
    uint8_t C1 = (g11 >>  2) & 1;
    uint8_t C2 = (g11 >>  1) & 1;
    uint8_t C4 = (g11 >>  0) & 1;

    // 500-ft компонент: стандартный 8-битный reflected Gray из D2..B4
    uint8_t gray500 = (uint8_t)( (D2<<7) | (D4<<6) | (A1<<5) | (A2<<4)
                               | (A4<<3) | (B1<<2) | (B2<<1) |  B4 );
    uint8_t n500 = gray8_to_bin(gray500); // 0..255

    // 100-ft компонент кодируется C1/C2/C4 по "зеркальной 5-состояний" схеме:
    // допустимы только коды: 001,011,010,110,100. Остальные → invalid.
    uint8_t c = (uint8_t)((C1<<2) | (C2<<1) | C4);
    int16_t off = 0;
    bool evenParity = (__builtin_parity((unsigned)n500) == 0); // true если чётная

    // Карта для чётного и нечётного n500 (см. ICAO / Wikipedia)
    // even: 001:-200, 011:-100, 010:0, 110:+100, 100:+200
    //  odd: 100:-200, 110:-100, 010:0, 011:+100, 001:+200
    switch (c) {
        case 0b010: off = 0; break;
        case 0b001: off = evenParity ? -200 : +200; break;
        case 0b011: off = evenParity ? -100 : +100; break;
        case 0b110: off = evenParity ? +100 : -100; break;
        case 0b100: off = evenParity ? +200 : -200; break;
        default:    return GILLHAM_INVALID_ALT; // 000,101,111 → не используется
    }

    // База −1000 ft + 500*n500 + off
    // Диапазон по стандарту: примерно −1200..126700 ft
    int32_t alt = -1000 + (int32_t)n500 * 500 + off;

    // Немного здравого смысла: отсечём нереальные выходы за 126700/−1200
    if (alt < -1200 || alt > 126700) return GILLHAM_INVALID_ALT;
    return (int16_t)alt;
}

// Построение полной таблицы
void adsb_gillham_init() {
    if (g_inited) return;
    for (uint16_t code = 0; code < 2048; ++code) {
        g_table[code] = gillham_decode_11(code);
    }
    g_inited = true;
}

// Восстановление 11-битного Gillham из Mode-S AC12 при Q=0.
// В AC12 (MSB..LSB = b11..b0) биты распределены как Gillham с "вставленными" M и Q.
// Стандартная укладка (см. 1090-MHz Riddle / Annex 10):
//  b11 D2, b10 D4, b9  A1, b8  A2, b7  A4, b6  B1, b5  B2, b4  B4,
//  b3  M,  b2  C1, b1  C2, b0  C4   (а Q — не в AC12, он извлекается заранее)
// Для Q=0 берём D2..B4, C1/C2/C4 как есть, M игнорируем.
uint16_t modes_ac12_to_g11(uint16_t ac12) {
    // Защитимся от "мусора" в старших битах
    ac12 &= 0x0FFF;

    // Собираем 11 бит D2..C4 в нашем порядке (MSB..LSB)
    uint16_t g11 = 0;
    // D2..B4
    g11 |= ((ac12 >> 11) & 1) << 10; // D2
    g11 |= ((ac12 >> 10) & 1) <<  9; // D4
    g11 |= ((ac12 >>  9) & 1) <<  8; // A1
    g11 |= ((ac12 >>  8) & 1) <<  7; // A2
    g11 |= ((ac12 >>  7) & 1) <<  6; // A4
    g11 |= ((ac12 >>  6) & 1) <<  5; // B1
    g11 |= ((ac12 >>  5) & 1) <<  4; // B2
    g11 |= ((ac12 >>  4) & 1) <<  3; // B4
    // b3 = M (пропускаем)
    g11 |= ((ac12 >>  2) & 1) <<  2; // C1
    g11 |= ((ac12 >>  1) & 1) <<  1; // C2
    g11 |= ((ac12 >>  0) & 1) <<  0; // C4

    return g11 & 0x07FF;
}

// Полный декодер Mode-S AC12 с учётом Q
int32_t modes_altitude_from_ac12(uint16_t ac12, bool q_bit) {
    if (q_bit) {
        // Q=1: 25-ft шаг
        // Формула (Annex 10): ( ( (ac12 & 0x0FE0) << 3 ) | (ac12 & 0x001F) ) даёт N
        // Но на практике N уже выделяется при парсинге; если нет — используем компактный путь:
        // Соберём 11-битное число N из AC12, пропустив M и Q.
        uint16_t N = 0;
        // Берём биты: b11 b10 b9 b8 b7 b6 b5 b4  b2 b1 b0 (пропуская b3(M) и b? (Q уже дан))
        N |= ((ac12 >> 11) & 1) << 10;
        N |= ((ac12 >> 10) & 1) <<  9;
        N |= ((ac12 >>  9) & 1) <<  8;
        N |= ((ac12 >>  8) & 1) <<  7;
        N |= ((ac12 >>  7) & 1) <<  6;
        N |= ((ac12 >>  6) & 1) <<  5;
        N |= ((ac12 >>  5) & 1) <<  4;
        N |= ((ac12 >>  4) & 1) <<  3;
        N |= ((ac12 >>  2) & 1) <<  2;
        N |= ((ac12 >>  1) & 1) <<  1;
        N |= ((ac12 >>  0) & 1) <<  0;
        return (int32_t)N * 25 - 1000;
    } else {
        // Q=0: полноценный Gillham
        if (!g_inited) adsb_gillham_init();
        uint16_t g11 = modes_ac12_to_g11(ac12);
        if (g11 == 0xFFFF) return GILLHAM_INVALID_ALT;
        return g_table[g11];
    }
}

// --- новая функция обратного кодирования ---
uint16_t gillham_encode_11(int32_t altitude_ft) {
    if (altitude_ft < -1000 || altitude_ft > 126700) return 0xFFFF;

    // Найдём ближайший 500-футовый шаг
    int32_t base = (altitude_ft + 1000) / 500;
    if (base < 0) base = 0;
    if (base > 255) base = 255;

    int32_t rem = altitude_ft - (-1000 + base*500);
    // rem ∈ {−200,−100,0,+100,+200}
    int16_t off = 0;
    switch (rem) {
        case -200: off=-200; break;
        case -100: off=-100; break;
        case 0: off=0; break;
        case 100: off=100; break;
        case 200: off=200; break;
        default: return 0xFFFF;
    }

    // Кодируем 500-ft часть в Gray
    uint8_t gray500 = bin_to_gray8((uint8_t)base);

    // Определим parity
    bool evenParity = (__builtin_parity((unsigned)base) == 0);

    // Кодируем offset в C1/C2/C4
    uint8_t c = 0b010; // default = 0
    if (off == -200) c = evenParity ? 0b001 : 0b100;
    if (off == -100) c = evenParity ? 0b011 : 0b110;
    if (off == +100) c = evenParity ? 0b110 : 0b011;
    if (off == +200) c = evenParity ? 0b100 : 0b001;

    // Собираем биты в 11-битный Gillham (MSB..LSB = D2..C4)
    uint16_t g11 = 0;
    g11 |= ((gray500 >> 7) & 1) << 10; // D2
    g11 |= ((gray500 >> 6) & 1) <<  9; // D4
    g11 |= ((gray500 >> 5) & 1) <<  8; // A1
    g11 |= ((gray500 >> 4) & 1) <<  7; // A2
    g11 |= ((gray500 >> 3) & 1) <<  6; // A4
    g11 |= ((gray500 >> 2) & 1) <<  5; // B1
    g11 |= ((gray500 >> 1) & 1) <<  4; // B2
    g11 |= ((gray500 >> 0) & 1) <<  3; // B4
    g11 |= ((c >> 2) & 1) <<  2;       // C1
    g11 |= ((c >> 1) & 1) <<  1;       // C2
    g11 |= ((c >> 0) & 1) <<  0;       // C4

    return g11;
}

// --- функции сохранения / загрузки ---
bool gillham_save_table(const char *path) {
    if (!g_inited) adsb_gillham_init();
    File f = LittleFS.open(path, "w");
    if (!f) return false;
    size_t n = f.write((uint8_t*)g_table, sizeof(g_table));
    f.close();
    return n == sizeof(g_table);
}

bool gillham_load_table(const char *path) {
    File f = LittleFS.open(path, "r");
    if (!f) return false;
    size_t n = f.read((uint8_t*)g_table, sizeof(g_table));
    f.close();
    if (n == sizeof(g_table)) {
        g_inited = true;
        return true;
    }
    return false;
}
