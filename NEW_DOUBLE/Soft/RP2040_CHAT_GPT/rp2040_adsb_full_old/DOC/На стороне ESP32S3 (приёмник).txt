#include <Arduino.h>
#include "adsb_packet.h"

#define UART_RX_PIN 42
#define UART_TX_PIN 41
#define UART_BAUD   115200

void setup() {
    Serial.begin(115200);
    Serial1.begin(UART_BAUD, SERIAL_8N1, UART_RX_PIN, UART_TX_PIN);
    Serial.println("ESP32S3 Receiver ready");
}

void loop() {
    static ToDUMP1090 pkt;
    if (Serial1.available() >= (int)sizeof(pkt)) {
        Serial1.readBytes((char*)&pkt, sizeof(pkt));
        if (pkt.endOfPacket[0] == (char)0xFF &&
            pkt.endOfPacket[1] == (char)0xFF &&
            pkt.endOfPacket[2] == (char)0xFF) {
            Serial.printf("Recv ICAO: %06X Flight:%s Alt:%d Speed:%d Lat:%.5f Lon:%.5f\n",
                          pkt.addr, pkt.flight, pkt.altitude, pkt.speed,
                          pkt.lat, pkt.lon);
        }
    }
}

ESP32S3: приём реальных пакетов
#include <Arduino.h>
#include "adsb_packet.h"

#define UART_RX_PIN 42
#define UART_TX_PIN 41
#define UART_BAUD   115200

void setup() {
    Serial.begin(115200);
    Serial1.begin(UART_BAUD, SERIAL_8N1, UART_RX_PIN, UART_TX_PIN);
    Serial.println("ESP32S3 ADS-B Receiver ready");
}

void loop() {
    static ToDUMP1090 pkt;
    if (Serial1.available() >= (int)sizeof(pkt)) {
        Serial1.readBytes((char*)&pkt, sizeof(pkt));
        if (pkt.endOfPacket[0] == (char)0xFF &&
            pkt.endOfPacket[1] == (char)0xFF &&
            pkt.endOfPacket[2] == (char)0xFF) {
            Serial.printf("ICAO:%06X  Flight:%s  Squawk:%04d  Alt:%dft  Spd:%dkn  Trk:%d  Vr:%d\n",
                          pkt.addr, pkt.flight, pkt.squawk, pkt.altitude,
                          pkt.speed, pkt.track, pkt.vert_rate);
            Serial.printf("Lat:%.6f  Lon:%.6f  Seen:%ds\n\n",
                          pkt.lat, pkt.lon, pkt.seen_time);
        }
    }
}
